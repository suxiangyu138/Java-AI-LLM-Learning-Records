## 1. 文件上传

```javascript
// 单文件上传
const formData = new FormData();
formData.append('file', file);              // file 来自 <input type="file">
formData.append('description', '头像');

await axios.post('/api/upload', formData, {
  headers: { 'Content-Type': 'multipart/form-data' },
  onUploadProgress: (e) => {
    const percent = Math.round((e.loaded / e.total) * 100);
    console.log(`上传进度: ${percent}%`);
  }
});

// 分片上传（大文件）
async function uploadChunks(file) {
  const CHUNK_SIZE = 5 * 1024 * 1024;  // 5MB
  const md5 = await calculateMD5(file);

  // 1. 检查秒传
  const { uploaded } = await api.get('/upload/check', { params: { md5 } });
  if (uploaded) return;  // 秒传！

  // 2. 上传分片
  const chunks = Math.ceil(file.size / CHUNK_SIZE);
  for (let i = 0; i < chunks; i++) {
    const chunk = file.slice(i * CHUNK_SIZE, (i + 1) * CHUNK_SIZE);
    const formData = new FormData();
    formData.append('file', chunk);
    formData.append('md5', md5);
    formData.append('index', i);
    await api.post('/upload/chunk', formData);
  }

  // 3. 合并分片
  await api.post('/upload/merge', { md5, filename: file.name });
}
```

## 2. 请求取消

```javascript
// ⭐ 使用 AbortController（现代浏览器）
const controller = new AbortController();

// 5 秒超时自动取消
const timer = setTimeout(() => controller.abort(), 5000);

try {
  const { data } = await axios.get('/api/users', {
    signal: controller.signal
  });
  clearTimeout(timer);
} catch (err) {
  if (axios.isCancel(err)) {
    console.log('请求已取消');
  }
}

// 搜索防抖 — 用户输入停止 300ms 后再发请求
let controller = null;
input.addEventListener('input', async (e) => {
  controller?.abort();              // 取消上一次请求
  controller = new AbortController();
  try {
    const { data } = await axios.get('/api/search', {
      params: { q: e.target.value },
      signal: controller.signal
    });
    renderResults(data);
  } catch (err) {
    if (!axios.isCancel(err)) console.error(err);
  }
});
```

## 3. Loading 与防重复提交

```javascript
// ⭐ 通用请求 Loading Hook
let requestCount = 0;
const loading = ref(false);

function showLoading() { loading.value = true; }
function hideLoading() {
  requestCount--;
  if (requestCount <= 0) loading.value = false;
}

// 拦截器中统一管理
api.interceptors.request.use(config => {
  if (!config.hideLoading) { requestCount++; showLoading(); }
  return config;
});

api.interceptors.response.use(
  response => { hideLoading(); return response; },
  error   => { hideLoading(); return Promise.reject(error); }
);

// 单个请求跳过 loading
await api.get('/api/config', { hideLoading: true });
```

### 防重复提交

```javascript
// 前端按钮防抖 — 方法1
let submitting = false;
async function submit() {
  if (submitting) return;
  submitting = true;
  try {
    await api.post('/api/orders', formData);
    message.success('提交成功');
  } finally {
    submitting = false;
  }
}

// 方法2 — 后端配合：接口返回唯一 token，提交时校验
```

## 4. 错误重试

```javascript
// 拦截器自动重试
api.interceptors.response.use(null, async (error) => {
  const config = error.config;
  config.__retryCount = config.__retryCount || 0;

  // 仅对 GET 和 5xx 重试，最多 3 次
  if (config.method === 'get'
      && error.response?.status >= 500
      && config.__retryCount < 3) {
    config.__retryCount++;
    await new Promise(resolve => setTimeout(resolve, 1000 * config.__retryCount)); // 退避
    return api(config);    // 重试
  }
  return Promise.reject(error);
});
```

```javascript
// ⭐ 封装通用请求方法
async function request(config) {
  const { data } = await api(config);
  return data;
}

// 使用 — 极简
const user = await request({ url: '/users/1' });
```

> 🎯 **后端看完能懂**：前端请求模式 = 上传用 FormData + onUploadProgress、取消用 AbortController、Loading 用拦截器计数、防抖用 abort + 新请求覆盖旧请求。

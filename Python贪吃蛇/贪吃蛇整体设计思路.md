## 整体设计思路

这个贪吃蛇游戏是一个纯静态 HTML 单文件应用，技术架构分三层：**渲染层（Canvas 2D）**、**逻辑层（状态机）**、**交互层（键盘/触摸/D-Pad）**。

***

## 状态机设计

游戏用一个 `state` 对象统一管理所有状态，`phase` 字段驱动整个流程：

```
idle → playing → paused → playing → dead → idle
```

```js
function initState() {
  const cx = Math.floor(COLS/2), cy = Math.floor(ROWS/2);
  return {
    phase: 'idle',       // 状态机：idle | playing | paused | dead
    snake: [{x:cx,y:cy},{x:cx-1,y:cy},{x:cx-2,y:cy}],  // 蛇身数组，头部是[0]
    dir: 'RIGHT',        // 当前实际方向
    nextDir: 'RIGHT',    // 下一帧将生效的方向（防止同帧连续按键引发穿越）
    food: null,
    bonus: null,         // 随机金星 { x, y, value, 倒计时 }
    bonusExpire: 0,
    score: 0,
    level: 1,
    tick: 0,
  };
}
```

**为什么要 `nextDir`？** 每帧 tick 时才真正切换方向，防止玩家在同一 tick 内连按两次（例如先按下再按左）导致蛇直接 180° 掉头穿越自身。

## 核心 tick 逻辑

游戏循环用 `setInterval` 驱动（而非 rAF），这样速度可以精确控制：

```js
function tick() {
  state.dir = state.nextDir;             // 本帧确认方向
  const d   = DIR[state.dir];            // 取方向向量 [dx, dy]
  const head = state.snake[0];
  const nx = head.x + d[0];
  const ny = head.y + d[1];

  // 1. 碰墙检测
  if (nx < 0 || nx >= COLS || ny < 0 || ny >= ROWS) { die(); return; }

  // 2. 碰自身检测（跳过最后一节尾巴，它即将消失）
  for (let i = 0; i < state.snake.length - 1; i++) {
    if (state.snake[i].x === nx && state.snake[i].y === ny) { die(); return; }
  }

  // 3. 移动：头部插入新坐标
  state.snake.unshift({ x: nx, y: ny });

  // 4. 吃食物/金星 → grew=true 则不弹出尾巴（蛇变长）
  let grew = false;
  if (state.food && nx === state.food.x && ny === state.food.y) {
    grew = true;
    state.score += 1;
    spawnFood();   // 重新随机食物
    spawnBonus();  // 概率刷新金星
  }
  if (state.bonus && nx === state.bonus.x && ny === state.bonus.y) {
    grew = true;
    state.score += state.bonus.value;  // +3/+5/+8
    state.bonus = null;
  }

  // 5. 没吃东西 → 弹出尾巴（保持长度）
  if (!grew) state.snake.pop();
}
```

**移动的本质**：每帧在头部 `unshift` 一个新格，在尾部 `pop` 一个旧格 —— 蛇就"动"了；吃到食物时不 pop，蛇就"长"了。这是经典的双端队列技巧。

## 等级与速度系统

```js
const BASE_SPEED  = 130;   // 初始每帧间隔 ms
const SPEED_SCALE = 0.88;  // 每升一级乘以 0.88（越小越快）
const LEVEL_EVERY = 5;     // 每 5 分升一级

function getSpeed() {
  // 指数衰减：130 → 114 → 100 → 88 → 78...
  return Math.round(BASE_SPEED * Math.pow(SPEED_SCALE, state.level - 1));
}
```

升级时清除旧定时器、用新速度重建：

```js
const newLevel = Math.floor(state.score / LEVEL_EVERY) + 1;
if (newLevel !== state.level) {
  state.level = newLevel;
  clearInterval(tickTimer);
  scheduleTickTimer();  // 重新 setInterval(tick, getSpeed())
}
```

### 圆角矩形工具函数

Canvas 没有原生 `roundRect`（旧版），手写一个：

```js
function roundRect(ctx, x, y, w, h, r) {
  r = Math.min(r, w/2, h/2);  // 防止 r 过大变形
  ctx.beginPath();
  ctx.moveTo(x + r, y);
  ctx.lineTo(x + w - r, y);
  ctx.arcTo(x + w, y,     x + w, y + r,   r);
  ctx.lineTo(x + w, y + h - r);
  ctx.arcTo(x + w, y + h, x + w - r, y + h, r);
  ctx.lineTo(x + r, y + h);
  ctx.arcTo(x,     y + h, x,     y + h - r, r);
  ctx.lineTo(x, y + r);
  ctx.arcTo(x,     y,     x + r, y,         r);
  ctx.closePath();
}
```

## 粒子爆炸效果

吃食物时触发 CSS 粒子（不用 Canvas，利用 CSS animation 减轻主线程压力）：

```js
function emitParticles(x, y, color, count) {
  for (let i = 0; i < count; i++) {
    const el = document.createElement('div');
    el.className = 'particle';

    const angle = (Math.PI * 2 * i / count) + Math.random() * 0.3;
    const speed = 20 + Math.random() * 45;
    const dur   = 0.4 + Math.random() * 0.5;

    // 用 CSS 自定义属性传参给 @keyframes
    el.style.cssText = `
      left:${x}px; top:${y}px;
      width:${size}px; height:${size}px;
      background:${color};
      --tx:${Math.cos(angle)*speed}px;   /* X 方向位移 */
      --ty:${Math.sin(angle)*speed}px;   /* Y 方向位移 */
      --dur:${dur}s;
    `;
    container.appendChild(el);
    setTimeout(() => el.remove(), dur * 1000 + 50);  // 自动回收 DOM
  }
}
```

对应 CSS：

```css
@keyframes particleFly {
  0%   { opacity: 1; transform: translate(0,0) scale(1); }
  100% { opacity: 0; transform: translate(var(--tx), var(--ty)) scale(0); }
}
```

**移动端滑动手势**：

```js
canvas.addEventListener('touchstart', e => {
  _touchStart = { x: e.touches[0].clientX, y: e.touches[0].clientY };
});
canvas.addEventListener('touchend', e => {
  const dx = e.changedTouches[0].clientX - _touchStart.x;
  const dy = e.changedTouches[0].clientY - _touchStart.y;
  // 取绝对值更大的轴作为方向
  if (Math.abs(dx) > Math.abs(dy))
    state.nextDir = dx > 0 ? 'RIGHT' : 'LEFT';
  else
    state.nextDir = dy > 0 ? 'DOWN' : 'UP';
});
```

<script>
const GRID=20, BASE_SPEED=130, SPEED_SCALE=0.88, LEVEL_EVERY=5;
const BONUS_CHANCE=0.35, BONUS_DURATION=6000;
const LEVELS=['慢速','普通','快速','极快','疯狂'];
const canvas=document.getElementById('gameCanvas'), ctx=canvas.getContext('2d'), wrap=document.getElementById('canvas-wrap');
let COLS,ROWS,TILE;

function sizeCanvas(){
  const vw=Math.min(window.innerWidth-32,600), vh=Math.min(window.innerHeight-170,560);
  TILE=Math.max(14,Math.floor(Math.min(vw,vh)/GRID));
  COLS=Math.floor(vw/TILE); ROWS=Math.floor(vh/TILE);
  canvas.width=COLS*TILE; canvas.height=ROWS*TILE;
  wrap.style.width=canvas.width+'px'; wrap.style.height=canvas.height+'px';
}

window.addEventListener('resize',()=>{sizeCanvas();if(state.phase==='playing'||state.phase==='paused')render();});

const DIR={UP:[0,-1],DOWN:[0,1],LEFT:[-1,0],RIGHT:[1,0]};
const OPPOSITE={UP:'DOWN',DOWN:'UP',LEFT:'RIGHT',RIGHT:'LEFT'};
let state,tickTimer,bestScore=0;

function initState(){
  const cx=Math.floor(COLS/2),cy=Math.floor(ROWS/2);
  return{phase:'idle',snake:[{x:cx,y:cy},{x:cx-1,y:cy},{x:cx-2,y:cy}],dir:'RIGHT',nextDir:'RIGHT',food:null,bonus:null,bonusExpire:0,score:0,level:1,tick:0};
}

function randomCell(exclude=[]){
  let x,y,tries=0;
  do{x=Math.floor(Math.random()*COLS);y=Math.floor(Math.random()*ROWS);tries++; if(tries>2000)return null;}
  while(exclude.some(c=>c.x===x&&c.y===y));
  return{x,y};
}
function spawnFood(){ state.food=randomCell(state.snake); }
function spawnBonus(){
  if(Math.random()<BONUS_CHANCE){
    const pos=randomCell([...state.snake,state.food]);
    if(pos){const v=[3,5,8][Math.floor(Math.random()*3)]; state.bonus={...pos,value:v}; state.bonusExpire=performance.now()+BONUS_DURATION;}
  }
}

function tick(){
  if(state.phase!=='playing')return;
  state.dir=state.nextDir;
  const d=DIR[state.dir],head=state.snake[0];

  if(state.bonus&&performance.now()>state.bonusExpire)state.bonus=null;
  if(!grew)state.snake.pop();
  const newLevel=Math.floor(state.score/LEVEL_EVERY)+1;
  if(newLevel!==state.level){state.level=newLevel;clearInterval(tickTimer);scheduleTickTimer();}
  if(state.score>bestScore){bestScore=state.score;document.getElementById('best-display').textContent=bestScore;}
  updateHUD(); render();
}

## 设计思路总结

| 模块 | 关键决策 | 原因 |
|------|---------|------|
| **游戏循环** | `setInterval` 控制逻辑，`rAF` 控制渲染 | 逻辑速度可变、渲染流畅不耦合 |
| **移动算法** | 数组头插 + 尾删 | O(n) 但足够简单，蛇长一般不超过几百节 |
| **方向防穿** | `nextDir` 缓冲机制 | 同帧连按不引发反向穿越死亡 |
| **粒子效果** | CSS 自定义属性 + animation | 不占用 Canvas 绘制资源，GPU 加速 |
| **视觉动画** | `Date.now()` 时间驱动 | 帧率无关，慢设备也不会加速/变形 |
| **响应式** | 动态计算 TILE + COLS + ROWS | 任意分辨率都完整显示，无变形 |
| **移动端** | D-Pad + 滑动手势双保险 | 覆盖不同用户习惯 |

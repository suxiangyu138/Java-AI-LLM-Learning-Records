## 2. FastAPI 服务封装

### 2.1 标准 AI 服务模板

```python
# app.py - 标准 AI 模型服务
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel, Field
from typing import Optional, List
import time
import asyncio
from contextlib import asynccontextmanager

# ========== 请求/响应模型 ==========
class ChatRequest(BaseModel):
    messages: List[dict] = Field(..., description="对话消息列表")
    temperature: Optional[float] = Field(0.7, ge=0.0, le=2.0)
    max_tokens: Optional[int] = Field(512, ge=1, le=4096)
    stream: Optional[bool] = Field(False)

class ChatResponse(BaseModel):
    id: str
    content: str
    model: str
    usage: dict
    latency_ms: float

# ========== 模型管理器 ==========
class ModelManager:
    """单例模式管理模型加载"""

    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            # 延迟加载（在 startup 事件中初始化）
            cls._instance.model = None
            cls._instance.tokenizer = None
        return cls._instance

    def load_model(self):
        """加载模型（启动时调用一次）"""
        from transformers import AutoModelForCausalLM, AutoTokenizer
        import torch

        self.model = AutoModelForCausalLM.from_pretrained(
            "Qwen/Qwen2.5-7B-Instruct",
            torch_dtype=torch.float16,
            device_map="auto"
        )
        self.tokenizer = AutoTokenizer.from_pretrained("Qwen/Qwen2.5-7B-Instruct")
        print("✅ 模型加载完成")

# ========== 生命周期管理 ==========
model_manager = ModelManager()

@asynccontextmanager
async def lifespan(app: FastAPI):
    """应用启动和关闭时的生命周期管理"""
    # 启动时：加载模型
    model_manager.load_model()
    yield
    # 关闭时：释放资源
    del model_manager.model
    torch.cuda.empty_cache()

app = FastAPI(title="AI Model Service", version="1.0.0", lifespan=lifespan)

# CORS 配置
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

# ========== API 端点 ==========
@app.post("/v1/chat", response_model=ChatResponse)
async def chat(request: ChatRequest):
    """对话接口"""
    start_time = time.time()

    try:
        # 构造输入
        text = model_manager.tokenizer.apply_chat_template(
            request.messages,
            tokenize=False
        )
        inputs = model_manager.tokenizer(text, return_tensors="pt").to("cuda")

        # 生成
        outputs = model_manager.model.generate(
            **inputs,
            max_new_tokens=request.max_tokens,
            temperature=request.temperature,
            do_sample=True
        )

        response_text = model_manager.tokenizer.decode(
            outputs[0][inputs.input_ids.shape[1]:],
            skip_special_tokens=True
        )

        latency = (time.time() - start_time) * 1000

        return ChatResponse(
            id=f"chatcmpl-{int(time.time())}",
            content=response_text,
            model="qwen-7b-instruct",
            usage={
                "prompt_tokens": inputs.input_ids.shape[1],
                "completion_tokens": len(outputs[0]) - inputs.input_ids.shape[1],
                "total_tokens": len(outputs[0])
            },
            latency_ms=latency
        )

    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


@app.get("/health")
async def health():
    """健康检查"""
    return {
        "status": "healthy",
        "model_loaded": model_manager.model is not None,
        "gpu_available": torch.cuda.is_available()
    }


@app.get("/metrics")
async def metrics():
    """Prometheus 指标端点"""
    return {
        "gpu_memory_used_mb": torch.cuda.memory_allocated() / 1024**2,
        "gpu_memory_total_mb": torch.cuda.get_device_properties(0).total_memory / 1024**2,
    }
```

### 2.2 流式输出（SSE）

```python
from fastapi.responses import StreamingResponse
import json

@app.post("/v1/chat/stream")
async def chat_stream(request: ChatRequest):
    """流式对话接口（Server-Sent Events）"""

    async def generate():
        inputs = model_manager.tokenizer.apply_chat_template(
            request.messages, tokenize=False
        )
        input_ids = model_manager.tokenizer(inputs, return_tensors="pt").to("cuda")

        # 流式生成
        from transformers import TextStreamer

        streamer = TextStreamer(
            model_manager.tokenizer,
            skip_prompt=True,
            skip_special_tokens=True
        )

        # 在后台线程中生成，逐 token 输出
        import threading
        output_queue = asyncio.Queue()

        def generate_tokens():
            model_manager.model.generate(
                **input_ids,
                max_new_tokens=request.max_tokens,
                temperature=request.temperature,
                streamer=streamer
            )

        thread = threading.Thread(target=generate_tokens)
        thread.start()

        # 等待流式输出（简化版，实际需要更复杂的事件循环集成）
        yield f"data: {json.dumps({'content': '流式输出中...'})}\n\n"
        yield "data: [DONE]\n\n"

    return StreamingResponse(
        generate(),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "Connection": "keep-alive"}
    )
```

## 3. Docker 容器化

### 3.1 优化 Dockerfile

```dockerfile
# ========== 多阶段构建：最小化镜像体积 ==========

# 阶段 1：基础环境
FROM nvidia/cuda:12.1-runtime-ubuntu22.04 AS runtime

# 系统依赖（合并 RUN 减少层数）
RUN apt-get update && apt-get install -y --no-install-recommends \
    python3.11 \
    python3-pip \
    && rm -rf /var/lib/apt/lists/* \
    && ln -s /usr/bin/python3.11 /usr/bin/python

WORKDIR /app

# 阶段 2：依赖安装
FROM runtime AS builder

# 先复制依赖文件（利用 Docker 缓存层）
COPY requirements.txt .

RUN pip install --no-cache-dir -r requirements.txt

# 阶段 3：最终镜像
FROM runtime AS production

# 复制已安装的依赖
COPY --from=builder /usr/local/lib/python3.11/dist-packages /usr/local/lib/python3.11/dist-packages

# 复制应用代码
COPY app.py .
COPY config.yaml .

# 非 root 用户运行
RUN useradd -m -s /bin/bash modeluser
USER modeluser

# 健康检查
HEALTHCHECK --interval=30s --timeout=10s --retries=3 \
    CMD curl -f http://localhost:8000/health || exit 1

EXPOSE 8000

# 启动命令
CMD ["uvicorn", "app:app", "--host", "0.0.0.0", "--port", "8000", "--workers", "1"]
```

## 5. 监控体系搭建

### 5.1 四大监控维度

```
AI 服务监控 = 基础设施 + 应用性能 + 模型质量 + 成本

基础设施层：
  - GPU 利用率、显存使用、温度
  - CPU、内存、磁盘、网络

应用性能层：
  - 请求延迟（P50/P95/P99）
  - 吞吐量（QPS/TPS）
  - 错误率、超时率

模型质量层：
  - Token 消耗（总量/日均）
  - 输出长度分布
  - 用户反馈（👍/👎）

成本层：
  - API 调用费用
  - GPU 实例费用
  - 缓存命中率（节省的成本）
```

### 5.2 Prometheus + Grafana 配置

```python
# metrics.py - 自定义 Prometheus 指标
from prometheus_client import Counter, Histogram, Gauge, generate_latest
import time

# 请求计数器
chat_requests_total = Counter(
    'chat_requests_total',
    'Total chat requests',
    ['model', 'status']  # status: success/error/timeout
)

# 延迟直方图
chat_latency_seconds = Histogram(
    'chat_latency_seconds',
    'Chat request latency',
    ['model'],
    buckets=[0.1, 0.5, 1.0, 2.0, 5.0, 10.0, 30.0, 60.0]
)

# Token 消耗
tokens_consumed_total = Counter(
    'tokens_consumed_total',
    'Total tokens consumed',
    ['type']  # type: prompt/completion
)

# GPU 显存
gpu_memory_used_bytes = Gauge(
    'gpu_memory_used_bytes',
    'GPU memory used'
)


# 在 API 中添加指标采集
@app.post("/v1/chat")
async def chat(request: ChatRequest):
    start = time.time()
    try:
        response = await process_chat(request)
        # 记录成功指标
        chat_requests_total.labels(model="qwen-7b", status="success").inc()
        chat_latency_seconds.labels(model="qwen-7b").observe(time.time() - start)
        tokens_consumed_total.labels(type="prompt").inc(response.usage["prompt_tokens"])
        tokens_consumed_total.labels(type="completion").inc(response.usage["completion_tokens"])
        return response
    except Exception as e:
        chat_requests_total.labels(model="qwen-7b", status="error").inc()
        raise


# Prometheus 指标端点
@app.get("/metrics")
async def get_metrics():
    # 更新 GPU 显存指标
    if torch.cuda.is_available():
        gpu_memory_used_bytes.set(torch.cuda.memory_allocated())
    return Response(content=generate_latest(), media_type="text/plain")
```

## 6. 弹性伸缩与降级

### 6.1 降级策略

```python
class DegradationManager:
    """降级管理器：多层 Fallback"""

    def __init__(self):
        self.model_chain = [
            {"name": "qwen-70b", "timeout": 30},
            {"name": "qwen-7b",  "timeout": 15},
            {"name": "qwen-1.5b","timeout": 5},
            {"name": "static",   "timeout": 1},   # 静态回复
        ]
        self.current_level = 0

    async def call_with_degradation(self, messages: list) -> str:
        """带降级的模型调用"""
        for i, model in enumerate(self.model_chain[self.current_level:], self.current_level):
            try:
                result = await asyncio.wait_for(
                    self._call_model(model["name"], messages),
                    timeout=model["timeout"]
                )
                # 成功后尝试恢复级别
                if self.current_level > 0:
                    self.current_level -= 1
                return result
            except asyncio.TimeoutError:
                self.current_level = i + 1
                continue
            except Exception:
                self.current_level = i + 1
                continue

        return "服务暂时不可用，请稍后重试。"


degradation = DegradationManager()

@app.post("/v1/chat/resilient")
async def resilient_chat(request: ChatRequest):
    """带降级保护的对话接口"""
    return await degradation.call_with_degradation(request.messages)
```

## 核心要点回顾

- 标准部署链路：**FastAPI 封装 → Docker 容器化 → K8s 编排 → Prometheus 监控**
- **Liveness/Readiness Probe** 是生产必备，initialDelaySeconds 要足够大（模型加载慢）
- 监控四大维度：基础设施 + 应用性能 + 模型质量 + 成本
- **流式输出（SSE）** 显著改善用户体验，是对话类 AI 应用的标准做法
- **多层降级** 保障服务可用性：大模型 → 小模型 → 静态回复

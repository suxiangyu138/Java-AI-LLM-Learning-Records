## 1. 文档处理全景

```
原始文档
   ├── PDF → PyPDF2 / pdfplumber → 纯文本
   ├── Markdown → markdown 库 → 结构化内容
   ├── 网页 → BeautifulSoup → 清洗后的文本
   ├── Word → python-docx → 纯文本
   └── 代码 → AST 解析 → 函数/类级别切分
        ↓
   Text Splitter → Chunks → Embedding → 向量数据库
```

from PyPDF2 import PdfReader

def extract_pdf_text(file_path: str) -> list[dict]:
    reader = PdfReader(file_path)
    pages = []
    for i, page in enumerate(reader.pages):
        text = page.extract_text()
        if text.strip():
            pages.append({
                "page": i + 1,
                "content": text.strip(),
                "source": file_path
            })
    return pages
```

import pdfplumber

def extract_with_tables(file_path: str) -> list[dict]:
    results = []
    with pdfplumber.open(file_path) as pdf:
        for i, page in enumerate(pdf.pages):
            text = page.extract_text() or ""
            tables = page.extract_tables()
            table_texts = [_table_to_markdown(t) for t in tables]
            results.append({
                "page": i + 1,
                "text": text,
                "tables": table_texts,
                "combined": text + "\n\n" + "\n\n".join(table_texts)
            })
    return results

def _table_to_markdown(table):
    """将二维数组表格转换为 Markdown 格式"""
    if not table:
        return ""
    lines = ["| " + " | ".join(table[0]) + " |"]
    lines.append("| " + " | ".join(["---"] * len(table[0])) + " |")
    for row in table[1:]:
        lines.append("| " + " | ".join(str(cell or "") for cell in row) + " |")
    return "\n".join(lines)
```

def extract_images_from_pdf(pdf_path: str, output_dir: str):
    doc = fitz.open(pdf_path)
    for i, page in enumerate(doc):
        images = page.get_images(full=True)
        for j, img in enumerate(images):

            img_path = f"{output_dir}/page{i+1}_img{j+1}.{ext}"
            with open(img_path, "wb") as f:
                f.write(base_image["image"])

## 3. Markdown 解析

### 3.1 按标题层级切分

```python
import re

def parse_markdown_sections(md_text: str) -> list[dict]:
    sections = []
    current_section = None
    current_content = []

    for line in md_text.split("\n"):
        header_match = re.match(r"^(#{1,6})\s+(.+)", line)
        if header_match:
            if current_section:
                current_section["content"] = "\n".join(current_content).strip()
            level = len(header_match.group(1))
            title = header_match.group(2)
            current_section = {"level": level, "title": title, "content": ""}
            sections.append(current_section)
            current_content = []
        else:
            current_content.append(line)

    if current_section:
        current_section["content"] = "\n".join(current_content).strip()
    return sections
```

### 3.2 代码块提取

```python
def extract_code_blocks(md_text: str) -> list[dict]:
    pattern = r"```(\w+)?\n(.*?)```"
    blocks = []
    for match in re.finditer(pattern, md_text, re.DOTALL):
        blocks.append({
            "language": match.group(1) or "text",
            "code": match.group(2).strip(),
        })
    return blocks
```

## 4. 网页爬取

```python
import requests
from bs4 import BeautifulSoup

def scrape_page(url: str) -> dict:
    headers = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"}
    response = requests.get(url, headers=headers, timeout=10)
    response.encoding = response.apparent_encoding
    soup = BeautifulSoup(response.text, "lxml")

    for tag in soup(["script", "style", "nav", "footer", "header"]):
        tag.decompose()

    title = soup.title.string if soup.title else ""
    content_tag = soup.find("article") or soup.find("main") or soup.find("div", class_="content") or soup.body
    text = content_tag.get_text(separator="\n", strip=True) if content_tag else ""
    return {"title": title, "content": text, "url": url}
```

### 5.2 中文文本专项切分

```python
class ChineseTextSplitter:
    """按语义边界切分，避免截断句子"""
    def __init__(self, chunk_size: int = 500, chunk_overlap: int = 50):
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap

    def split(self, text: str) -> list[str]:
        sentences = re.split(r"(?<=[。！？；\n])\s*", text)
        sentences = [s.strip() for s in sentences if s.strip()]
        chunks = []
        current_chunk = ""
        for sentence in sentences:
            if len(current_chunk) + len(sentence) <= self.chunk_size:
                current_chunk += sentence
            else:
                if current_chunk:
                    chunks.append(current_chunk)
                current_chunk = current_chunk[-self.chunk_overlap:] + sentence if current_chunk else sentence
        if current_chunk:
            chunks.append(current_chunk)
        return chunks
```

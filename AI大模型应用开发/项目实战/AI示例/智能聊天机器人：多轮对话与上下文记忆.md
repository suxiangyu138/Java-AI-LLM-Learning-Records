## 1. 项目概述

构建一个 Web 版智能聊天机器人，支持多轮对话和上下文记忆。用户通过浏览器与 AI 对话，AI 能记住之前的聊天内容。

### 核心功能
- 多轮对话（上下文记忆）
- 流式输出（打字机效果）
- 对话历史管理
- Web 界面

## 3. 核心实现

### 3.1 对话管理器

```python
# conversation.py
class ConversationManager:
    """多轮对话管理器 — 核心组件"""

    def __init__(self, system_prompt=None, max_history=20):
        self.system_prompt = system_prompt
        self.max_history = max_history
        self.sessions = {}  # session_id → messages[]

    def create_session(self):
        session_id = str(uuid.uuid4())
        self.sessions[session_id] = []
        if self.system_prompt:
            self.sessions[session_id].append(
                {"role": "system", "content": self.system_prompt}
            )
        return session_id

    def add_message(self, session_id, role, content):
        if session_id not in self.sessions:
            self.create_session()
        self.sessions[session_id].append({"role": role, "content": content})
        self._trim(session_id)

    def get_messages(self, session_id):
        return self.sessions.get(session_id, [])

    def _trim(self, session_id):
        """保留 system prompt + 最近 N 条消息"""
        msgs = self.sessions[session_id]
        system = [m for m in msgs if m["role"] == "system"]
        others = [m for m in msgs if m["role"] != "system"]
        self.sessions[session_id] = system + others[-self.max_history:]

    def clear(self, session_id):
        if session_id in self.sessions:
            system = [m for m in self.sessions[session_id]
                       if m["role"] == "system"]
            self.sessions[session_id] = system
```

### 3.2 Flask 后端

```python
# app.py
from flask import Flask, request, jsonify, Response, stream_with_context
from flask_cors import CORS
from openai import OpenAI
import json
import uuid
from conversation import ConversationManager

app = Flask(__name__)
CORS(app)

client = OpenAI(
    api_key="sk-你的密钥",
    base_url="https://api.deepseek.com"
)

manager = ConversationManager(
    system_prompt="你是友好的AI助手，回答简洁准确。"
)

@app.route("/api/session", methods=["POST"])
def create_session():
    session_id = manager.create_session()
    return jsonify({"session_id": session_id})

@app.route("/api/chat", methods=["POST"])
def chat():
    """非流式对话"""
    data = request.json
    session_id = data.get("session_id")
    prompt = data.get("prompt")

    manager.add_message(session_id, "user", prompt)
    messages = manager.get_messages(session_id)

    response = client.chat.completions.create(
        model="deepseek-chat",
        messages=messages
    )
    reply = response.choices[0].message.content
    manager.add_message(session_id, "assistant", reply)

    return jsonify({"reply": reply})

@app.route("/api/chat/stream", methods=["POST"])
def chat_stream():
    """流式对话"""
    data = request.json
    session_id = data.get("session_id")
    prompt = data.get("prompt")

    manager.add_message(session_id, "user", prompt)
    messages = manager.get_messages(session_id)

    def generate():
        full_reply = ""
        response = client.chat.completions.create(
            model="deepseek-chat",
            messages=messages,
            stream=True
        )
        for chunk in response:
            if chunk.choices[0].delta.content:
                content = chunk.choices[0].delta.content
                full_reply += content
                yield f"data: {json.dumps({'content': content}, ensure_ascii=False)}\n\n"

        manager.add_message(session_id, "assistant", full_reply)
        yield "data: [DONE]\n\n"

    return Response(
        stream_with_context(generate()),
        mimetype="text/event-stream"
    )

if __name__ == "__main__":
    app.run(debug=True, port=5000)
```

<script>
const chatBox = document.getElementById("chat-box");
const input = document.getElementById("user-input");
const sendBtn = document.getElementById("send-btn");

let sessionId = null;

// 初始化会话
fetch("/api/session", { method: "POST" })
    .then(r => r.json())
    .then(data => { sessionId = data.session_id; });

function addMessage(role, content) {
    const div = document.createElement("div");
    div.className = `message ${role}`;
    div.innerHTML = `<div class="bubble">${content}</div>`;
    chatBox.appendChild(div);
    chatBox.scrollTop = chatBox.scrollHeight;
}

async function sendMessage() {
    const prompt = input.value.trim();
    if (!prompt) return;
    input.value = "";

    addMessage("user", prompt);

    // 流式接收
    const resp = await fetch("/api/chat/stream", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ session_id: sessionId, prompt })
    });

    const reader = resp.body.getReader();
    const decoder = new TextDecoder();
    const bubble = document.createElement("div");
    bubble.className = "message assistant";
    const content = document.createElement("div");
    content.className = "bubble";
    bubble.appendChild(content);
    chatBox.appendChild(bubble);

    let fullText = "";
    while (true) {
        const { done, value } = await reader.read();
        if (done) break;
        const text = decoder.decode(value);
        for (const line of text.split("\n")) {
            if (line.startsWith("data: ") && line !== "data: [DONE]") {
                const data = JSON.parse(line.slice(6));
                fullText += data.content;
                content.textContent = fullText;
                chatBox.scrollTop = chatBox.scrollHeight;
            }
        }
    }
}

sendBtn.addEventListener("click", sendMessage);
input.addEventListener("keydown", e => {
    if (e.key === "Enter") sendMessage();
});
</script>
</body>
</html>

# gradio_app.py
import gradio as gr
from openai import OpenAI

client = OpenAI(api_key="sk-xxx", base_url="https://api.deepseek.com")

def chat(message, history):
    messages = [{"role": "system", "content": "你是友好助手"}]
    for h in history:
        messages.append({"role": "user", "content": h[0]})
        messages.append({"role": "assistant", "content": h[1]})
    messages.append({"role": "user", "content": message})

    response = client.chat.completions.create(
        model="deepseek-chat", messages=messages
    )
    return response.choices[0].message.content

gr.ChatInterface(chat, title="AI 聊天机器人").launch()

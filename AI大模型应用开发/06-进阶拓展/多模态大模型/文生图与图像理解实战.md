## 一、Stable Diffusion 文生图实战

### 1.1 核心参数

```python
import torch
from diffusers import StableDiffusionPipeline

pipe = StableDiffusionPipeline.from_pretrained(
    "runwayml/stable-diffusion-v1-5",
    torch_dtype=torch.float16
).to("cuda")

def generate(prompt: str, **kwargs):
    defaults = {
        "negative_prompt": "blurry, low quality, ugly, distorted, bad anatomy, watermark, text",
        "num_inference_steps": 30,    # 步数（越大质量越好但慢）
        "guidance_scale": 7.5,        # Prompt 引导强度（越高越听 Prompt，范围 5-15）
        "width": 768,
        "height": 512,
        "generator": torch.Generator("cuda").manual_seed(42),
    }
    defaults.update(kwargs)
    return pipe(**defaults).images[0]
```

> **重点**：参数调优经验——`num_inference_steps` 建议 20（草稿）/ 30（日常）/ 50+（精修）；`guidance_scale` 建议 3-5（创意）/ 7-8（日常）/ 10-15（严格遵循 Prompt）；固定 seed 可复现相同图片，调试时非常有价值。

### 1.3 批量生成 + 自动筛选

```python
import random

def batch_generate(prompt: str, count: int = 4) -> list:
    """批量生成多个变体，人工筛选最佳"""
    images = []
    for i in range(count):
        seed = random.randint(1, 1000000)
        img = generate(prompt, seed=seed)
        img.save(f"output/gen_{i}_seed_{seed}.png")
        images.append({"image": img, "seed": seed})
        print(f"Generated {i+1}/{count} (seed={seed})")
    return images
```

## 二、ControlNet 精准控制

### 2.1 线稿上色

```python
from diffusers import StableDiffusionControlNetPipeline, ControlNetModel
from PIL import Image
import cv2
import numpy as np

controlnet = ControlNetModel.from_pretrained(
    "lllyasviel/sd-controlnet-canny", torch_dtype=torch.float16
).to("cuda")

pipe = StableDiffusionControlNetPipeline.from_pretrained(
    "runwayml/stable-diffusion-v1-5", controlnet=controlnet, torch_dtype=torch.float16
).to("cuda")

def image_to_image_with_edge(
    reference_image_path: str, prompt: str,
    low_threshold: int = 100, high_threshold: int = 200
) -> Image:
    """基于参考图的边缘结构生成新图"""
    ref_img = Image.open(reference_image_path).convert("RGB")
    ref_np = np.array(ref_img)
    edges = cv2.Canny(ref_np, low_threshold, high_threshold)
    edges_img = Image.fromarray(edges)

    output = pipe(
        prompt=prompt, image=edges_img,
        num_inference_steps=30, guidance_scale=7.5
    ).images[0]
    return output

# 场景：手绘草图 → 精修图
result = image_to_image_with_edge(
    "hand_sketch.png",
    "A modern web application dashboard, clean UI design, professional"
)
```

### 3.2 UI 设计稿 → 代码

```python
def ui_screenshot_to_code(image_path: str, framework: str = "html-tailwind") -> str:
    """将 UI 截图转为前端代码"""
    FRAMEWORK_PROMPTS = {
        "html-tailwind": "生成使用 Tailwind CSS 的 HTML 代码",
        "react": "生成 React + Tailwind 组件代码，包含状态管理",
        "vue": "生成 Vue3 + Tailwind 单文件组件",
        "flutter": "生成 Flutter Widget 代码",
    }
    prompt = f"""请根据这张UI设计图，{FRAMEWORK_PROMPTS[framework]}。
要求：1. 保持与设计图最大程度的视觉一致性 2. 使用语义化的结构 3. 处理好边界情况（空状态、加载态） 4. 只输出代码，不需要解释"""
    code = analyze_image_local(image_path, prompt)
    import re
    match = re.search(r"```(?:\w+)?\n(.*?)```", code, re.DOTALL)
    return match.group(1) if match else code
```

### 3.3 视频帧理解

```python
import cv2

def analyze_video_frames(video_path: str, interval: int = 30) -> list[dict]:
    """每隔 interval 帧抽取关键帧并分析"""
    cap = cv2.VideoCapture(video_path)
    fps = int(cap.get(cv2.CAP_PROP_FPS))
    frame_count = 0
    analyses = []
    while True:
        ret, frame = cap.read()
        if not ret:
            break
        if frame_count % interval == 0:
            frame_path = f"frames/frame_{frame_count}.jpg"
            cv2.imwrite(frame_path, frame)
            desc = analyze_image_local(frame_path, "描述这张视频帧中的内容。")
            analyses.append({
                "frame": frame_count,
                "timestamp": f"{frame_count // fps}s",
                "description": desc
            })
        frame_count += 1
    cap.release()
    return analyses
```

## 四、项目实战：智能简历生成器完整链路

```python
class SmartResumeGenerator:
    """多模态智能简历生成器"""
    def __init__(self):
        self.vision = QwenVLService()
        self.writer = DeepSeekService()
        self.image_gen = StableDiffusionService()  # 可换 DALL-E

    def generate(self, photo_path: str, old_resume_path: str, target_job: str) -> dict:
        # Step 1: 处理证件照
        photo_analysis = self.vision.analyze(
            photo_path,
            "分析这张照片：是否适合做证件照？背景、光线、表情如何？给出评分和改进建议(1-10)"
        )
        enhanced_photo = self._enhance_photo(photo_path, photo_analysis)

        # Step 2: OCR 提取旧简历
        resume_text = self.vision.extract_text(old_resume_path)

        # Step 3: AI 分析 + 优化简历内容
        analysis = self.writer.analyze_resume(resume_text, target_job)
        optimized = self.writer.rewrite_resume(resume_text, analysis, target_job)

        # Step 4: 生成 PDF
        pdf = self._format_to_pdf(optimized, enhanced_photo)

        # Step 5: 生成简历封面图
        cover_image = self.image_gen.generate(
            f"A professional resume cover image with abstract technology elements, featuring {target_job} keywords, modern clean design, blue gradient"
        )
        return {
            "photo_analysis": photo_analysis, "enhanced_photo": enhanced_photo,
            "resume_text": optimized, "pdf": pdf, "cover_image": cover_image,
        }
```

## 核心要点回顾

- Stable Diffusion 的 Prompt 使用英文效果最佳，结合 `negative_prompt` 可显著提升生成质量
- ControlNet 支持基于边缘图（Canny）和姿态图（OpenPose）的精准控制
- Qwen-VL 支持 OCR、QA、代码生成、Bug 分析等多种图像理解任务
- 视频帧分析通过关键帧提取 + 逐帧分析实现，需注意帧间隔选择
- 批量生成时使用不同 seed 可获得多样化的变体结果

## 快速调试检查清单

- [ ] Stable Diffusion 的 `prompt` 是否用英文？（SD 对英文 Prompt 理解最好）

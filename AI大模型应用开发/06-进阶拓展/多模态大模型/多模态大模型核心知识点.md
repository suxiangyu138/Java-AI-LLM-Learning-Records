### 2.2 本地图片转 Base64

```python
import base64
from pathlib import Path

def image_to_base64(image_path: str) -> str:
    """将本地图片转为 Base64（API 需要）"""
    with open(image_path, "rb") as f:
        return base64.b64encode(f.read()).decode("utf-8")

def analyze_image_local(image_path: str, question: str) -> str:
    """分析本地图片"""
    b64 = image_to_base64(image_path)
    ext = Path(image_path).suffix.lower().replace(".", "")
    response = client.chat.completions.create(
        model="qwen-vl-max",
        messages=[{
            "role": "user",
            "content": [
                {"type": "text", "text": question},
                {"type": "image_url", "image_url": {"url": f"data:image/{ext};base64,{b64}"}}
            ]
        }]
    )
    return response.choices[0].message.content
```

**典型应用场景**：UI 截图生成前端代码、架构图生成文字描述、错误截图自动定位问题、证件照信息提取。

### 3.2 Stable Diffusion 本地部署

```python
from diffusers import StableDiffusionPipeline
import torch

# 加载模型（首次会下载约 4GB）
pipe = StableDiffusionPipeline.from_pretrained(
    "runwayml/stable-diffusion-v1-5",
    torch_dtype=torch.float16,
    safety_checker=None  # 开发调试时关闭安全检测加速
)
pipe = pipe.to("cuda")

def generate_image(
    prompt: str,
    negative_prompt: str = "blurry, low quality, distorted",
    width: int = 512,
    height: int = 512,
    num_inference_steps: int = 25,
    seed: int = 42
):
    """生成图片"""
    generator = torch.Generator("cuda").manual_seed(seed)
    image = pipe(
        prompt=prompt,
        negative_prompt=negative_prompt,
        width=width,
        height=height,
        num_inference_steps=num_inference_steps,
        generator=generator
    ).images[0]
    return image

img = generate_image(
    prompt="A modern tech office with clean design, natural lighting, 8k detailed",
    negative_prompt="people, text, watermark"
)
img.save("office.png")
```

## 四、多模态应用架构

### 4.1 图片理解 + 文本生成链路

```python
class MultimodalPipeline:
    """多模态处理流水线"""
    def __init__(self, vision_model: str = "qwen-vl-max", text_model: str = "deepseek-chat"):
        self.vision_client = OpenAI(base_url="https://dashscope.aliyuncs.com/compatible-mode/v1")
        self.text_client = OpenAI(base_url="https://api.deepseek.com/v1")
        self.vision_model = vision_model
        self.text_model = text_model

    def analyze_and_generate(self, image_path: str, task: str) -> dict:
        """图片分析 + 内容生成"""
        description = analyze_image_local(image_path, "请详细描述这张图片的内容。")
        prompt = f"根据以下图片描述，{task}\n\n图片描述：\n{description}"
        result = self.text_client.chat.completions.create(
            model=self.text_model,
            messages=[{"role": "user", "content": prompt}],
            temperature=0.7
        )
        return {"description": description, "result": result.choices[0].message.content}
```

### 4.2 服务化封装

```python
from abc import ABC, abstractmethod

class MultimodalService(ABC):
    @abstractmethod
    def image_to_text(self, image: bytes, prompt: str) -> str: ...
    @abstractmethod
    def text_to_image(self, prompt: str) -> bytes: ...

class QwenVLService(MultimodalService):
    def image_to_text(self, image: bytes, prompt: str) -> str:
        b64 = base64.b64encode(image).decode()
        response = self.client.chat.completions.create(
            model="qwen-vl-max",
            messages=[{
                "role": "user",
                "content": [
                    {"type": "text", "text": prompt},
                    {"type": "image_url", "image_url": {"url": f"data:image/png;base64,{b64}"}}
                ]
            }]
        )
        return response.choices[0].message.content

    def text_to_image(self, prompt: str) -> bytes:
        response = self.client.images.generate(
            model="dall-e-3", prompt=prompt, size="1024x1024", n=1
        )
        return requests.get(response.data[0].url).content
```

## 六、实战项目：简历优化器

```python
class ResumeOptimizer:
    """多模态简历优化器"""
    def __init__(self):
        self.vision = OpenAI(base_url="...")  # Qwen-VL
        self.writer = OpenAI(base_url="...")  # DeepSeek

    def optimize(self, resume_image_path: str, target_job: str) -> dict:
        """输入简历截图，输出优化后的简历"""
        resume_text = extract_text_from_image(resume_image_path)
        analysis = self._analyze_resume(resume_text, target_job)
        optimized = self._rewrite_resume(resume_text, analysis, target_job)
        md = self._format_to_markdown(optimized)
        return {"original_text": resume_text, "analysis": analysis, "optimized_text": optimized, "markdown": md}

    def _analyze_resume(self, resume: str, job: str) -> str:
        prompt = f"分析以下简历，找出与目标岗位 '{job}' 的差距：\n简历：{resume}\n请指出：1. 缺少的技能 2. 表达不清晰的项目描述 3. 格式问题 4. 改进建议"
        return self.writer.chat(prompt)

    def _rewrite_resume(self, resume: str, analysis: str, job: str) -> str:
        prompt = f"根据分析结果重写简历，面向岗位：{job}\n原简历：{resume}\n分析：{analysis}\n要求：1. 用STAR法则描述项目 2. 突出量化成果 3. 匹配岗位关键词"
        return self.writer.chat(prompt)

    def _format_to_markdown(self, text: str) -> str:
        prompt = f"将以下简历内容格式化为美观的 Markdown：\n{text}"
        return self.writer.chat(prompt)
```

## 核心要点回顾

- 掌握 Qwen-VL 和 DALL-E 3 的 API 调用即可覆盖大部分多模态应用场景
- 本地图片处理需先转为 Base64 格式，注意 `data:image/{ext};base64,{b64}` 格式
- Stable Diffusion 本地部署需要约 4GB 磁盘空间和 4GB 显存
- 文生图 Prompt 使用英文效果最佳，采用"主体 + 风格 + 光照 + 构图 + 质量"公式
- 多模态应用架构可采用"视觉模型理解 + 文本模型生成"的两阶段流水线模式

## 快速调试检查清单

- [ ] 图片 Base64 编码是否正确？（检查 `data:image/{ext};base64,{b64}` 格式）
- [ ] Vision API 是否支持该图片格式？（常见：PNG、JPEG、WebP）
- [ ] 图片大小是否超过 API 限制？（一般小于 20MB）
- [ ] Stable Diffusion 首次加载是否下载了足够空间？（模型文件约 4GB）
- [ ] GPU 显存是否足够运行 SD？（512x512 需要约 4GB）

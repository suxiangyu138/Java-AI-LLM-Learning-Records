## 1. 评估体系总览

```
模型评估
├── 自动化指标（快速、低成本）
│   ├── BLEU — 翻译/生成质量
│   ├── ROUGE — 摘要/文档质量
│   ├── BERTScore — 语义相似度
│   └── Perplexity — 语言流畅度
│
├── 模型评判（用更强的模型打分）
│   └── GPT-4 / Claude 作为评判
│
└── 人工评估（最终把关）
    ├── A/B 盲测
    ├── Likert 量表评分
    └── 偏好标注（Win / Tie / Loss）
```

## 2. 自动化指标

### 2.1 BLEU（双语评估替补）

```python
from nltk.translate.bleu_score import sentence_bleu, SmoothingFunction
import jieba


def compute_bleu(reference: str, candidate: str, lang: str = "zh") -> float:
    """计算 BLEU 分数（0-1，越高越好）"""
    if lang == "zh":
        # 中文需要分词
        ref_tokens = [list(jieba.cut(reference))]
        cand_tokens = list(jieba.cut(candidate))
    else:
        ref_tokens = [reference.split()]
        cand_tokens = candidate.split()

    # SmoothingFunction 避免 0 分（短文本匹配不上时）
    smoothie = SmoothingFunction().method1
    return sentence_bleu(ref_tokens, cand_tokens, smoothing_function=smoothie)


# 示例
ref = "在 Spring Boot 中，使用 @Transactional 注解可以声明式地管理事务"
cand = "Spring Boot 通过 @Transactional 注解来实现声明式事务管理"
print(f"BLEU: {compute_bleu(ref, cand):.3f}")  # ~0.35
```

**BLEU 的局限**：只比较 n-gram 重叠，不理解语义。同义词替换会被扣分。

```python
from rouge_score import rouge_scorer


def compute_rouge(reference: str, candidate: str, lang: str = "zh") -> dict:
    """计算 ROUGE 分数"""
    if lang == "zh":
        reference = " ".join(jieba.cut(reference))
        candidate = " ".join(jieba.cut(candidate))

    scorer = rouge_scorer.RougeScorer(
        ["rouge1", "rouge2", "rougeL"],
        use_stemmer=False
    )
    scores = scorer.score(reference, candidate)

    return {
        "rouge-1": round(scores["rouge1"].fmeasure, 4),  # 单词级别
        "rouge-2": round(scores["rouge2"].fmeasure, 4),  # 双词级别
        "rouge-L": round(scores["rougeL"].fmeasure, 4),  # 最长公共子序列
    }


# 示例
ref = "使用 @Transactional 注解可以自动管理数据库事务的开启和提交"
cand = "@Transactional 注解能自动处理事务的提交和回滚"
print(compute_rouge(ref, cand))
# {'rouge-1': 0.6154, 'rouge-2': 0.3333, 'rouge-L': 0.4615}
```

### 2.3 BERTScore（语义级评估，推荐）


```python
from bert_score import score


def compute_bertscore(references: list[str], candidates: list[str], lang: str = "zh"):
    """BERTScore：用预训练模型计算语义相似度"""
    P, R, F1 = score(
        candidates,
        references,
        lang=lang,
        model_type="bert-base-chinese",  # 中文用 bert-base-chinese
        verbose=False
    )
    return {
        "precision": P.mean().item(),
        "recall": R.mean().item(),
        "f1": F1.mean().item(),
    }


# BERTScore 能识别同义表达
refs = ["这个方法的执行效率很低"]
cands = ["该函数运行速度非常慢"]  # 用词不同但语义一致
print(compute_bertscore(refs, cands))
# f1 ≈ 0.85（比 BLEU 更能反映真实质量）
```

### 2.4 Perplexity（困惑度）

```python
import torch
from transformers import AutoModelForCausalLM, AutoTokenizer


def compute_perplexity(model, tokenizer, text: str) -> float:
    """计算困惑度（越低越好，说明模型对该文本越"不困惑"）"""
    inputs = tokenizer(text, return_tensors="pt").to(model.device)
    with torch.no_grad():
        outputs = model(**inputs, labels=inputs["input_ids"])
        loss = outputs.loss
    return torch.exp(loss).item()


# 评估：微调前后，模型对目标领域文本的困惑度是否下降
base_ppl = compute_perplexity(base_model, tokenizer, java_code_text)
finetuned_ppl = compute_perplexity(finetuned_model, tokenizer, java_code_text)
print(f"基础模型 PPL: {base_ppl:.1f} → 微调后 PPL: {finetuned_ppl:.1f}")
```

## 3. LLM-as-Judge（用强模型做评判）

```python
def llm_judge(prompt: str, response_a: str, response_b: str, judge_client) -> dict:
    """用 GPT-4 / Claude 作为裁判，比较两个回答的质量"""
    judge_prompt = f"""你是一个公正的 AI 评测员。请比较以下两个 AI 助手对同一问题的回答。

问题：
{prompt}

回答 A：
{response_a}

回答 B：
{response_b}

请从以下维度评分（1-5分）：
1. 准确性：回答是否包含事实错误
2. 完整性：是否完整回答了问题的所有方面
3. 实用性：回答是否可直接用于解决问题
4. 语言质量：表达是否清晰、专业

输出 JSON 格式：
{{
    "A": {{"accuracy": 分, "completeness": 分, "practicality": 分, "language": 分, "total": 总分}},
    "B": {{"accuracy": 分, "completeness": 分, "practicality": 分, "language": 分, "total": 总分}},
    "winner": "A" / "B" / "tie",
    "reason": "简短理由"
}}"""

    result = judge_client.chat(judge_prompt)
    return json.loads(result)


# 批量评估
def benchmark_vs_baseline(
    test_queries: list[str],
    finetuned_model,
    baseline_model,
    judge_client
) -> dict:
    """微调模型 vs 基准模型 A/B 测试"""
    wins = 0
    ties = 0
    losses = 0

    for query in test_queries:
        response_a = finetuned_model.generate(query)  # 随机分配 A/B 避免位置偏差
        response_b = baseline_model.generate(query)

        if random.random() > 0.5:
            response_a, response_b = response_b, response_a
            swapped = True
        else:
            swapped = False

        judgment = llm_judge(query, response_a, response_b, judge_client)
        winner = judgment["winner"]

        if (winner == "A" and not swapped) or (winner == "B" and swapped):
            wins += 1
        elif winner == "tie":
            ties += 1
        else:
            losses += 1

    total = len(test_queries)
    return {
        "win_rate": wins / total,
        "tie_rate": ties / total,
        "loss_rate": losses / total,
    }
```

## 4. 人工评估流程

### 4.1 评估表格模板

```markdown
| 维度 | 1分（差） | 2分（一般） | 3分（好） | 4分（很好） | 5分（优秀） |
|------|-----------|-------------|-----------|-------------|-------------|
| 代码质量 | 有严重语法错误 | 能跑但有明显问题 | 实现正确 | 实现优雅 | 生产级别 |

| 实用性 | 建议不能落地 | 部分建议可用 | 大部分可用 | 可直接采用 | 超出预期 |
```

## 5. 完整评估脚本

```python
class ModelEvaluator:
    """模型评估器——一键运行所有指标"""

    def __init__(self, model, tokenizer, test_data: list[dict]):
        """
        test_data = [
            {"input": "...", "reference": "标准答案"},
            ...
        ]
        """
        self.model = model
        self.tokenizer = tokenizer
        self.test_data = test_data

    def evaluate(self) -> dict:
        predictions = []
        references = []

        for item in self.test_data:
            pred = self._generate(item["input"])
            predictions.append(pred)
            references.append(item["reference"])

        return {
            "bleu": self._batch_bleu(predictions, references),
            "rouge": self._batch_rouge(predictions, references),
            "bertscore": compute_bertscore(references, predictions),
            "num_samples": len(self.test_data),
        }

    def _generate(self, prompt: str) -> str:
        inputs = self.tokenizer(prompt, return_tensors="pt").to(self.model.device)
        outputs = self.model.generate(**inputs, max_new_tokens=256)
        return self.tokenizer.decode(outputs[0], skip_special_tokens=True)

    def _batch_bleu(self, preds, refs):
        scores = [compute_bleu(r, p) for r, p in zip(refs, preds)]
        return sum(scores) / len(scores)

    def _batch_rouge(self, preds, refs):
        r1, r2, rl = [], [], []
        for r, p in zip(refs, preds):
            s = compute_rouge(r, p)
            r1.append(s["rouge-1"])
            r2.append(s["rouge-2"])
            rl.append(s["rouge-L"])
        return {
            "rouge-1": sum(r1)/len(r1),
            "rouge-2": sum(r2)/len(r2),
            "rouge-L": sum(rl)/len(rl),
        }
```

## 快速调试检查清单

- [ ] BLEU/ROUGE 的取值范围是 0-1，0 分检查分词是否正确
- [ ] BERTScore 的 `model_type` 是否匹配语言？（中文用 `bert-base-chinese`）
- [ ] LLM-as-Judge 是否随机分配了 A/B 顺序？（避免位置偏差）
- [ ] 测试集数据是否和训练集**无重叠**？（数据泄露会导致虚高指标）

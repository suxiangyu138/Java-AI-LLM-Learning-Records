## 2. 视觉理解（图→文）

### 2.1 多模态模型调用

```java
@Service
public class VisionService {

    private final ChatClient visionClient;

    public VisionService(ChatClient.Builder builder) {
        this.visionClient = builder.build();
    }

    /**
     * URL 方式传入图片
     */
    public String analyzeImageByUrl(String imageUrl, String question) {
        return visionClient.prompt()
            .user(userSpec -> userSpec
                .text(question)
                .media(Media.Format.IMAGE_PNG, imageUrl)
            )
            .options(OpenAiChatOptions.builder()
                .withModel("gpt-4o")  // 支持视觉的模型
                .build())
            .call()
            .content();
    }

    /**
     * 本地文件/Base64 方式传入图片
     */
    public String analyzeImageByFile(MultipartFile imageFile, String question) throws IOException {
        byte[] imageBytes = imageFile.getBytes();
        String base64Image = Base64.getEncoder().encodeToString(imageBytes);
        String dataUri = "data:" + imageFile.getContentType() + ";base64," + base64Image;

        return visionClient.prompt()
            .user(userSpec -> userSpec
                .text(question)
                .media(Media.Format.IMAGE_PNG, dataUri)
            )
            .call()
            .content();
    }

    /**
     * 多图分析
     */
    public String compareImages(List<MultipartFile> images, String question)
            throws IOException {
        UserMessageSpec spec = userSpec -> {
            userSpec.text(question);
            for (MultipartFile image : images) {
                String b64 = Base64.getEncoder()
                    .encodeToString(image.getBytes());
                userSpec.media(Media.Format.IMAGE_PNG,
                    "data:" + image.getContentType() + ";base64," + b64);
            }
        };

        return visionClient.prompt()
            .user(spec)
            .call()
            .content();
    }
}
```

/**
 * 多模态 RAG：文档中的图片也能被检索
 */
@Service
public class MultiModalRagService {

    // 1. 文档处理时，提取图片并生成描述
    public List<Document> processDocumentWithImages(InputStream pdfStream) {
        List<Document> documents = new ArrayList<>();

        // 提取文本
        String text = extractText(pdfStream);

        // 提取图片并为每张图生成描述
        List<ImageWithCaption> images = extractImages(pdfStream);
        for (var img : images) {
            String caption = visionClient.prompt()
                .user(userSpec -> userSpec
                    .text("请详细描述这张图片的内容，包括所有文字、数字、图表含义")
                    .media(Media.Format.IMAGE_PNG, img.base64())
                )
                .call()
                .content();

            // 图片描述也作为一个 Document
            documents.add(new Document(
                String.format("[图片描述] 第%d页: %s", img.page(), caption),
                Map.of("type", "image_caption", "page", String.valueOf(img.page()))
            ));
        }

        // 分割文本
        documents.addAll(splitText(text));
        return documents;
    }
}

@Service
public class TextToImageService {

    private final ImageModel imageModel;

    // DALL-E 3 生成
    public String generateWithDalle(String prompt) {
        ImageResponse response = imageModel.call(
            new ImagePrompt(prompt,
                OpenAiImageOptions.builder()
                    .withModel("dall-e-3")
                    .withQuality("hd")           // standard / hd
                    .withStyle("vivid")          // vivid / natural
                    .withN(1)
                    .withHeight(1024)
                    .withWidth(1024)
                    .build()
            )
        );
        return response.getResult().getOutput().getUrl();
    }

    // 批量生成
    public List<String> generateBatch(String prompt, int count) {
        ImageResponse response = imageModel.call(
            new ImagePrompt(prompt,
                OpenAiImageOptions.builder()
                    .withModel("dall-e-2")
                    .withN(Math.min(count, 10))  // DALL-E 2 最多 10 张
                    .withHeight(1024)
                    .withWidth(1024)
                    .build()
            )
        );
        return response.getResults().stream()
            .map(r -> r.getOutput().getUrl())
            .toList();
    }
}

## 5. 多模态流水线

### 5.1 完整流水线示例

```java
/**
 * 完整多模态流水线：
 * 语音输入 → ASR → 文本 → LLM → TTS → 语音输出
 */
@Service
public class MultimodalPipeline {

    private final AudioTranscriptionModel asrModel;
    private final ChatClient chatClient;
    private final AudioSpeechModel ttsModel;

    /**
     * 语音对话：语音问 → 语音答
     */
    public byte[] voiceChat(byte[] audioInput) {
        // 1. ASR：语音 → 文本
        Resource audioResource = new ByteArrayResource(audioInput);
        String userText = asrModel.call(
            new AudioTranscriptionPrompt(audioResource)
        ).getResult().getOutput();

        // 2. LLM：文本 → 文本
        String responseText = chatClient.prompt()
            .user(userText)
            .call()
            .content();

        // 3. TTS：文本 → 语音
        byte[] audioOutput = ttsModel.call(
            new SpeechPrompt(responseText)
        ).getResult().getOutput();

        return audioOutput;
    }
}
```

        for (int i = 0; i < frames.size(); i += 30) {  // 每 30 秒分析一帧

                .user(userSpec -> userSpec

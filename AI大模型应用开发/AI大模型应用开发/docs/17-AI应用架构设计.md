## 2. 数据库设计

### 2.1 核心表结构

```sql
-- ===== 用户与权限 =====
CREATE TABLE users (
    id          BIGSERIAL PRIMARY KEY,
    username    VARCHAR(64) UNIQUE NOT NULL,
    email       VARCHAR(128),
    api_key     VARCHAR(128) UNIQUE,       -- 个人 API Key
    role        VARCHAR(32) DEFAULT 'USER', -- USER / ADMIN / EDITOR
    tier        VARCHAR(32) DEFAULT 'FREE', -- FREE / PRO / ENTERPRISE
    quota_daily BIGINT DEFAULT 10000,       -- 每日 Token 配额
    created_at  TIMESTAMP DEFAULT NOW()
);

-- ===== 对话会话 =====
CREATE TABLE conversations (
    id          BIGSERIAL PRIMARY KEY,
    user_id     BIGINT REFERENCES users(id),
    title       VARCHAR(255),               -- 自动生成标题
    model       VARCHAR(64),                -- 使用的模型
    message_count INT DEFAULT 0,
    total_tokens  BIGINT DEFAULT 0,
    created_at  TIMESTAMP DEFAULT NOW(),
    updated_at  TIMESTAMP DEFAULT NOW()
);

-- ===== 对话消息 =====
CREATE TABLE messages (
    id              BIGSERIAL PRIMARY KEY,
    conversation_id BIGINT REFERENCES conversations(id),
    role            VARCHAR(16) NOT NULL,     -- system / user / assistant / tool
    content         TEXT NOT NULL,
    token_count     INT,
    metadata        JSONB,                   -- tool_calls / citations 等
    created_at      TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_messages_conv ON messages(conversation_id, created_at);

-- ===== RAG 文档 =====
CREATE TABLE rag_documents (
    id          BIGSERIAL PRIMARY KEY,
    filename    VARCHAR(255),
    file_type   VARCHAR(32),
    file_size   BIGINT,
    chunk_count INT,
    status      VARCHAR(32) DEFAULT 'PROCESSING',  -- PROCESSING / READY / FAILED
    uploaded_by BIGINT REFERENCES users(id),
    created_at  TIMESTAMP DEFAULT NOW()
);

-- ===== Prompt 模板 =====
CREATE TABLE prompt_templates (
    id            BIGSERIAL PRIMARY KEY,
    name          VARCHAR(128) UNIQUE NOT NULL,
    version       VARCHAR(32) DEFAULT 'v1.0.0',
    system_prompt TEXT,
    user_template TEXT DEFAULT '{message}',
    status        VARCHAR(32) DEFAULT 'DRAFT', -- DRAFT / ACTIVE / ARCHIVED
    avg_rating    DOUBLE PRECISION,
    usage_count   BIGINT DEFAULT 0,
    created_by    BIGINT REFERENCES users(id),
    created_at    TIMESTAMP DEFAULT NOW()
);

-- ===== AI 调用日志（成本分析）=====
CREATE TABLE ai_usage_logs (
    id            BIGSERIAL PRIMARY KEY,
    user_id       BIGINT REFERENCES users(id),
    model         VARCHAR(64),
    endpoint      VARCHAR(64),              -- chat / rag / agent
    input_tokens  INT,
    output_tokens INT,
    cache_hit     BOOLEAN DEFAULT FALSE,
    latency_ms    INT,
    cost          NUMERIC(10, 6),
    created_at    TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_usage_user_date ON ai_usage_logs(user_id, created_at);
CREATE INDEX idx_usage_model_date ON ai_usage_logs(model, created_at);
```

### 3.2 多租户实现

```java
// 租户上下文
public class TenantContext {
    private static final ThreadLocal<String> currentTenant = new ThreadLocal<>();

    public static void set(String tenantId) { currentTenant.set(tenantId); }
    public static String get() { return currentTenant.get(); }
    public static void clear() { currentTenant.remove(); }
}

// 租户过滤器
@Component
public class TenantFilter extends OncePerRequestFilter {
    @Override
    protected void doFilterInternal(HttpServletRequest request,
                                     HttpServletResponse response,
                                     FilterChain chain)
            throws ServletException, IOException {
        String tenantId = request.getHeader("X-Tenant-Id");
        if (tenantId != null) {
            TenantContext.set(tenantId);
        }
        try {
            chain.doFilter(request, response);
        } finally {
            TenantContext.clear();
        }
    }
}

// 多租户 RAG 服务：每个租户独立的向量集合
@Service
public class MultiTenantRagService {

    public List<Document> search(String query, int topK) {
        String tenant = TenantContext.get();
        return vectorStore.similaritySearch(
            SearchRequest.query(query)
                .withTopK(topK)
                .withFilterExpression("tenant_id == '" + tenant + "'")
        );
    }

    public void addDocuments(List<Document> docs) {
        String tenant = TenantContext.get();
        docs.forEach(d -> d.getMetadata().put("tenant_id", tenant));
        vectorStore.add(docs);
    }
}
```

## 4. 高可用设计

### 4.1 高可用策略

```java
/**
 * 多模型 Fallback 策略
 */
@Service
public class HighAvailabilityService {

    private final Map<String, ChatModel> primaryModels;
    private final Map<String, ChatModel> fallbackModels;
    private final CircuitBreakerRegistry cbRegistry;

    public String chatWithFallback(String prompt) {
        return Try.of(() -> callWithCircuitBreaker("gpt-4o", prompt))
            .recover(TimeoutException.class,
                () -> callWithCircuitBreaker("gpt-4o-mini", prompt))
            .recover(CircuitBreakerOpenException.class,
                () -> callWithCircuitBreaker("local-llm", prompt))
            .get();
    }

    private String callWithCircuitBreaker(String model, String prompt) {
        CircuitBreaker cb = cbRegistry.circuitBreaker(model);
        return cb.executeCallable(() -> {
            ChatModel chatModel = primaryModels.get(model);
            return chatModel.call(prompt);
        });
    }
}
```

## 5. 异步任务处理

### 5.1 文档摄入异步化

```java
@Service
public class AsyncDocumentIngestion {

    // 使用 RabbitMQ / Kafka 处理大批量文档摄入

    @RabbitListener(queues = "document.ingestion")
    public void handleIngestionTask(IngestionTask task) {
        try {
            // 1. 解析文档
            String text = documentParser.parse(task.getFileContent(), task.getFileType());

            // 2. 分割
            List<String> chunks = textSplitter.split(text);

            // 3. 向量化 + 存储
            for (String chunk : chunks) {
                float[] embedding = embeddingModel.embed(chunk);
                vectorStore.add(new Document(chunk, embedding));
            }

            // 4. 通知完成
            notifyUser(task.getUserId(), "文档摄入完成: " + task.getFilename());
        } catch (Exception e) {
            notifyUser(task.getUserId(), "文档摄入失败: " + e.getMessage());
        }
    }
}
```

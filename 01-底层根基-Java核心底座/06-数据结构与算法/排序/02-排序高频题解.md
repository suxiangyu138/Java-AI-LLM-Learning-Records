## 1. 912. 排序数组（手撕三件套）

### 1.1 快排（含三数取中）

```java
/**
 * LeetCode 912. 排序数组（快排版）
 * 面试标准答案：三数取中防退化
 */
public int[] sortArray(int[] nums) {
    quickSort(nums, 0, nums.length - 1);
    return nums;
}

private void quickSort(int[] arr, int lo, int hi) {
    if (lo >= hi) return;

    // ⚠️ 三数取中（防有序数组退化）
    int mid = lo + (hi - lo) / 2;
    if (arr[lo] > arr[mid]) swap(arr, lo, mid);
    if (arr[lo] > arr[hi]) swap(arr, lo, hi);
    if (arr[mid] > arr[hi]) swap(arr, mid, hi);
    swap(arr, mid, hi);

    // partition：小于基准放左边
    int pivot = arr[hi], i = lo - 1;
    for (int j = lo; j < hi; j++) {
        if (arr[j] < pivot) swap(arr, ++i, j);
    }
    swap(arr, i + 1, hi);

    quickSort(arr, lo, i);           // ⚠️ 基准在 i+1
    quickSort(arr, i + 2, hi);
}

private void swap(int[] arr, int i, int j) {
    int tmp = arr[i]; arr[i] = arr[j]; arr[j] = tmp;
}
// 平均 O(n log n)，最坏 O(n²)（三数取中后几乎不触发）
// ⚠️ 面试易错：递归边界 [lo,i] 与 [i+2,hi]（基准在 i+1）
```

### 1.2 归并版

```java
/**
 * 归并版（稳定，无退化）
 */
public int[] sortArray_Merge(int[] nums) {
    int[] temp = new int[nums.length];      // 全局辅助数组
    mergeSort(nums, 0, nums.length - 1, temp);
    return nums;
}

private void mergeSort(int[] arr, int lo, int hi, int[] temp) {
    if (lo >= hi) return;
    int mid = lo + (hi - lo) / 2;
    mergeSort(arr, lo, mid, temp);
    mergeSort(arr, mid + 1, hi, temp);
    merge(arr, lo, mid, hi, temp);
}

private void merge(int[] arr, int lo, int mid, int hi, int[] temp) {
    System.arraycopy(arr, lo, temp, lo, hi - lo + 1);
    int i = lo, j = mid + 1, k = lo;
    while (i <= mid && j <= hi) {
        if (temp[i] <= temp[j]) arr[k++] = temp[i++];   // <= 稳定
        else arr[k++] = temp[j++];
    }
    while (i <= mid) arr[k++] = temp[i++];
    while (j <= hi)  arr[k++] = temp[j++];
}
// 最好=最坏=平均 O(n log n)
// ⚠️ 面试追问："为什么归并没有退化？"
//   → 拆分与数据无关，递归树恒 log n 层
```

### 1.3 手撕选择

```
面试手撕顺序：快排 → 归并 → 堆排
  快排：三数取中 + partition（默认答案）
  归并：追问稳定性时展示
  堆排：追问 O(1) 空间时展示

⚠️ 写完用 [1,3,2,5,4] 快速验证
```

## 2. 215. 数组中的第 K 个最大元素

### 2.1 快速选择（平均 O(n)）

```java
/**
 * LeetCode 215. 数组中的第 K 个最大元素
 * 快选：partition 后只递归一侧
 * 第 K 大 = 第 n-K 小（0-indexed）
 */
public int findKthLargest(int[] nums, int k) {
    return quickSelect(nums, 0, nums.length - 1, nums.length - k);
}

private int quickSelect(int[] arr, int lo, int hi, int target) {
    if (lo == hi) return arr[lo];

    // ⚠️ 随机基准（防有序数组退化 O(n²)）
    int randomIdx = lo + new Random().nextInt(hi - lo + 1);
    swap(arr, randomIdx, hi);

    int pivot = arr[hi], i = lo - 1;
    for (int j = lo; j < hi; j++) {
        if (arr[j] < pivot) swap(arr, ++i, j);
    }
    swap(arr, i + 1, hi);

    if (i + 1 == target) return arr[i + 1];
    else if (i + 1 < target) return quickSelect(arr, i + 2, hi, target);
    else return quickSelect(arr, lo, i, target);
}
// 平均 O(n)：n + n/2 + n/4... = 2n（等比求和）
// 最坏 O(n²)：随机基准缓解
```

### 2.2 堆解法（O(n log k)）

```java
/**
 * 堆：小根堆维护最大的 k 个
 */
public int findKthLargest_Heap(int[] nums, int k) {
    PriorityQueue<Integer> minHeap = new PriorityQueue<>();
    for (int x : nums) {
        minHeap.offer(x);
        if (minHeap.size() > k) minHeap.poll();  // 弹最小
    }
    return minHeap.peek();   // 堆顶 = 第 k 大
}
// O(n log k) 时间 O(k) 空间
// ⚠️ 选型：静态数组快选更快；数据流用堆
```

## 3. 75. 颜色分类

### 3.1 三指针（荷兰国旗）

```java
/**
 * LeetCode 75. 颜色分类
 * 只有 0/1/2 → 三指针原地排序（O(n)/O(1)）
 */
public void sortColors(int[] nums) {
    int p0 = 0, p2 = nums.length - 1, curr = 0;

    while (curr <= p2) {
        if (nums[curr] == 0) {
            swap(nums, curr, p0);    // 0 交换到左边
            p0++;
            curr++;                  // ⚠️ curr 前进（换来的必是 0/1）
        } else if (nums[curr] == 2) {
            swap(nums, curr, p2);    // 2 交换到右边
            p2--;                    // ⚠️ curr 不前进（可能还是 2）
        } else {
            curr++;                  // 1 直接跳过
        }
    }
}
// 推演 [2,0,2,1,1,0]：
//   curr=0(2) 换 p2 → [0,0,2,1,1,2], p2=4（curr 不动）
//   curr=0(0) 换 p0 → [0,0,2,1,1,2], p0=1, curr=1
//   ... → [0,0,1,1,2,2] ✓
// ⚠️ 面试必讲："2 交换后 curr 不前进——
//   换回来的可能还是 2，要再检查一次"
```

### 3.2 计数排序版（另一种思路）

```java
/**
 * 计数版：0/1/2 计数后回填
 */
public void sortColors_Count(int[] nums) {
    int[] count = new int[3];
    for (int x : nums) count[x]++;

    int idx = 0;
    for (int i = 0; i < 3; i++) {
        while (count[i]-- > 0) {
            nums[idx++] = i;
        }
    }
}
// O(n) 时间 O(1) 空间（固定 3 个计数）
// ⚠️ 面试两种解法都能写 = 加分
```

## 4. 148. 排序链表

### 4.1 完整实现（归并）

```java
/**
 * LeetCode 148. 排序链表
 * 链表无随机访问 → 归并（O(n log n)）
 */
public ListNode sortList(ListNode head) {
    if (head == null || head.next == null) return head;

    // ① 快慢指针找中点并断开
    ListNode slow = head, fast = head;
    while (fast.next != null && fast.next.next != null) {
        slow = slow.next;
        fast = fast.next.next;
    }
    ListNode second = slow.next;
    slow.next = null;                    // ⚠️ 断开！

    // ② 递归排序两半
    ListNode left = sortList(head);
    ListNode right = sortList(second);

    // ③ 合并（21 题复用）
    return mergeTwoLists(left, right);
}

private ListNode mergeTwoLists(ListNode l1, ListNode l2) {
    ListNode dummy = new ListNode(-1);
    ListNode tail = dummy;
    while (l1 != null && l2 != null) {
        if (l1.val <= l2.val) { tail.next = l1; l1 = l1.next; }
        else { tail.next = l2; l2 = l2.next; }
        tail = tail.next;
    }
    tail.next = (l1 != null) ? l1 : l2;
    return dummy.next;
}
// O(n log n) 时间 O(log n) 空间（递归栈）
// ⚠️ 面试必答："为什么链表选归并？
//   快排 partition 需要双向遍历（链表低效）；
//   归并只需 next 指针 → 天然适配"
```

## 5. 347. 前 K 个高频元素

### 5.1 频次桶（O(n)）

```java
/**
 * LeetCode 347. 前 K 个高频元素
 * 频次桶：下标 = 频次 → 免排序 O(n)
 */
public int[] topKFrequent(int[] nums, int k) {
    // ① 计数
    Map<Integer, Integer> freq = new HashMap<>();
    for (int x : nums) freq.merge(x, 1, Integer::sum);

    // ② 频次桶
    List<Integer>[] buckets = new List[nums.length + 1];
    for (var e : freq.entrySet()) {
        int f = e.getValue();
        if (buckets[f] == null) buckets[f] = new ArrayList<>();
        buckets[f].add(e.getKey());
    }

    // ③ 从高频往低频收集
    int[] result = new int[k];
    int idx = 0;
    for (int f = buckets.length - 1; f > 0 && idx < k; f--) {
        if (buckets[f] != null) {
            for (int x : buckets[f]) result[idx++] = x;
        }
    }
    return result;
}
// O(n)：计数 + 分桶 + 收集（免排序）
// ⚠️ 对比堆 O(n log k)：频次桶更快（静态数组）
```

### 5.2 堆解法

```java
/**
 * 堆：计数 + 小根堆（堆顶是第 k 高频）
 */
public int[] topKFrequent_Heap(int[] nums, int k) {
    Map<Integer, Integer> freq = new HashMap<>();
    for (int x : nums) freq.merge(x, 1, Integer::sum);

    PriorityQueue<Integer> heap =
        new PriorityQueue<>((a, b) -> freq.get(a) - freq.get(b));
    for (int key : freq.keySet()) {
        heap.offer(key);
        if (heap.size() > k) heap.poll();
    }
    return heap.stream().mapToInt(Integer::intValue).toArray();
}
// O(n log k) 时间 O(k) 空间
// ⚠️ 选型：数据流/在线 → 堆；静态 → 频次桶
```

## 6. 剑指51. 数组中的逆序对

### 6.1 完整实现（归并计数）

```java
/**
 * 剑指 Offer 51. 数组中的逆序对
 * 归并排序的隐藏能力：合并时统计
 * 右半先入 → 与左半剩余 (mid-i+1) 个都构成逆序对
 */
public int reversePairs(int[] nums) {
    int[] temp = new int[nums.length];
    return mergeSort(nums, 0, nums.length - 1, temp);
}

private int mergeSort(int[] arr, int lo, int hi, int[] temp) {
    if (lo >= hi) return 0;

    int mid = lo + (hi - lo) / 2;
    int count = mergeSort(arr, lo, mid, temp)       // 左半内部
              + mergeSort(arr, mid + 1, hi, temp);  // 右半内部

    // 合并时统计跨两半的逆序对
    System.arraycopy(arr, lo, temp, lo, hi - lo + 1);
    int i = lo, j = mid + 1, k = lo;
    while (i <= mid && j <= hi) {
        if (temp[i] <= temp[j]) {
            arr[k++] = temp[i++];
        } else {
            count += mid - i + 1;      // ⚠️ 灵魂：左半剩余都是逆序对
            arr[k++] = temp[j++];
        }
    }
    while (i <= mid) arr[k++] = temp[i++];
    while (j <= hi)  arr[k++] = temp[j++];
    return count;
}
// 推演 [7,5,6,4]：
//   左半 [7,5]：合并时 5<7 → +1
//   右半 [6,4]：合并时 4<6 → +1
//   跨半：4 先入 → 与左半 7,5 → +2；6 后入 → 与 7 → +1
//   总 = 5 ✓
// ⚠️ 为什么"右半先入 = 逆序对"？
//   temp[j] < temp[i] → 小于左半剩余全部
//   → 构成 (mid-i+1) 个逆序对
```

## 7. 56. 合并区间

### 7.1 完整实现（排序 + 扫描）

```java
/**
 * LeetCode 56. 合并区间
 * 按开始排序 → 覆盖式扫描
 */
public int[][] merge(int[][] intervals) {
    Arrays.sort(intervals, (a, b) -> a[0] - b[0]);   // ⚠️ 按开始排序

    List<int[]> result = new ArrayList<>();
    int[] current = intervals[0];        // 当前合并区间

    for (int i = 1; i < intervals.length; i++) {
        if (intervals[i][0] <= current[1]) {
            // 重叠 → 扩展右边界（取较大结束）
            current[1] = Math.max(current[1], intervals[i][1]);
        } else {
            result.add(current);         // 不重叠 → 输出
            current = intervals[i];      // 开始新区间
        }
    }
    result.add(current);                 // ⚠️ 最后一个
    return result.toArray(new int[0][]);
}
// 推演 [[1,3],[2,6],[8,10],[15,18]]：
//   [2,6] 与 [1,3] 重叠 → [1,6]
//   [8,10] 不重叠 → 输出 [1,6]
//   → [[1,6],[8,10],[15,18]] ✓
// ⚠️ 为什么按开始排序？
//   从早到晚扫描才能顺序合并（按结束会漏合并）
```

## 8. 493. 翻转对

### 8.1 完整实现（归并变体）

```java
/**
 * LeetCode 493. 翻转对
 * i < j 且 nums[i] > 2 × nums[j]
 * 归并变体：合并前先统计（双指针），再正常合并
 */
public int reversePairs(int[] nums) {
    int[] temp = new int[nums.length];
    return mergeSort(nums, 0, nums.length - 1, temp);
}

private int mergeSort(int[] arr, int lo, int hi, int[] temp) {
    if (lo >= hi) return 0;

    int mid = lo + (hi - lo) / 2;
    int count = mergeSort(arr, lo, mid, temp)
              + mergeSort(arr, mid + 1, hi, temp);

    // ⚠️ 合并前先统计（两半已各自有序）
    int i = lo, j = mid + 1;
    while (i <= mid && j <= hi) {
        if ((long) arr[i] > 2L * arr[j]) {   // ⚠️ long 防溢出
            count += mid - i + 1;            // 左半剩余都满足
            j++;
        } else {
            i++;
        }
    }

    // 正常合并（与标准归并相同）
    System.arraycopy(arr, lo, temp, lo, hi - lo + 1);
    i = lo; j = mid + 1;
    int k = lo;
    while (i <= mid && j <= hi) {
        if (temp[i] <= temp[j]) arr[k++] = temp[i++];
        else arr[k++] = temp[j++];
    }
    while (i <= mid) arr[k++] = temp[i++];
    while (j <= hi)  arr[k++] = temp[j++];

    return count;
}
// 推演 [1,3,2,3,1]：
//   翻转对：(3,1),(2,1),(3,1) → 3 ✓
// ⚠️ 与剑指 51 的区别：
//   51：合并时统计（条件 temp[i] > temp[j]）
//   493：合并前统计（条件 arr[i] > 2×arr[j]）
//   → 493 需要"双指针 + 两半有序"前提
```

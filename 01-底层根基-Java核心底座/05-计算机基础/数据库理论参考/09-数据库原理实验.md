## 1. 实验一：数据库与表创建（DDL）

### 1.1 创建数据库

```sql
CREATE DATABASE bookstore CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE bookstore;
```

### 1.2 创建表与约束

```sql
-- 用户表
CREATE TABLE users (
    id         BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '主键',
    username   VARCHAR(50)  NOT NULL UNIQUE COMMENT '用户名',
    email      VARCHAR(100) NOT NULL COMMENT '邮箱',
    age        TINYINT      DEFAULT 0 COMMENT '年龄',
    phone      VARCHAR(20)  COMMENT '手机号',
    created_at DATETIME     DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at DATETIME     DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

-- 订单表
CREATE TABLE orders (
    id         BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_id    BIGINT      NOT NULL,
    amount     DECIMAL(10,2) DEFAULT 0.00,
    status     TINYINT     DEFAULT 0,
    created_at DATETIME    DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

### 1.3 修改表结构

```sql
ALTER TABLE users ADD COLUMN address VARCHAR(200);
ALTER TABLE users MODIFY COLUMN age TINYINT DEFAULT 18;
ALTER TABLE users CHANGE COLUMN phone mobile VARCHAR(20);
ALTER TABLE users DROP COLUMN address;
```

### 1.4 创建与删除索引

```sql
CREATE INDEX idx_username ON users(username);
CREATE UNIQUE INDEX idx_email ON users(email);
CREATE INDEX idx_user_created ON users(created_at);
DROP INDEX idx_username ON users;
```

## 2. 实验二：单表查询（DQL）

### 2.1 基本查询

```sql
-- 插入测试数据
INSERT INTO users (username, email, age) VALUES
('zhangsan', 'zhang@test.com', 25),
('lisi', 'li@test.com', 30),
('wangwu', 'wang@test.com', 28),
('zhaoliu', 'zhao@test.com', 22);

-- 基本 SELECT
SELECT username, email FROM users;
SELECT * FROM users;            -- 生产环境避免 *
SELECT DISTINCT age FROM users; -- 去重
SELECT username AS name FROM users; -- 别名

-- WHERE 条件
SELECT * FROM users WHERE age > 25;
SELECT * FROM users WHERE age BETWEEN 22 AND 28;
SELECT * FROM users WHERE username LIKE 'zhang%';
SELECT * FROM users WHERE email IS NOT NULL;

-- 排序
SELECT * FROM users ORDER BY age DESC, username ASC;

-- 分页
SELECT * FROM users ORDER BY id LIMIT 2 OFFSET 0;
SELECT * FROM users ORDER BY id LIMIT 2, 2;
```

### 2.2 聚合与分组

```sql
-- 聚合
SELECT COUNT(*) FROM users;
SELECT AVG(age), MAX(age), MIN(age) FROM users;

-- 分组
SELECT age, COUNT(*) AS cnt FROM users GROUP BY age;

-- HAVING
SELECT age, COUNT(*) AS cnt FROM users
GROUP BY age HAVING cnt >= 1;
```

## 3. 实验三：多表查询（JOIN/子查询）

### 3.1 插入多表数据

```sql
INSERT INTO orders (user_id, amount, status) VALUES
(1, 100.00, 1), (1, 200.00, 0), (2, 150.00, 1), (3, 300.00, 1);
```

### 3.2 JOIN 查询

```sql
-- 内连接
SELECT u.username, o.amount, o.status
FROM users u INNER JOIN orders o ON u.id = o.user_id;

-- 左外连接（含未下单用户）
SELECT u.username, o.amount
FROM users u LEFT JOIN orders o ON u.id = o.user_id;

-- 聚合 JOIN
SELECT u.username, COUNT(o.id) AS order_count, SUM(o.amount) AS total
FROM users u LEFT JOIN orders o ON u.id = o.user_id
GROUP BY u.id;
```

### 3.3 子查询

```sql
-- 订单金额高于平均的用户
SELECT * FROM users WHERE id IN (
    SELECT user_id FROM orders WHERE amount > (SELECT AVG(amount) FROM orders)
);

-- EXISTS
SELECT * FROM users u WHERE EXISTS (
    SELECT 1 FROM orders o WHERE o.user_id = u.id AND o.amount > 200
);
```

## 4. 实验四：数据操作与事务

### 4.1 DML

```sql
-- 插入
INSERT INTO users (username, email) VALUES ('test1', 't1@test.com');
INSERT INTO users (username, email) VALUES ('test2', 't2@test.com'), ('test3', 't3@test.com');

-- 更新
UPDATE users SET age = 26 WHERE username = 'zhangsan';

-- 删除
DELETE FROM users WHERE username = 'test1';

-- 逻辑删除推荐
UPDATE users SET status = 0 WHERE id = 2;
```

### 4.2 事务

```sql
START TRANSACTION;
UPDATE users SET age = age + 1 WHERE id = 1;
UPDATE users SET age = age - 1 WHERE id = 2;
COMMIT;

-- 回滚测试
START TRANSACTION;
DELETE FROM users WHERE id = 3;
ROLLBACK;  -- 数据恢复

-- 保存点
START TRANSACTION;
INSERT INTO users(username, email) VALUES('a', 'a@test.com');
SAVEPOINT sp1;
INSERT INTO users(username, email) VALUES('b', 'b@test.com');
ROLLBACK TO sp1;  -- b 被回滚，a 还在
COMMIT;
```

## 5. 实验五：索引与视图

### 5.1 索引实验

```sql
-- 创建测试大表
CREATE TABLE big_table (
    id INT PRIMARY KEY AUTO_INCREMENT,
    val VARCHAR(100),
    created_at DATETIME
);

-- 插入 10 万条数据测试
-- 未加索引的查询
SELECT * FROM big_table WHERE val = 'test';  -- 全表扫描

-- 加索引后
CREATE INDEX idx_val ON big_table(val);
SELECT * FROM big_table WHERE val = 'test';  -- 索引查找

-- 查看执行计划
EXPLAIN SELECT * FROM big_table WHERE val = 'test';
```

### 5.2 视图

```sql
-- 创建视图
CREATE VIEW v_user_order_summary AS
SELECT u.id, u.username, COUNT(o.id) AS order_count, SUM(o.amount) AS total_amount
FROM users u LEFT JOIN orders o ON u.id = o.user_id
GROUP BY u.id;

-- 使用视图
SELECT * FROM v_user_order_summary WHERE total_amount > 100;
```

## 6. 实验六：用户权限管理

```sql
-- 创建用户
CREATE USER 'app_user'@'localhost' IDENTIFIED BY 'password123';
CREATE USER 'readonly_user'@'%' IDENTIFIED BY 'readonly_pass';

-- 授权
GRANT SELECT, INSERT, UPDATE, DELETE ON bookstore.* TO 'app_user'@'localhost';
GRANT SELECT ON bookstore.* TO 'readonly_user'@'%';

-- 查看权限
SHOW GRANTS FOR 'app_user'@'localhost';
SHOW GRANTS FOR 'readonly_user'@'%';

-- 撤销权限
REVOKE DELETE ON bookstore.* FROM 'app_user'@'localhost';

-- 修改密码
ALTER USER 'app_user'@'localhost' IDENTIFIED BY 'new_password';

-- 删除用户
DROP USER 'readonly_user'@'%';

-- 刷新权限
FLUSH PRIVILEGES;
```

> 🎯 **每个实验都值得亲手操作一遍**——数据库是实践性很强的学科，光看不练无法真正掌握。

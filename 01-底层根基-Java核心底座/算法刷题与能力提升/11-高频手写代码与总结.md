// 1. 反转链表（5 行）
public ListNode reverseList(ListNode head) {
    ListNode prev = null, curr = head;
    while (curr != null) {
        ListNode next = curr.next; curr.next = prev; prev = curr; curr = next;
    }
    return prev;
}

// 2. 二分查找（5 行）
public int binarySearch(int[] nums, int target) {
    int l = 0, r = nums.length - 1;
    while (l <= r) {
        int m = l + (r - l) / 2;
        if (nums[m] == target) return m;
        else if (nums[m] < target) l = m + 1; else r = m - 1;
    }
    return -1;
}

// 3. 快排核心 partition（6 行）
int partition(int[] a, int l, int r) {
    int pivot = a[r], i = l;
    for (int j = l; j < r; j++) if (a[j] < pivot) swap(a, i++, j);
    swap(a, i, r);
    return i;
}

// 4. 二叉树最大深度（1 行）
public int maxDepth(TreeNode root) {
    return root == null ? 0 : 1 + Math.max(maxDepth(root.left), maxDepth(root.right));
}

// 5. 有效括号（8 行）
public boolean isValid(String s) {
    Deque<Character> st = new ArrayDeque<>();
    var m = Map.of(')', '(', ']', '[', '}', '{');
    for (char c : s.toCharArray()) {
        if (m.containsKey(c)) { if (st.isEmpty() || st.pop() != m.get(c)) return false; }
        else st.push(c);
    }
    return st.isEmpty();
}

// 6. 两数之和 HashMap（5 行）
public int[] twoSum(int[] nums, int target) {
    Map<Integer, Integer> map = new HashMap<>();
    for (int i = 0; i < nums.length; i++) {
        int c = target - nums[i];
        if (map.containsKey(c)) return new int[]{map.get(c), i};
        map.put(nums[i], i);
    }
    return new int[]{-1, -1};
}

// 7. 无重复最长子串（滑动窗口，8 行）
public int lengthOfLongestSubstring(String s) {
    Set<Character> set = new HashSet<>();
    int l = 0, max = 0;
    for (int r = 0; r < s.length(); r++) {
        while (set.contains(s.charAt(r))) set.remove(s.charAt(l++));
        set.add(s.charAt(r));
        max = Math.max(max, r - l + 1);
    }
    return max;
}

// 8. 爬楼梯 DP（5 行）
public int climbStairs(int n) {
    if (n <= 2) return n;
    int[] dp = new int[n + 1]; dp[1] = 1; dp[2] = 2;
    for (int i = 3; i <= n; i++) dp[i] = dp[i - 1] + dp[i - 2];
    return dp[n];
}

// 9. 二叉树中序遍历（迭代，8 行）
public List<Integer> inorder(TreeNode root) {
    List<Integer> res = new ArrayList<>();
    Deque<TreeNode> st = new ArrayDeque<>();
    TreeNode cur = root;
    while (cur != null || !st.isEmpty()) {
        while (cur != null) { st.push(cur); cur = cur.left; }
        cur = st.pop(); res.add(cur.val); cur = cur.right;
    }
    return res;
}

// 10. 全排列（回溯，12 行）
void permute(List<List<Integer>> res, List<Integer> path, int[] nums, boolean[] used) {
    if (path.size() == nums.length) { res.add(new ArrayList<>(path)); return; }
    for (int i = 0; i < nums.length; i++) {
        if (used[i]) continue;
        used[i] = true; path.add(nums[i]);
        permute(res, path, nums, used);
        used[i] = false; path.removeLast();
    }
}

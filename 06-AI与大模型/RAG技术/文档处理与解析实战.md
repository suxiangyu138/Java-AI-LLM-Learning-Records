# 文档处理与解析实战

> **核心认知**：RAG 系统的数据源 80% 是非结构化的 PDF 和 Markdown 文档。文档加载和切分的质量直接决定检索效果的上限。
> **前置知识**：LangChain Document Loaders / Text Splitters 基础

## 1. 文档处理全景

```
原始文档
   │
   ├── PDF ─→ PyPDF2 / pdfplumber ─→ 纯文本
   ├── Markdown ─→ markdown 库 ─→ 结构化内容
   ├── 网页 ─→ BeautifulSoup ─→ 清洗后的文本
   ├── Word ─→ python-docx ─→ 纯文本
   └── 代码 ─→ AST 解析 ─→ 函数/类级别切分
        │
        └──→ Text Splitter ─→ Chunks ─→ Embedding ─→ 向量数据库
```

```python
from PyPDF2 import PdfReader


def extract_pdf_text(file_path: str) -> list[dict]:
    """提取 PDF 每页文本 + 页码"""
    reader = PdfReader(file_path)
    pages = []
    for i, page in enumerate(reader.pages):
        text = page.extract_text()
        if text.strip():  # 过滤空页
            pages.append({
                "page": i + 1,
                "content": text.strip(),
                "source": file_path
            })
    return pages


# 批量处理
import os

def load_all_pdfs(pdf_dir: str) -> list[str]:
    """加载目录下所有 PDF 的文本"""
    all_texts = []
    for fname in os.listdir(pdf_dir):
        if fname.endswith(".pdf"):
            pages = extract_pdf_text(os.path.join(pdf_dir, fname))
            for p in pages:
                all_texts.append(
                    f"[来源: {fname}, 第{p['page']}页]\n{p['content']}"
                )
    return all_texts
```

```python
import pdfplumber


def extract_with_tables(file_path: str) -> list[dict]:
    """提取 PDF 文本 + 表格"""
    results = []
    with pdfplumber.open(file_path) as pdf:
        for i, page in enumerate(pdf.pages):
            text = page.extract_text() or ""

            # 提取表格
            tables = page.extract_tables()
            table_texts = []
            for table in tables:
                # 表格 → Markdown 格式（LLM 更容易理解）
                md_table = _table_to_markdown(table)
                table_texts.append(md_table)

            results.append({
                "page": i + 1,
                "text": text,
                "tables": table_texts,
                "combined": text + "\n\n" + "\n\n".join(table_texts)
            })
    return results


def _table_to_markdown(table: list[list[str]]) -> str:
    """将二维数组表格转换为 Markdown 格式"""
    if not table:
        return ""
    lines = []
    # 表头
    lines.append("| " + " | ".join(table[0]) + " |")
    lines.append("| " + " | ".join(["---"] * len(table[0])) + " |")
    # 数据行
    for row in table[1:]:
        lines.append("| " + " | ".join(str(cell or "") for cell in row) + " |")
    return "\n".join(lines)
```

```python
def extract_images_from_pdf(pdf_path: str, output_dir: str):
    """提取 PDF 中的图片"""
    doc = fitz.open(pdf_path)
    for i, page in enumerate(doc):
        images = page.get_images(full=True)
        for j, img in enumerate(images):
            xref = img[0]
            base_image = doc.extract_image(xref)
            image_bytes = base_image["image"]
            ext = base_image["ext"]
            img_path = f"{output_dir}/page{i+1}_img{j+1}.{ext}"
            with open(img_path, "wb") as f:
                f.write(image_bytes)
```

## 3. Markdown 解析

### 3.1 按标题层级切分

```python
import re
from typing import Optional


def parse_markdown_sections(md_text: str) -> list[dict]:
    """按 # 标题层级解析 Markdown，保留层级关系"""
    sections = []
    current_section: Optional[dict] = None
    current_content: list[str] = []

    for line in md_text.split("\n"):
        header_match = re.match(r"^(#{1,6})\s+(.+)", line)
        if header_match:
            # 保存上一节
            if current_section:
                current_section["content"] = "\n".join(current_content).strip()

            level = len(header_match.group(1))
            title = header_match.group(2)
            current_section = {"level": level, "title": title, "content": ""}
            sections.append(current_section)
            current_content = []
        else:
            current_content.append(line)

    # 保存最后一节
    if current_section:
        current_section["content"] = "\n".join(current_content).strip()

    return sections


# 使用
md = """
# Spring Boot 实战
## 第一章 快速开始
Spring Boot 让创建独立的应用变得简单。
### 1.1 环境准备
需要 JDK 17+ 和 Maven 3.6+。
## 第二章 自动配置
自动配置是 Spring Boot 的核心特性。
"""

sections = parse_markdown_sections(md)
for s in sections:
    print(f"{'#' * s['level']} {s['title']}: {len(s['content'])} chars")
```

### 3.2 代码块提取

```python
def extract_code_blocks(md_text: str) -> list[dict]:
    """提取 Markdown 中的代码块"""
    pattern = r"```(\w+)?\n(.*?)```"
    blocks = []
    for match in re.finditer(pattern, md_text, re.DOTALL):
        blocks.append({
            "language": match.group(1) or "text",
            "code": match.group(2).strip(),
            "start": match.start(),
        })
    return blocks
```

```python
import requests
from bs4 import BeautifulSoup


def scrape_page(url: str) -> dict:
    """爬取网页并提取主要内容"""
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
    }
    response = requests.get(url, headers=headers, timeout=10)
    response.encoding = response.apparent_encoding  # 自动检测编码
    soup = BeautifulSoup(response.text, "lxml")

    # 移除无关元素
    for tag in soup(["script", "style", "nav", "footer", "header"]):
        tag.decompose()

    # 提取标题
    title = soup.title.string if soup.title else ""

    # 提取正文（优先 article > main > body）
    content_tag = (
        soup.find("article") or
        soup.find("main") or
        soup.find("div", class_="content") or
        soup.body
    )
    text = content_tag.get_text(separator="\n", strip=True) if content_tag else ""

    # 提取所有链接
    links = [
        {"text": a.get_text(strip=True), "href": a.get("href")}
        for a in soup.find_all("a", href=True)
    ]

    return {"title": title, "content": text, "links": links, "url": url}
```

### 5.2 中文文本专项切分

```python
import re


class ChineseTextSplitter:
    """中文文本切分器——按语义边界切分，避免截断句子"""

    def __init__(self, chunk_size: int = 500, chunk_overlap: int = 50):
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap

    def split(self, text: str) -> list[str]:
        # 按中文句末标点切分句子
        sentences = re.split(r"(?<=[。！？；\n])\s*", text)
        sentences = [s.strip() for s in sentences if s.strip()]

        chunks = []
        current_chunk = ""
        for sentence in sentences:
            if len(current_chunk) + len(sentence) <= self.chunk_size:
                current_chunk += sentence
            else:
                if current_chunk:
                    chunks.append(current_chunk)
                # 保持重叠
                current_chunk = current_chunk[-self.chunk_overlap:] + sentence if current_chunk else sentence

        if current_chunk:
            chunks.append(current_chunk)
        return chunks
```

## 6. 完整文档处理流水线

```python
import os
from typing import Callable
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_community.document_loaders import PyPDFLoader, TextLoader


class DocumentPipeline:
    """文档处理流水线"""

    def __init__(
        self,
        chunk_size: int = 500,
        chunk_overlap: int = 50,
        file_types: dict[str, Callable] = None
    ):
        self.splitter = RecursiveCharacterTextSplitter(
            chunk_size=chunk_size,
            chunk_overlap=chunk_overlap,
            separators=["\n## ", "\n", "。", ".", " "]
        )
        self.file_types = file_types or {
            ".pdf": self._load_pdf,
            ".md": self._load_text,
            ".txt": self._load_text,
        }

    def process_directory(self, root_dir: str) -> list[dict]:
        """递归处理目录下所有支持的文件"""
        all_results = []
        for dirpath, _, filenames in os.walk(root_dir):
            for fname in filenames:
                ext = os.path.splitext(fname)[1].lower()
                if ext in self.file_types:
                    file_path = os.path.join(dirpath, fname)
                    try:
                        results = self.file_types[ext](file_path)
                        all_results.extend(results)
                        print(f"√ {file_path} → {len(results)} chunks")
                    except Exception as e:
                        print(f"× {file_path}: {e}")
        return all_results

    def _load_pdf(self, path: str) -> list[dict]:
        loader = PyPDFLoader(path)
        docs = loader.load()
        chunks = self.splitter.split_documents(docs)
        return [{"content": c.page_content, "source": path, "page": c.metadata.get("page")} for c in chunks]

    def _load_text(self, path: str) -> list[dict]:
        loader = TextLoader(path, encoding="utf-8")
        docs = loader.load()
        chunks = self.splitter.split_documents(docs)
        return [{"content": c.page_content, "source": path} for c in chunks]
```

## 7. 文档质量检查清单

| 检查项 | 方法 | 标准 |
|--------|------|------|
| 是否残留乱码 | 抽样人工检查 | 乱码率 < 1% |
| Chunk 是否完整 | 检查 chunk 的起止位置是否切割句子 | 语义完整率 > 95% |
| 是否遗漏页面 | 对比原始 PDF 页数 vs 提取页数 | 页面覆盖率 100% |
| 表格是否正确解析 | 检查表格是否被正确转为 Markdown | 表头 + 数据无缺失 |
| 代码块是否保留 | 检查 code block 是否完整 | 代码块保留率 100% |

## 快速调试检查清单

- [ ] `PyPDF2` 提取中文是否乱码？（尝试 `pdfplumber` 或 `pymupdf`）
- [ ] Chunk 大小是否合适？（太小缺上下文，太大超模型 Token 限制）
- [ ] Overlap 是否合理？（一般 chunk_size 的 10-20%）
- [ ] BeautifulSoup 的目标网页是否有反爬？（加 User-Agent、延迟请求）

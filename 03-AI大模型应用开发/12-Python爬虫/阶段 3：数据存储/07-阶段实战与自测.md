## 1. 综合实战：增量新闻爬虫 + 数据库 + 清洗流水线

```python
"""
incremental_spider.py —— 本阶段综合实战
功能：增量新闻爬虫（任务状态机 + MySQL 去重 + pandas 清洗导出）
"""
import random
import time
import pymysql
import pandas as pd
import requests
from bs4 import BeautifulSoup
from contextlib import closing

# ═══ ① 配置 ═══
DB_CONFIG = dict(host="localhost", user="root", password="xxx",
                 database="crawler_db", charset="utf8mb4",
                 cursorclass=pymysql.cursors.DictCursor)
BASE_URL = "https://example.com/news"       # 练习站点
HEADERS = {"User-Agent": "StudyCrawler/1.0 (learning)"}

# ═══ ② 建表（幂等）═══
def init_db():
    with closing(pymysql.connect(**DB_CONFIG)) as conn:
        with conn.cursor() as cur:
            cur.execute("""
                CREATE TABLE IF NOT EXISTS news (
                    id BIGINT AUTO_INCREMENT PRIMARY KEY,
                    title VARCHAR(512) NOT NULL,
                    url VARCHAR(1024) NOT NULL,
                    content MEDIUMTEXT,
                    views INT DEFAULT 0,
                    crawl_time DATETIME DEFAULT CURRENT_TIMESTAMP,
                    UNIQUE KEY uk_url (url(255))
                ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
            """)
            cur.execute("""
                CREATE TABLE IF NOT EXISTS crawl_tasks (
                    id BIGINT AUTO_INCREMENT PRIMARY KEY,
                    url VARCHAR(1024) NOT NULL,
                    status TINYINT DEFAULT 0,
                    retry TINYINT DEFAULT 0,
                    UNIQUE KEY uk_url (url(255))
                ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
            """)
        conn.commit()

# ═══ ③ 列表页发现新 URL（入队）═══
def discover_urls(cur):
    session = requests.Session()
    session.headers.update(HEADERS)
    new_urls = []
    for page in range(1, 6):                      # 前 5 页
        resp = session.get(BASE_URL, params={"page": page}, timeout=10)
        if resp.status_code != 200:
            continue
        soup = BeautifulSoup(resp.text, "lxml")
        items = soup.select("ul.news-list li a")
        urls = [a.get("href", "") for a in items if a.get("href")]
        new_urls.extend(urls)
        time.sleep(0.5)
    # 入队（INSERT IGNORE：已存在的跳过）
    cur.executemany(
        "INSERT IGNORE INTO crawl_tasks (url) VALUES (%s)",
        [(u,) for u in new_urls],
    )
    conn.commit() if False else None

# ═══ ④ 抓取详情 + 入库 ═══
def crawl_detail(cur, url):
    session = requests.Session()
    session.headers.update(HEADERS)
    resp = session.get(url, timeout=(3, 15))
    resp.raise_for_status()
    soup = BeautifulSoup(resp.text, "lxml")

    title = soup.select_one("h1.title")
    content = soup.select_one("div.article-content")
    title = title.get_text(strip=True) if title else url
    content = " ".join(content.get_text(strip=True).split()) if content else ""

    cur.execute(
        "INSERT IGNORE INTO news (title, url, content) VALUES (%s, %s, %s)",
        (title, url, content),
    )
    return cur.rowcount == 1                      # True = 新插入

# ═══ ⑤ 主流程：任务状态机 ═══
def run():
    init_db()
    with closing(pymysql.connect(**DB_CONFIG)) as conn:
        with conn.cursor() as cur:
            discover_urls(cur)
            conn.commit()

            # 处理待抓任务（每轮 100 条）
            for _ in range(3):
                cur.execute("SELECT url FROM crawl_tasks WHERE status = 0 LIMIT 100")
                tasks = cur.fetchall()
                if not tasks:
                    break
                for task in tasks:
                    try:
                        new = crawl_detail(cur, task["url"])
                        cur.execute(
                            "UPDATE crawl_tasks SET status = 2 WHERE url = %s",
                            (task["url"],))
                        if new:
                            print(f"✅ 新入库: {task['url']}")
                    except requests.RequestException:
                        cur.execute(
                            "UPDATE crawl_tasks SET status = 3 WHERE url = %s",
                            (task["url"],))
                    conn.commit()
                    time.sleep(random.uniform(0.5, 1))

            # 统计
            cur.execute("SELECT status, COUNT(*) c FROM crawl_tasks GROUP BY status")
            print("任务统计:", cur.fetchall())

# ═══ ⑥ pandas 清洗导出 ═══
def export_clean():
    with closing(pymysql.connect(**DB_CONFIG)) as conn:
        df = pd.read_sql("SELECT * FROM news", conn)
    df["title"] = df["title"].str.replace("\xa0", " ").str.strip()
    df["views"] = pd.to_numeric(df["views"], errors="coerce").fillna(0).astype(int)
    df = df.drop_duplicates(subset=["url"], keep="first")
    df.to_csv("news_clean.csv", index=False, encoding="utf-8-sig")
    print(f"清洗导出: {len(df)} 条 → news_clean.csv")

if __name__ == "__main__":
    run()
    export_clean()
```

## 2. 运行与验证

```bash
# 前置：本地 MySQL 建库 crawler_db
python incremental_spider.py

# 预期流程
# ① 建表（不存在则创建）
# ② 发现 URL 入队 → 抓取 → 状态流转
# ③ 任务统计: [{'status': 0, ...}, {'status': 2, ...}, ...]
# ④ 清洗导出: N 条 → news_clean.csv

# 验证（重跑第二次）：
#  · 任务统计 status=2 增多、status=0 减少（增量生效）
#  · 数据无重复（UNIQUE 去重生效）
#  · news_clean.csv Excel 打开中文正常
```

| 验证项 | 检查 |
|--------|------|
| 幂等 | 重跑两次结果一致（无重复入库） |
| 增量 | 第二次只抓新 URL（任务状态机） |
| 去重 | SELECT COUNT(*) 与 COUNT(DISTINCT url) 相等 |
| 失败处理 | 模拟断网 → status=3 → 重跑自动捞回 |
| 清洗 | CSV 无 \xa0、views 是数字 |

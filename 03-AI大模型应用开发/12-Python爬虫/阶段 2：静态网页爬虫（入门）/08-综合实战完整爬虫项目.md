## 2. 完整代码

```python
"""
news_spider.py —— 静态网页爬虫综合实战
结构：请求模块 / 解析模块 / 存储模块 / 主流程
"""
import csv
import random
import time
import requests
from bs4 import BeautifulSoup
from urllib.parse import urljoin

# ═══════════ ① 请求模块 ═══════════
HEADERS = {
    "User-Agent": ("Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                   "AppleWebKit/537.36 (KHTML, like Gecko) "
                   "Chrome/126.0.0.0 Safari/537.36"),
    "Accept-Language": "zh-CN,zh;q=0.9",
}
BASE_URL = "https://example.com"          # 替换为练习站点

session = requests.Session()
session.headers.update(HEADERS)


def fetch(url: str, retries: int = 3) -> str:
    """请求 + 重试 + 礼貌间隔"""
    for attempt in range(retries):
        try:
            resp = session.get(url, timeout=(3, 15))
            resp.raise_for_status()
            time.sleep(random.uniform(0.5, 1.5))     # 随机间隔（更像人）
            return resp.text
        except requests.RequestException as e:
            if attempt == retries - 1:
                print(f"请求失败（{retries} 次）: {url} — {e}")
                return ""
            time.sleep(2 ** attempt)


# ═══════════ ② 解析模块 ═══════════
def parse_list(html_text: str) -> list:
    """解析列表页：返回 [{title, url, date}]"""
    soup = BeautifulSoup(html_text, "lxml")
    items = []
    for li in soup.select("ul.news-list li"):
        title_el = li.select_one("a.news-title")
        date_el = li.select_one("span.date")
        if not title_el:
            continue
        items.append({
            "title": title_el.get_text(strip=True).replace("\xa0", " "),
            "url": urljoin(BASE_URL, title_el.get("href", "")),   # 链接补全
            "date": date_el.get_text(strip=True) if date_el else "",
        })
    return items


def parse_detail(html_text: str) -> str:
    """解析详情页：返回正文文本"""
    soup = BeautifulSoup(html_text, "lxml")
    content = soup.select_one("div.article-content")
    if not content:
        return ""
    return " ".join(content.get_text(strip=True).split())   # 空白归一


# ═══════════ ③ 存储模块 ═══════════
def save_csv(filename: str, rows: list, fieldnames: list) -> None:
    """CSV 存储（utf-8-sig 兼容 Excel）"""
    with open(filename, "w", newline="", encoding="utf-8-sig") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(rows)
    print(f"已保存 {len(rows)} 条 → {filename}")


# ═══════════ ④ 主流程 ═══════════
def main():
    # 1. 列表页翻页采集（前 5 页）
    list_rows = []
    for page in range(1, 6):
        html = fetch(f"{BASE_URL}/news?page={page}")
        if not html:
            continue
        items = parse_list(html)
        list_rows.extend(items)
        print(f"第 {page} 页：{len(items)} 条，累计 {len(list_rows)}")
        if not items:
            break                      # 空页终止

    # 2. 详情页逐条采集（每篇抓正文）
    seen = set()                       # URL 去重
    for row in list_rows:
        if row["url"] in seen:
            continue
        seen.add(row["url"])
        detail_html = fetch(row["url"])
        row["content"] = parse_detail(detail_html) if detail_html else ""
        print(f"详情: {row['title'][:20]}... ({len(row['content'])} 字)")

    # 3. 存储
    save_csv("news_list.csv", list_rows,
             ["title", "url", "date"])
    save_csv("news_detail.csv", list_rows,
             ["title", "url", "date", "content"])
    print("✅ 采集完成")


if __name__ == "__main__":
    main()
```

## 3. 运行与验证

```bash
# 运行
python news_spider.py

# 第 1 页：20 条，累计 20
# 第 2 页：20 条，累计 40
# ...
# 详情: 新闻标题一... (1500 字)
# ✅ 采集完成

# 验证数据
# ① Excel/记事本打开 news_list.csv（中文不乱码 = utf-8-sig 生效）
# ② 抽查详情链接可访问（urljoin 补全正确）
# ③ 数据量核对（列表条数 = 详情条数）
```

| 验证项 | 检查什么 |
|--------|---------|
| 状态码 | 全部 200（无 403/429 日志） |
| 数据量 | 列表 = 详情（无遗漏） |
| 中文 | CSV 打开不乱码 |
| 链接 | 抽查 url 是完整地址 |
| 去重 | 无重复行 |
| 节奏 | 观察日志间隔（0.5-1.5s） |

> ⚠️ 跑不通时按层排查：**请求层**（curl 先试 URL → 通了再怪解析）→ **解析层**（F12 看实际 class 名 vs 代码选择器）→ **存储层**（打印 rows 看数据对不对）。排查方法论见 [09-常见问题](09-阶段自测与常见问题.md)。

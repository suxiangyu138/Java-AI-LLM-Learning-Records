## 1. 综合实战：动态页三路爬虫

```python
"""
dynamic_spider.py —— 本阶段综合实战
功能：动态页三路方案（直取/捕获/渲染）+ 稳定性三件套
"""
import requests
from bs4 import BeautifulSoup
from playwright.sync_api import sync_playwright

# ═══ ① 路线一：接口直取（02 章）═══
def fetch_api_direct(url, api_keyword, session=None):
    """直接请求接口（需先在 F12 找到接口）"""
    session = session or requests.Session()
    session.headers.update({
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                      "AppleWebKit/537.36 Chrome/126.0.0.0 Safari/537.36",
        "Referer": url,
    })
    # 示例：假设接口是 /api/list?page=N（按实际 F12 结果改）
    api_url = url.replace("/list", "/api/list")
    resp = session.get(api_url, params={"page": 1, "size": 50}, timeout=10)
    if resp.status_code != 200:
        return None
    data = resp.json()
    return data.get("data", {}).get("list", [])

# ═══ ② 路线二：响应捕获（05 §5）═══
def fetch_by_capture(url, api_keyword):
    """浏览器渲染 + 监听接口（绕开签名）"""
    captured = []
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=True, args=["--no-sandbox"])
        page = browser.new_page()
        page.on("response", lambda r: (
            captured.append(r.json())
            if api_keyword in r.url and "json" in (r.headers.get("content-type") or "")
            else None
        ))
        page.goto(url, timeout=30000)
        page.wait_for_timeout(3000)                 # 等接口返回
        browser.close()
    return captured

# ═══ ③ 路线三：渲染后解析（07 §2）═══
def fetch_by_render(url, item_selector):
    """渲染 + bs4 解析（兜底）"""
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=True, args=["--no-sandbox"])
        page = browser.new_page()
        try:
            page.goto(url, timeout=30000)
            page.wait_for_selector(item_selector, timeout=15000)
            html = page.content()
        except Exception as e:
            page.screenshot(path="debug.png")       # 稳定性①：截图取证
            print(f"渲染失败: {e}（已截图 debug.png）")
            html = ""
        browser.close()

    soup = BeautifulSoup(html, "lxml")
    items = []
    for li in soup.select(item_selector):
        a = li.select_one("a")
        if a:
            items.append({"title": a.get_text(strip=True),
                          "url": a.get("href", "")})
    return items

# ═══ ④ 主流程：三路自动选（07 §4 混合架构）═══
def crawl_dynamic(url, api_keyword=None, item_selector=None):
    """动态页三路爬取（直取 → 捕获 → 渲染）"""
    # 路线一：接口直取
    if api_keyword:
        data = fetch_api_direct(url, api_keyword)
        if data:
            print(f"✅ 接口直取: {len(data)} 条")
            return data
        print("接口直取失败，尝试浏览器...")

    # 路线二：响应捕获
    if api_keyword:
        captured = fetch_by_capture(url, api_keyword)
        if captured:
            print(f"✅ 响应捕获: {len(captured)} 个接口")
            return captured

    # 路线三：渲染后解析
    if item_selector:
        items = fetch_by_render(url, item_selector)
        if items:
            print(f"✅ 渲染解析: {len(items)} 条")
            return items

    print("❌ 三路均失败——检查 URL/选择器/是否被拦截")
    return []


if __name__ == "__main__":
    # 按目标站点配置（练习站点自行替换）
    results = crawl_dynamic(
        "https://example.com/list",
        api_keyword="/api/",                 # 接口关键词（F12 确认）
        item_selector="ul.news-list li",     # 数据元素
    )
    print("结果预览:", results[:3])
```

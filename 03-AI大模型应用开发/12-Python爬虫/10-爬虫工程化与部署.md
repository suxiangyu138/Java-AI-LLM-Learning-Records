## 1. 定时调度

### 1.1 APScheduler 定时任务

```python
# pip install apscheduler
from apscheduler.schedulers.blocking import BlockingScheduler
from apscheduler.triggers.cron import CronTrigger

scheduler = BlockingScheduler()

# 每天凌晨 2 点执行
@scheduler.scheduled_job(CronTrigger(hour=2, minute=0))
def daily_crawl():
    print("开始每日爬取...")
    run_crawler()

# 每小时执行
@scheduler.scheduled_job(CronTrigger(minute=0))
def hourly_crawl():
    run_quick_crawl()

# 每 30 分钟执行
@scheduler.scheduled_job("interval", minutes=30)
def frequent_crawl():
    run_incremental_crawl()

scheduler.start()
```

## 2. 异常处理与自愈

### 2.1 全局异常捕获

```python
import traceback
import sys

def resilient_crawl(urls: list[str], max_retries: int = 3):
    """带自愈能力的爬虫"""
    failed = []
    results = []

    for url in urls:
        for attempt in range(max_retries):
            try:
                data = fetch_and_parse(url)
                results.append(data)
                break
            except requests.Timeout:
                if attempt == max_retries - 1:
                    failed.append((url, "Timeout"))
                time.sleep(2 ** attempt)
            except requests.HTTPError as e:
                if e.response.status_code == 429:
                    retry_after = int(e.response.headers.get("Retry-After", 60))
                    print(f"429 限流，等待 {retry_after}s")
                    time.sleep(retry_after)
                else:
                    failed.append((url, f"HTTP {e.response.status_code}"))
                    break
            except Exception as e:
                failed.append((url, str(e)))
                break

    # 失败记录持久化
    if failed:
        with open("failed_urls.json", "w") as f:
            json.dump(failed, f, ensure_ascii=False, indent=2)

    return results, failed
```

### 2.2 断点续爬

```python
class CheckpointCrawler:
    """支持断点续爬的爬虫"""

    def __init__(self, checkpoint_file="checkpoint.json"):
        self.checkpoint_file = checkpoint_file
        self.completed = self._load_checkpoint()

    def _load_checkpoint(self) -> set:
        try:
            with open(self.checkpoint_file) as f:
                return set(json.load(f))
        except FileNotFoundError:
            return set()

    def _save_checkpoint(self):
        with open(self.checkpoint_file, "w") as f:
            json.dump(list(self.completed), f)

    def run(self, urls: list[str]):
        for i, url in enumerate(urls):
            if url in self.completed:
                continue

            try:
                data = crawl(url)
                self.completed.add(url)

                # 每 100 条保存一次
                if i % 100 == 0:
                    self._save_checkpoint()

            except Exception as e:
                print(f"[{i}] {url} 失败: {e}, 下次从断点继续")
                self._save_checkpoint()
                break  # 停止，保留断点
```

## 4. 监控与告警

### 4.1 关键监控指标

```python
class CrawlerMonitor:
    """爬虫监控器"""

    def __init__(self):
        self.start_time = None
        self.stats = {
            "total_urls": 0,
            "success": 0,
            "failed": 0,
            "skipped": 0,
            "bytes_downloaded": 0,
            "start_time": None,
            "end_time": None,
        }

    def start(self):
        self.start_time = time.time()
        self.stats["start_time"] = datetime.now().isoformat()

    def finish(self):
        self.stats["end_time"] = datetime.now().isoformat()
        self._generate_report()

    def _generate_report(self):
        elapsed = time.time() - self.start_time
        rate = self.stats["success"] / elapsed if elapsed > 0 else 0

        report = f"""
====== 爬虫运行报告 ======
时间: {self.stats['start_time']} → {self.stats['end_time']}
耗时: {elapsed:.1f}s
总数: {self.stats['total_urls']}
成功: {self.stats['success']}
失败: {self.stats['failed']}
跳过: {self.stats['skipped']}
速率: {rate:.1f} req/s
下载: {self.stats['bytes_downloaded'] / 1024 / 1024:.1f} MB
成功率: {self.stats['success'] / max(self.stats['total_urls'], 1) * 100:.1f}%
========================
"""
        print(report)

        # 写入日志
        with open("crawl_report.log", "a") as f:
            f.write(report)

        # 失败率告警
        fail_rate = self.stats["failed"] / max(self.stats["total_urls"], 1)
        if fail_rate > 0.3:
            send_alert(f"⚠️ 爬虫失败率过高: {fail_rate:.1%}")
```

## 1. Excel 自动化

### 1.1 openpyxl 样式化输出

```python
import pandas as pd
from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
from openpyxl.utils.dataframe import dataframe_to_rows

# 准备数据
wb = Workbook()
ws = wb.active
ws.title = "销售月报"

# 写入 DataFrame
for r in dataframe_to_rows(df_summary, index=False, header=True):
    ws.append(r)

# ---- 样式设置 ----
# 标题行
header_font = Font(name="微软雅黑", bold=True, size=12, color="FFFFFF")
header_fill = PatternFill(start_color="4472C4", end_color="4472C4", fill_type="solid")

for cell in ws[1]:
    cell.font = header_font
    cell.fill = header_fill
    cell.alignment = Alignment(horizontal="center", vertical="center")

# 数据行样式
for row in ws.iter_rows(min_row=2, max_row=ws.max_row):
    for i, cell in enumerate(row):
        if i == 0:
            cell.alignment = Alignment(horizontal="left")   # 第一列左对齐
        else:
            cell.alignment = Alignment(horizontal="center")  # 其余居中

# 条件格式（高亮 > 阈值）
from openpyxl.formatting.rule import CellIsRule
ws.conditional_formatting.add(
    f"C2:C{ws.max_row}",
    CellIsRule(operator="greaterThan", formula=["1000000"],
               fill=PatternFill(start_color="C6EFCE", fill_type="solid"))
)

# 自动列宽
for col in ws.columns:
    max_len = max(len(str(cell.value or "")) for cell in col)
    ws.column_dimensions[col[0].column_letter].width = min(max_len + 2, 30)

# 冻结首行
ws.freeze_panes = "A2"

wb.save("销售月报_2026年7月.xlsx")
```

## 3. 定时任务与邮件推送

```python
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from email.mime.base import MIMEBase
from email import encoders

def generate_and_send_report():
    """生成报告并发送邮件"""
    # 1. 数据分析
    df = pd.read_csv("sales_data.csv")
    summary = analyze_sales(df)

    # 2. 生成图表
    fig = create_dashboard(summary)
    fig.savefig("dashboard.png", dpi=150, bbox_inches="tight")

    # 3. 生成 Excel 附件
    with pd.ExcelWriter("weekly_report.xlsx") as writer:
        summary.to_excel(writer, index=False)

    # 4. 构建邮件
    msg = MIMEMultipart()
    msg["Subject"] = f"销售周报 ({datetime.now().strftime('%Y-%m-%d')})"
    msg["From"] = "analytics@company.com"
    msg["To"] = "manager@company.com"

    # HTML 正文
    html = f"""
    <h2>📊 销售周报</h2>
    <p>总销售额: <b>¥{summary['total']:,.0f}</b></p>
    <p>周环比: <b>{summary['wow']:+.1%}</b></p>
    <p>同比增长: <b>{summary['yoy']:+.1%}</b></p>
    <p><img src="cid:dashboard"></p>
    """
    msg.attach(MIMEText(html, "html"))

    # 图片附件（内嵌）
    with open("dashboard.png", "rb") as f:
        img = MIMEBase("image", "png", filename="dashboard.png")
        img.add_header("Content-ID", "<dashboard>")
        img.set_payload(f.read())
        encoders.encode_base64(img)
        msg.attach(img)

    # Excel 附件
    with open("weekly_report.xlsx", "rb") as f:
        xlsx = MIMEBase("application", "octet-stream")
        xlsx.set_payload(f.read())
        encoders.encode_base64(xlsx)
        xlsx.add_header("Content-Disposition",
                        "attachment", filename="weekly_report.xlsx")
        msg.attach(xlsx)

    # 5. 发送
    with smtplib.SMTP("smtp.company.com", 587) as server:
        server.starttls()
        server.login("username", "password")
        server.send_message(msg)

    print("✅ 报告已发送")
```

## 核心要点回顾

- Excel 自动化：`openpyxl` 控制样式/颜色/条件格式/冻结
- PDF 报告：Matplotlib → PdfPages（图表）或 ReportLab（文字+排版）
- 多 Sheet 输出：`pd.ExcelWriter` + `sheet_name` 参数
- 邮件推送：SMTP + MIMEMultipart（HTML正文 + 图片 + Excel附件）
- 定时：APScheduler（Python）或 crontab + Python 脚本

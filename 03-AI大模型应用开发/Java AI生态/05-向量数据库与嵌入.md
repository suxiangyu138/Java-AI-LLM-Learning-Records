## 2. PGVector：PostgreSQL 扩展

### 2.1 为什么首选 PGVector

```text
PGVector 的核心优势：
┌─────────────────────────────────────────────┐
│ 1. 零额外运维 —— 与现有 PostgreSQL 共用      │
│ 2. ACID 事务 —— 关系数据和向量在同一事务中     │
│ 3. 生态完整 —— PG 连接池/监控/备份全部复用     │
│ 4. sparsevec（2026）—— 原生混合检索           │
│ 5. HNSW 索引 —— 生产级检索性能               │
│ 6. 团队熟悉 —— 不需要学新工具                  │
└─────────────────────────────────────────────┘
```

### 2.2 安装与配置

```sql
-- 1. 安装 pgvector 扩展
CREATE EXTENSION IF NOT EXISTS vector;

-- 2. 创建文档表
CREATE TABLE documents (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    content     TEXT NOT NULL,
    source      VARCHAR(500),
    metadata    JSONB DEFAULT '{}',
    embedding   VECTOR(1024)  -- ⚠️ 维度必须匹配 Embedding 模型
);

-- 3. 创建 HNSW 索引（生产必备）
CREATE INDEX idx_documents_embedding
    ON documents
    USING hnsw (embedding vector_cosine_ops)
    WITH (
        m = 24,               -- 每层最大连接数（越大越精确，越占内存）
        ef_construction = 128  -- 构建搜索宽度（越大越精确，构建越慢）
    );
```

### 2.3 HNSW 参数调优指南

| 参数 | 范围 | 默认 | 小数据集 (<100K) | 大数据集 (>1M) | 说明 |
|------|:---:|:---:|:---:|:---:|------|
| **m** | 2-100 | 16 | 16 | 32-64 | 越大=越精确+越占内存 |
| **ef_construction** | 4-1000 | 64 | 64 | 200-500 | 构建精度，不影响查询速度 |
| **ef_search** | 1-1000 | 40 | 40 | 100-200 | 查询精度，运行时设置 |

### 2.4 Java 配置

```yaml
# Spring AI + PGVector 配置
spring:
  ai:
    vectorstore:
      pgvector:
        host: localhost
        port: 5432
        database: knowledge_base
        username: ${DB_USER}
        password: ${DB_PASSWORD}
        table-name: documents
        dimensions: 1024          # 必须与 Embedding 模型一致
        index-type: hnsw          # hnsw 或 ivfflat
        initialize-schema: true   # 自动建表
```

### 2.5 sparsevec：2026 年新能力

```sql
-- sparsevec：在 PostgreSQL 内实现原生混合检索，无需 Elasticsearch
CREATE TABLE docs_hybrid (
    id         UUID PRIMARY KEY,
    content    TEXT,
    dense_vec  VECTOR(1024),     -- 语义向量
    sparse_vec SPARSEVEC(250000) -- 关键词稀疏向量（SPLADE）
);

-- 分别建索引
CREATE INDEX ON docs_hybrid USING hnsw (dense_vec vector_cosine_ops);
CREATE INDEX ON docs_hybrid USING hnsw (sparse_vec vector_cosine_ops);
```

## 3. Elasticsearch 向量检索

### 3.1 配置

```json

PUT /knowledge_base
{
  "mappings": {
    "properties": {
      "content": {
        "type": "text",
        "analyzer": "standard"
      },
      "content_vector": {
        "type": "dense_vector",
        "dims": 1024,
        "index": true,
        "similarity": "cosine"
      },
      "metadata": {
        "type": "object"
      }
    }
  }
}
```

### 3.2 Java 端使用

```java
@Service
public class ElasticsearchVectorSearch {

    @Autowired
    private ElasticsearchClient esClient;

    /**
     * ES kNN 向量检索
     */
    public List<SearchHit<Document>> knnSearch(float[] queryVector, int k) {
        var response = esClient.search(s -> s
            .index("knowledge_base")
            .knn(knn -> knn
                .field("content_vector")
                .queryVector(asFloatList(queryVector))
                .k(k)
                .numCandidates(100)  // 候选集大小，建议 k 的 5-10 倍
            ),
            Document.class
        );

        return response.hits().hits();
    }

    /**
     * 混合检索：kNN 向量 + BM25 关键词
     */
    public List<SearchHit<Document>> hybridSearch(
            String queryText, float[] queryVector, int k) {

        var response = esClient.search(s -> s
            .index("knowledge_base")
            .query(q -> q.bool(b -> b
                .should(sh -> sh.knn(knn -> knn
                    .field("content_vector")
                    .queryVector(asFloatList(queryVector))
                    .k(k)
                    .boost(1.0f)       // 向量权重
                ))
                .should(sh -> sh.match(m -> m
                    .field("content")
                    .query(queryText)
                    .boost(0.5f)       // 关键词权重
                ))
            ))
            .size(k),
            Document.class
        );

        return response.hits().hits();
    }
}
```

## 4. 专业向量数据库

### 4.1 Milvus（十亿级规模）

```yaml
# 适用：海量向量检索、推荐系统、以图搜图
# 特点：分布式、GPU 加速、十亿级
milvus:
  host: localhost
  port: 19530
  collection: documents
  index-type: IVF_FLAT    # 或 HNSW, DISKANN
  metric-type: COSINE
```

### 4.2 Qdrant（高 QPS）

```yaml
# 适用：高并发 API、实时检索
# 特点：Rust 实现、量化压缩、内置 Payload 过滤
qdrant:
  host: localhost
  port: 6334
  collection: documents
  quantization: scalar    # 压缩方式
  on-disk: true           # 磁盘存储（大规模节省内存）
```

### 4.3 Redis Stack（低延迟缓存）

```yaml
# 适用：缓存 + 向量一体、毫秒级延迟
# 特点：内存级速度、与业务缓存复用
spring:
  ai:
    vectorstore:
      redis:
        host: localhost
        port: 6379
        index: documents_idx
        prefix: "doc:"
        algorithm: hnsw       # hnsw 或 flat
        distance-metric: cosine
```

### 4.4 对比速查

| 数据库 | 规模上限 | 延迟 | QPS | 运维复杂度 | 成本 |
|------|:---:|:---:|:---:|:---:|:---:|
| **PGVector** | 百万级 | 10-50ms | 中 | 低（复用 PG） | 低 |
| **Elasticsearch** | 百万级 | 10-100ms | 中 | 中 | 中 |
| **Milvus** | 十亿级 | 1-10ms | 高 | 高 | 高 |
| **Qdrant** | 亿级 | 1-5ms | 极高 | 中 | 中 |
| **Redis Stack** | 内存限制 | <1ms | 极高 | 低 | 中 |
| **Chroma** | 十万级 | 10-50ms | 低 | 极低 | 免费 |

### 5.2 本地部署 vs API

```java
// 方案 A：本地 Ollama 部署（免费、私密、离线可用）
@Bean
public EmbeddingModel localEmbedding() {
    return OllamaEmbeddingModel.builder()
        .baseUrl("http://localhost:11434")
        .modelName("bge-m3")
        .build();

// 方案 B：OpenAI API（方便、高精度、成本透明）

    return OpenAiEmbeddingModel.builder()
        .apiKey(System.getenv("OPENAI_API_KEY"))
        .modelName("text-embedding-3-small")
        .build();

// 方案 C：Spring AI 自动配置（切换只需改配置）
// application.yml
spring:
  ai:
    ollama:
      embedding:
        model: bge-m3
        base-url: http://localhost:11434
```

## 6. 性能优化实战

### 6.1 HNSW 索引调优

```sql
-- 入库完成后重建索引（重要！）
REINDEX INDEX CONCURRENTLY idx_documents_embedding;

-- 查询前设置搜索宽度（提高精度）
SET hnsw.ef_search = 64;

-- 查看索引大小和性能
SELECT
    indexname,
    pg_size_pretty(pg_relation_size(indexname::regclass)) AS size,
    idx_scan AS scans,
    idx_tup_read AS tuples_read,
    idx_tup_fetch AS tuples_fetched
FROM pg_stat_all_indexes
WHERE indexname LIKE '%embedding%';
```

### 6.2 批量入库最佳实践

```java
@Service
public class BatchIngestionService {

    private static final int BATCH_SIZE = 100;

    public IngestionStats batchIngest(List<Document> documents) {
        IngestionStats stats = new IngestionStats();
        List<Document> batch = new ArrayList<>(BATCH_SIZE);

        for (Document doc : documents) {
            batch.add(doc);

            if (batch.size() >= BATCH_SIZE) {
                try {
                    vectorStore.add(batch);
                    stats.incrementSuccess(batch.size());
                } catch (Exception e) {
                    log.error("批量入库失败，记录失败 ID 以便重试", e);
                    stats.incrementFailed(batch.size());
                }
                batch.clear();
            }
        }

        // 处理剩余批次
        if (!batch.isEmpty()) {
            try {
                vectorStore.add(batch);
                stats.incrementSuccess(batch.size());
            } catch (Exception e) {
                stats.incrementFailed(batch.size());
            }
        }

        return stats;
    }
}
```

### 6.3 连接池与并发


## 7. 选型决策框架

```text
                     你的向量数据库需求？
                           │
          ┌────────────────┼────────────────┐
          ▼                ▼                ▼
    已有 PostgreSQL   已有 Elasticsearch   从零开始
          │                │                │
          ▼                ▼                ▼
    用 PGVector！     用 ES 向量功能！     数据量多大？
    ┌─────────────┐  ┌──────────────┐       │
    │ ✅ ACID 事务  │  │ ✅ 混合检索   │  ┌────┴────┐
    │ ✅ 零新运维   │  │ ✅ 已有集群   │  ▼         ▼
    │ ✅ sparsevec │  │ ⚠️ 内存消耗大 │ <100万   >1000万
    └─────────────┘  └──────────────┘    │         │
                                         ▼         ▼
                                     PGVector   Milvus
                                    / Redis     / Qdrant

需要低延迟（<1ms）？
  → Redis Stack

需要全托管（零运维）？
  → Pinecone

原型开发/小项目？
  → Chroma

纯 Java 嵌入式（零依赖）？
  → Lucene
```

---

> 🎯 **核心要点**：90% 的场景用 PGVector 就够了——ACID 事务 + 关系数据和向量在同一事务中 + sparsevec 原生混合检索。只有在十亿级以上规模、毫秒级延迟要求、或已有 ES 集群时才考虑专业向量数据库。

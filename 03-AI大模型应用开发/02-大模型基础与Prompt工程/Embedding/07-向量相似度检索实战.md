## 1. 检索问题定义

```text
向量检索问题：

  给定：
    - 查询向量 q (dim 维)
    - 候选向量库 D = {d₁, d₂, ..., d_N}，每个 d_i 是 dim 维向量
  
  目标：
    找出与 q 最相似的 Top-K 个候选向量

  相似度度量：余弦相似度 / 内积 / 欧氏距离
  K 的典型值：3-10

复杂度：
  暴力检索：O(N × dim) — N 个向量逐个比较
```

## 2. 暴力检索：最简实现

### 2.1 纯 Python 实现

```python
import numpy as np

class SimpleVectorSearch:
    """最简向量检索 — 理解原理用"""
    
    def __init__(self):
        self.ids = []        # 文档 ID
        self.texts = []      # 原始文本
        self.vectors = []    # 向量
    
    def add(self, doc_id, text, vector):
        self.ids.append(doc_id)
        self.texts.append(text)
        self.vectors.append(vector)
    
    def search(self, query_vec, top_k=5):
        """暴力检索 Top-K"""
        if not self.vectors:
            return []
        
        # 计算所有候选的余弦相似度
        scores = []
        for i, vec in enumerate(self.vectors):
            sim = np.dot(query_vec, vec) / (
                np.linalg.norm(query_vec) * np.linalg.norm(vec)
            )
            scores.append((sim, i))
        
        # 排序取 Top-K
        scores.sort(reverse=True)
        results = []
        for sim, idx in scores[:top_k]:
            results.append({
                "id": self.ids[idx],
                "text": self.texts[idx],
                "score": sim
            })
        
        return results
```

## 3. NumPy 向量化加速

### 3.1 批量余弦相似度

```python
def batch_cosine_similarity(query_vec: np.ndarray, 
                            doc_matrix: np.ndarray) -> np.ndarray:
    """批量计算余弦相似度 — 一次矩阵运算替代循环"""
    # 归一化
    query_norm = query_vec / np.linalg.norm(query_vec)
    doc_norms = doc_matrix / np.linalg.norm(doc_matrix, axis=1, keepdims=True)
    
    # 归一化后的点积 = 余弦相似度
    scores = np.dot(doc_norms, query_norm)
    return scores

def fast_search(query_vec, doc_matrix, doc_texts, top_k=5):
    """向量化快速检索"""
    scores = batch_cosine_similarity(query_vec, doc_matrix)
    
    # argsort 降序 → 取 Top-K 索引
    top_indices = np.argsort(scores)[::-1][:top_k]
    
    return [
        {"text": doc_texts[i], "score": float(scores[i])}
        for i in top_indices
    ]

# 速度对比：
# 10000 个 1024 维向量
#   纯 Python 循环： ~500ms
#   NumPy 矩阵运算：~5ms → 提速 100×
```

### 3.2 批量查询

```python
def batch_search(query_matrix, doc_matrix, top_k=5):
    """一次检索多个 query — 极致效率"""
    # 归一化
    q_norm = query_matrix / np.linalg.norm(query_matrix, axis=1, keepdims=True)
    d_norm = doc_matrix / np.linalg.norm(doc_matrix, axis=1, keepdims=True)
    
    # 相似度矩阵 (num_queries, num_docs)
    scores = np.dot(q_norm, d_norm.T)
    
    # 每个 query 的 Top-K
    top_indices = np.argsort(scores, axis=1)[:, ::-1][:, :top_k]
    
    return top_indices, np.take_along_axis(scores, top_indices, axis=1)
```

## 4. FAISS 索引入门

### 4.1 什么是 FAISS

```text
FAISS (Facebook AI Similarity Search) = 工业级向量检索库

特性：
  ✅ 支持 10 亿级向量检索
  ✅ GPU 加速
  ✅ 多种索引类型（精确/近似）
  ✅ 支持向量压缩（PQ 量化）
  ✅ C++ 底层，Python 接口
```

### 4.2 基础使用

```python
import faiss
import numpy as np

# ① 创建索引
dim = 1024  # 向量维度
index = faiss.IndexFlatIP(dim)  # 内积索引（需向量已归一化）
# 或 IndexFlatL2(dim) — L2 距离索引

# ② 添加向量
vectors = np.random.randn(10000, dim).astype('float32')
# 内积索引需要归一化
faiss.normalize_L2(vectors)
index.add(vectors)

print(f"索引中向量数: {index.ntotal}")

# ③ 检索
query = np.random.randn(1, dim).astype('float32')
faiss.normalize_L2(query)
k = 5
distances, indices = index.search(query, k)

print(f"Top-{k} 索引: {indices[0]}")
print(f"相似度: {distances[0]}")  # 内积值（归一化后=余弦相似度）
```

### 4.3 索引类型选择

| 索引 | 原理 | 速度 | 精度 | 适用规模 |
|------|------|:---:|:---:|:---:|
| **IndexFlatIP** | 暴力内积搜索 | 慢 | 100% | <10万 |
| **IndexIVFFlat** | 倒排索引+暴力 | 中 | ~95% | 10万~1000万 |
| **IndexIVFPQ** | 倒排+乘积量化 | 快 | ~90% | 100万~10亿 |
| **IndexHNSWFlat** | 图索引 | 最快 | ~98% | <100万 |

## 5. 近似最近邻 ANN

### 5.1 为什么需要近似

```text
暴力检索：O(N × dim) — 100 万 × 1024 维 → 慢但 100% 准确
近似检索：牺牲少量精度（<5%）→ 速度提升 100-1000×

ANN 的 trade-off：
  精度 vs 速度 vs 内存
  
  三种主流方法：
  ① 图索引 (HNSW)：构建"高速公路图" → 快速导航到最近邻
  ② 倒排索引 (IVF)：聚类 → 只搜索最近的几个簇
  ③ 哈希 (LSH)：相似的向量哈希到同一个桶
```

### 5.2 HNSW 索引

```python
# HNSW — 基于图的最快 ANN 索引
dim = 1024
index = faiss.IndexHNSWFlat(dim, 32)  # 32 = 每个节点的连接数
# M 越大 → 精度越高 → 内存越大 → 构建越慢

vectors = np.random.randn(100000, dim).astype('float32')
index.add(vectors)

query = np.random.randn(1, dim).astype('float32')
distances, indices = index.search(query, k=5)
```

## 6. 检索质量评估

```python
def evaluate_retrieval(index, queries, ground_truth, k=5):
    """
    评估检索质量
    
    queries: 查询向量列表
    ground_truth: 每个 query 对应的正确答案 ID 列表
    """
    recall_count = 0
    mrr_sum = 0  # Mean Reciprocal Rank
    
    for i, q in enumerate(queries):
        _, indices = index.search(q.reshape(1, -1), k)
        retrieved = set(indices[0])
        
        # Recall@K
        if ground_truth[i] in retrieved:
            recall_count += 1
        
        # MRR
        for rank, idx in enumerate(indices[0], 1):
            if idx == ground_truth[i]:
                mrr_sum += 1.0 / rank
                break
    
    recall = recall_count / len(queries)
    mrr = mrr_sum / len(queries)
    
    return {"Recall@K": recall, "MRR": mrr}
```

## 7. 生产级检索封装

```python
import faiss
import numpy as np
from typing import Optional

class VectorSearchEngine:
    """生产级向量检索引擎"""
    
    def __init__(self, dim: int, index_type="flat"):
        self.dim = dim
        self.texts = []
        self.metadata = []
        
        if index_type == "flat":
            self.index = faiss.IndexFlatIP(dim)
        elif index_type == "hnsw":
            self.index = faiss.IndexHNSWFlat(dim, 32)
        elif index_type == "ivf":
            quantizer = faiss.IndexFlatIP(dim)
            nlist = 100  # 聚类中心数
            self.index = faiss.IndexIVFFlat(quantizer, dim, nlist)
            self.index.train  # IVF 需要训练
    
    def add(self, texts, vectors, metadata=None):
        """批量添加文档"""
        vecs = np.array(vectors).astype('float32')
        faiss.normalize_L2(vecs)  # 内积=余弦相似度
        
        start_idx = len(self.texts)
        self.index.add(vecs)
        self.texts.extend(texts)
        if metadata:
            self.metadata.extend(metadata)
    
    def search(self, query_vec, top_k=5):
        """检索 Top-K"""
        q = np.array(query_vec).astype('float32').reshape(1, -1)
        faiss.normalize_L2(q)
        
        scores, indices = self.index.search(q, top_k)
        
        results = []
        for score, idx in zip(scores[0], indices[0]):
            if idx == -1:  # FAISS 用 -1 表示无效
                continue
            results.append({
                "text": self.texts[idx],
                "score": float(score),
                "metadata": self.metadata[idx] if self.metadata else None
            })
        return results
    
    def save(self, path):
        faiss.write_index(self.index, f"{path}.index")
    
    def load(self, path):
        self.index = faiss.read_index(f"{path}.index")
```

## 核心要点回顾

- NumPy 向量化检索比纯 Python 快 100×（矩阵运算替代循环）
- FAISS 是工业级向量检索事实标准（Meta 出品，支持 10 亿级）
- HNSW 图索引是当前最快的 ANN 方案（精度/速度最优平衡）
- 近似检索牺牲 <5% 精度换取 100-1000× 速度提升
- 生产建议：<10 万用 IndexFlatIP，10 万~1000 万用 IndexIVFPQ，>1000 万用 GPU 分布式

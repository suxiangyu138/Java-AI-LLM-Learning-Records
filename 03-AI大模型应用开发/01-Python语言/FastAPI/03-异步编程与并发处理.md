## 1. async/await 核心概念

### 1.1 事件循环模型

```text
FastAPI 基于 Starlette → asyncio 事件循环

  同步（def）：
    请求1 ────────────────────────► 返回
    请求2                       等待 ────────────────► 返回
    请求3                       等待  等待 ──────────► 返回
    时间 → 三请求总耗时 = 累加

  异步（async def）：
    请求1 ── IO等待 ── IO等待 ───► 返回
    请求2 ──── IO等待 ────────► 返回
    请求3 ────── IO等待 ────► 返回
    时间 → 三请求总耗时 ≈ 最长的一个

  async def 让 IO 密集型任务（网络/数据库/文件）不阻塞其他请求
```

### 1.2 async/await 基础

```python
import asyncio
import httpx
from fastapi import FastAPI

app = FastAPI()

# ===== 基本语法 =====
async def fetch_data(url: str) -> dict:
    """异步 HTTP 请求 — 不阻塞事件循环"""
    async with httpx.AsyncClient() as client:
        response = await client.get(url)  # ← await 释放控制权
        return response.json()


@app.get("/async-demo")
async def async_endpoint():
    """异步端点：await 释放控制权给其他请求"""
    result = await fetch_data("https://api.example.com/data")
    return result
```

## 2. async vs def 路由选择

### 2.1 抉择规则

```text
┌─────────────────────────────────────────────────────────┐
│              async def vs def 决策树                      │
├─────────────────────────────────────────────────────────┤
│                                                          │
│  路由函数是否包含 await 调用？                             │
│  ├── 是 → 必须 async def                                  │
│  │   ✅ async def fetch():                                │
│  │        data = await db.query()                         │
│  │        result = await client.get(url)                  │
│  │                                                        │
│  └── 否 → 可以 def，也可以 async def                      │
│       ├── def（推荐）：Starlette 在线程池运行，不阻塞     │
│       │   ✅ def compute():                               │
│       │        return heavy_pandas_operation()            │
│       │                                                   │
│       └── async def：在事件循环中运行，但无收益           │
│           ⚠️ async def compute():                         │
│                return heavy_pandas_operation()  # 阻塞!   │
│                                                          │
│  规则：                                                   │
│  • IO 密集型（DB/HTTP/文件）→ async def + await          │
│  • CPU 密集型（计算/ML推理） → def（线程池）              │
│  • 混合                 → async def + run_in_executor    │
└─────────────────────────────────────────────────────────┘
```

### 2.2 错误示例

```python
# ❌ 错误：async def 中直接调用同步代码 — 阻塞事件循环！
import time

@app.get("/wrong")
async def wrong():
    time.sleep(5)  # ⚠️ 阻塞整个事件循环 5 秒！
    return {"status": "done"}

# ✅ 正确：用 run_in_executor 在线程池中执行
@app.get("/correct")
async def correct():
    loop = asyncio.get_event_loop()
    await loop.run_in_executor(None, time.sleep, 5)  # 不阻塞
    return {"status": "done"}


# ❌ 错误：def 中调用 async 函数 — 需要事件循环
@app.get("/wrong2")
def wrong2():
    result = await fetch_data(...)  # SyntaxError!
    return result

# ✅ 正确：def 中导入 sync 版本或用 asyncio.run
@app.get("/correct2")
def correct2():
    result = httpx.get("https://api.example.com/data").json()
    return result
```

## 3. 并发调用 LLM API

### 3.1 asyncio.gather — 并行调用

```python
"""
AI 应用核心场景：同时调用多个 LLM / 向量数据库 / 搜索引擎
"""
import asyncio
from fastapi import FastAPI
import httpx

app = FastAPI()


@app.get("/api/compare-models")
async def compare_models(prompt: str):
    """
    并发调用 3 个 LLM，比较结果
    总耗时 ≈ max(三个调用)，而不是 sum
    """

    async def call_gpt4o(prompt: str) -> dict:
        async with httpx.AsyncClient() as client:
            resp = await client.post(
                "https://api.openai.com/v1/chat/completions",
                headers={"Authorization": f"Bearer {OPENAI_KEY}"},
                json={"model": "gpt-4o", "messages": [{"role": "user", "content": prompt}]},
                timeout=60.0,
            )
            return {"model": "gpt-4o", "result": resp.json()}

    async def call_claude(prompt: str) -> dict:
        async with httpx.AsyncClient() as client:
            resp = await client.post(
                "https://api.anthropic.com/v1/messages",
                headers={"x-api-key": ANTHROPIC_KEY, "anthropic-version": "2023-06-01"},
                json={"model": "claude-sonnet-5", "max_tokens": 1000, "messages": [{"role": "user", "content": prompt}]},
                timeout=60.0,
            )
            return {"model": "claude-opus-5", "result": resp.json()}

    async def call_deepseek(prompt: str) -> dict:
        async with httpx.AsyncClient() as client:
            resp = await client.post(
                "https://api.deepseek.com/v1/chat/completions",
                headers={"Authorization": f"Bearer {DEEPSEEK_KEY}"},
                json={"model": "deepseek-chat", "messages": [{"role": "user", "content": prompt}]},
                timeout=60.0,
            )
            return {"model": "deepseek-v4-pro", "result": resp.json()}

    # 并发执行：三个调用同时进行
    results = await asyncio.gather(
        call_gpt4o(prompt),
        call_claude(prompt),
        call_deepseek(prompt),
        return_exceptions=True,  # 单个失败不影响其他
    )

    # 分别处理成功和失败
    output = []
    for i, result in enumerate(results):
        if isinstance(result, Exception):
            output.append({"error": str(result)})
        else:
            output.append(result)

    return {"comparisons": output}
```

### 3.2 asyncio.as_completed — 先到先处理

```python
@app.get("/api/chat-race")
async def chat_race(prompt: str):
    """
    竞速模式：同时调用多个模型，返回最先完成的
    适合追求响应速度的场景
    """

    tasks = [
        call_gpt4o(prompt),
        call_claude(prompt),
        call_deepseek(prompt),
    ]

    # 谁先完成用谁的结果
    for coro in asyncio.as_completed(tasks):
        try:
            result = await coro
            # 取消其他还没完成的任务
            for t in tasks:
                t.cancel()
            return {"winner": result}
        except Exception:
            continue  # 这个失败了，等下一个

    raise HTTPException(500, "所有模型调用都失败了")
```

### 3.3 Semaphore — 并发控制

```python
# 限制最大并发数，防止打爆 API/数据库
SEMAPHORE = asyncio.Semaphore(10)  # 最多 10 个并发


@app.get("/api/batch-translate")
async def batch_translate(texts: list[str]):
    """批量翻译 — 控制并发数"""

    async def translate_one(text: str) -> str:
        async with SEMAPHORE:  # 获取信号量许可
            async with httpx.AsyncClient() as client:
                resp = await client.post(
                    "https://api.openai.com/v1/chat/completions",
                    json={"model": "gpt-4o", "messages": [
                        {"role": "user", "content": f"翻译为英文：{text}"}
                    ]},
                )
                return resp.json()["choices"][0]["message"]["content"]

    results = await asyncio.gather(*[translate_one(t) for t in texts])
    return {"translations": list(zip(texts, results))}
```

## 4. CPU 密集型任务的线程池处理

### 4.1 run_in_executor

```python
import asyncio
from concurrent.futures import ThreadPoolExecutor, ProcessPoolExecutor
from functools import partial

# 线程池（默认）
EXECUTOR = ThreadPoolExecutor(max_workers=4)  # CPU 核心数的 2 倍

# 进程池（真正的 CPU 并行）
PROCESS_EXECUTOR = ProcessPoolExecutor(max_workers=2)


@app.post("/api/embed-text")
async def embed_text(text: str):
    """
    文本 Embedding — CPU 密集型（模型推理）
    放到线程池/进程池执行，不阻塞事件循环
    """
    loop = asyncio.get_event_loop()

    # 方式 1：run_in_executor
    embedding = await loop.run_in_executor(
        EXECUTOR,
        embedding_model.encode,  # 同步函数
        text,
    )

    return {"embedding": embedding.tolist(), "dim": len(embedding)}


@app.post("/api/embed-batch")
async def embed_batch(texts: list[str]):
    """批量 Embedding — 并发 + 线程池"""
    loop = asyncio.get_event_loop()

    async def embed_one(text: str):
        return await loop.run_in_executor(
            EXECUTOR,
            embedding_model.encode,
            text,
        )

    embeddings = await asyncio.gather(*[embed_one(t) for t in texts])
    return {"embeddings": [e.tolist() for e in embeddings]}
```

### 4.2 自定义线程池 + 上下文管理

```python
from contextlib import asynccontextmanager

# 在应用生命周期中管理线程池
@asynccontextmanager
async def lifespan(app: FastAPI):
    # 启动
    app.state.executor = ThreadPoolExecutor(max_workers=8)
    yield
    # 关闭
    app.state.executor.shutdown(wait=True)


app = FastAPI(lifespan=lifespan)


@app.post("/api/process")
async def process(data: str):
    executor = app.state.executor
    result = await asyncio.get_event_loop().run_in_executor(
        executor,
        heavy_computation,
        data,
    )
    return result
```

## 5. BackgroundTasks 后台任务

### 5.1 基础用法

```python
from fastapi import BackgroundTasks


def write_log(message: str):
    """后台写日志 — 同步函数"""
    with open("app.log", "a") as f:
        f.write(f"{datetime.now()}: {message}\n")


async def send_notification(user_id: int, content: str):
    """后台发通知 — 异步函数"""
    await httpx.AsyncClient().post(
        "https://notify.example.com/send",
        json={"user_id": user_id, "content": content},
    )


@app.post("/api/chat")
async def chat(
    request: ChatRequest,
    background_tasks: BackgroundTasks,
):
    # 主流程：立即返回 LLM 响应
    response = await call_llm(request)

    # 添加后台任务（响应返回后执行）
    background_tasks.add_task(write_log, f"User chat: {request.messages[-1].content[:50]}")
    background_tasks.add_task(send_notification, user_id=123, content="Chat completed")

    return response
```

### 5.2 AI 应用中的典型后台任务

```python
@app.post("/api/rag/upload")
async def upload_document(
    file: UploadFile,
    background_tasks: BackgroundTasks,
    db = Depends(get_db),
):
    """文档上传 — 接受后异步处理"""
    # 1. 保存文件（同步，但很快）
    content = await file.read()
    doc_id = await db.save_document(file.filename, content)

    # 2. 立即返回
    # 3. 后台异步处理：分块 → Embedding → 存入向量数据库
    background_tasks.add_task(process_document_async, doc_id, content)

    return {
        "doc_id": doc_id,
        "status": "processing",
        "message": "文档已接收，正在后台处理中。请稍后查询状态",
    }


async def process_document_async(doc_id: str, content: bytes):
    """后台文档处理（可能耗时数分钟）"""
    # 分块
    chunks = split_into_chunks(content)
    # 批量 Embedding（带并发控制）
    embeddings = await batch_embed(chunks)
    # 存入向量数据库
    await vector_db.upsert(doc_id, chunks, embeddings)
    # 更新状态
    await db.update_status(doc_id, "ready")
```

> 🎯 **核心要点**：IO 用 async/await（DB/HTTP）、CPU 用 run_in_executor（Embedding/计算）、后处理用 BackgroundTasks（日志/通知/异步入库）。asyncio.gather 实现 LLM 并发调用，Semaphore 控并发

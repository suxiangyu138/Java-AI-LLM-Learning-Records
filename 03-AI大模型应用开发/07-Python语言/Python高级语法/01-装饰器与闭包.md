## 1. 闭包：装饰器的前置知识

```python
# 闭包 = 函数 + 它捕获的外部变量
def make_multiplier(n):
    def multiplier(x):
        return x * n          # n 被"捕获"了，即使 make_multiplier 已返回
    return multiplier

double = make_multiplier(2)
print(double(5))   # 10 —— 闭包记住了 n=2
print(double(10))  # 20
```

## 2. 装饰器基础

```python
# 装饰器 = 接受函数 → 返回增强版函数
def log(func):
    def wrapper(*args, **kwargs):
        print(f"调用 {func.__name__}({args})")
        result = func(*args, **kwargs)
        print(f"返回 {result}")
        return result
    return wrapper

@log  # ← 等价于 greet = log(greet)
def greet(name):
    return f"Hello, {name}"

greet("Alice")
# 调用 greet(('Alice',))
# 返回 Hello, Alice
```

### 保留元信息

```python
import functools

def log(func):
    @functools.wraps(func)  # ← 保留原函数的 __name__ / __doc__
    def wrapper(*args, **kwargs):
        print(f"[{func.__name__}]")
        return func(*args, **kwargs)
    return wrapper
# 不用 @wraps → wrapper.__name__ = "wrapper"（调试时看不出原始函数）
```

## 3. 带参数 + 类装饰器

```python
# 带参数的装饰器：再加一层函数
def retry(max_attempts=3, delay=1):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    time.sleep(delay)
            return None
        return wrapper
    return decorator

@retry(max_attempts=3, delay=2)  # 重试 3 次，间隔 2 秒
def call_api(prompt):
    return openai.chat(prompt)
```

```python
# 类装饰器：用类实现装饰器（可维护状态）
class CountCalls:
    def __init__(self, func):
        self.func = func
        self.count = 0

    def __call__(self, *args, **kwargs):
        self.count += 1
        print(f"第 {self.count} 次调用 {self.func.__name__}")
        return self.func(*args, **kwargs)

@CountCalls
def say_hello():
    print("Hello!")

say_hello()  # 第 1 次调用 say_hello
say_hello()  # 第 2 次调用 say_hello
```

## 4. AI 开发实战

```python
# 1. 计时装饰器（对比模型推理速度）
def timed(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"{func.__name__}: {elapsed:.3f}s")
        return result
    return wrapper

# 2. GPU 显存监控
def track_memory(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        torch.cuda.reset_peak_memory_stats()
        result = func(*args, **kwargs)
        peak_mb = torch.cuda.max_memory_allocated() / 1024**2
        print(f"{func.__name__}: peak GPU memory {peak_mb:.1f} MB")
        return result
    return wrapper

# 3. 批量处理（自动处理单个/列表输入）
def batchable(func):
    @functools.wraps(func)
    def wrapper(inputs, *args, **kwargs):
        if isinstance(inputs, list):
            return [func(item, *args, **kwargs) for item in inputs]
        return func(inputs, *args, **kwargs)
    return wrapper
```

## 核心要点回顾

- 闭包 = 函数捕获外部变量 → 装饰器的前置知识
- 装饰器 = 高阶函数：接受函数 → 返回增强版
- `@functools.wraps` 必须加（保留原函数元信息）
- 带参数装饰器 = 三层嵌套函数
- 装饰器链 = 从下往上执行（@A → @B → def f = A(B(f))）

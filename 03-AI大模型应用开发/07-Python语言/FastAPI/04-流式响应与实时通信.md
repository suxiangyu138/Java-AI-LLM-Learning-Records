## 2. StreamingResponse — LLM Token 流式输出

### 2.1 核心实现

```python
"""
LLM 流式输出 — FastAPI 标准模式
"""
from fastapi import FastAPI
from fastapi.responses import StreamingResponse
import httpx
import json
import asyncio

app = FastAPI()


async def stream_openai(prompt: str):
    """异步生成器：逐 token 转发 OpenAI 的 SSE 流"""
    async with httpx.AsyncClient() as client:
        async with client.stream(
            "POST",
            "https://api.openai.com/v1/chat/completions",
            headers={
                "Authorization": f"Bearer {OPENAI_API_KEY}",
                "Content-Type": "application/json",
            },
            json={
                "model": "gpt-4o",
                "messages": [{"role": "user", "content": prompt}],
                "stream": True,
            },
            timeout=120.0,
        ) as response:
            async for line in response.aiter_lines():
                if line.startswith("data: "):
                    data = line[6:]
                    if data == "[DONE]":
                        yield "data: [DONE]\n\n"
                        break

                    chunk = json.loads(data)
                    delta = chunk["choices"][0].get("delta", {})
                    content = delta.get("content", "")

                    if content:
                        # 将 OpenAI 格式转为标准 SSE
                        sse_data = json.dumps({
                            "token": content,
                            "model": chunk.get("model", ""),
                        })
                        yield f"data: {sse_data}\n\n"


@app.post("/api/chat/stream")
async def chat_stream(request: ChatRequest):
    """
    流式 LLM 对话 — 前端逐字显示
    """
    return StreamingResponse(
        stream_openai(request.messages[-1]["content"]),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",  # Nginx 禁用缓冲
        },
    )
```

### 2.2 多模型流式聚合

```python
async def stream_multi_model(prompt: str):
    """同时流式输出多个模型的结果，用 SSE event 字段区分"""

    async def stream_model(model: str, api_func, prompt: str):
        async for chunk in api_func(prompt):
            yield f"event: {model}\ndata: {chunk}\n\n"

    # 合并多个流：交替输出（哪个先有数据先输出）
    # 使用 asyncio.Queue 合并
    queue = asyncio.Queue()

    async def producer(name: str, gen):
        async for chunk in gen:
            await queue.put(f"event: {name}\ndata: {chunk}\n\n")
        await queue.put(f"event: {name}\ndata: [DONE]\n\n")

    producers = [
        producer("gpt4o", openai_stream(prompt)),
        producer("claude", claude_stream(prompt)),
    ]

    tasks = [asyncio.create_task(p) for p in producers]

    done_count = 0
    while done_count < len(producers):
        data = await queue.get()
        yield data
        if "[DONE]" in data:
            done_count += 1

    await asyncio.gather(*tasks)
```

### 2.3 带缓冲的流式输出

```python
async def stream_with_buffering(prompt: str, buffer_size: int = 5):
    """
    缓冲模式：积累一定 token 数再发送
    减少 HTTP chunk 数量，提高网络效率
    """
    buffer = []
    async for token in raw_token_stream(prompt):
        buffer.append(token)
        if len(buffer) >= buffer_size:
            yield f"data: {json.dumps({'tokens': ''.join(buffer)})}\n\n"
            buffer.clear()

    # 发送剩余 token
    if buffer:
        yield f"data: {json.dumps({'tokens': ''.join(buffer)})}\n\n"
    yield "data: [DONE]\n\n"
```

## 3. SSE (Server-Sent Events)

### 3.1 SSE vs StreamingResponse

```text
StreamingResponse：通用流式（自定义格式）
SSE：标准化的 text/event-stream（浏览器原生 EventSource 支持）

SSE 消息格式：
  event: <事件类型>\n
  id: <消息ID>\n
  data: <数据>\n
  retry: <重连间隔ms>\n
  \n

何时用 SSE 而不是原始 StreamingResponse：
  ✅ 前端是浏览器（原生 EventSource API）
  ✅ 需要多种事件类型区分
  ✅ 需要自动重连
  ✅ 需要消息 ID 去重
```

### 3.2 SSE 实现

```python
"""
SSE — 适合进度推送、日志实时、Agent 状态更新
"""
import asyncio
from fastapi import FastAPI, Request
from fastapi.responses import StreamingResponse
from sse_starlette.sse import EventSourceResponse


@app.get("/api/agent/{agent_id}/events")
async def agent_events(agent_id: str, request: Request):
    """Agent 执行进度推送 — 标准 SSE"""

    async def event_generator():
        queue = subscribe_to_agent_events(agent_id)

        try:
            while True:
                # 等待事件（带客户端断开检测）
                if await request.is_disconnected():
                    break

                try:
                    event = await asyncio.wait_for(queue.get(), timeout=30.0)
                except asyncio.TimeoutError:
                    # 30 秒无事件 → 发心跳
                    yield {"event": "heartbeat", "data": ""}
                    continue

                if event is None:
                    break  # Agent 执行完毕

                yield {
                    "event": event["type"],   # step_start | step_end | progress | error | done
                    "id": event["id"],
                    "data": json.dumps(event["data"], ensure_ascii=False),
                }

        except asyncio.CancelledError:
            pass
        finally:
            unsubscribe_from_agent_events(agent_id, queue)

    return EventSourceResponse(
        event_generator(),
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
        },
    )


# 前端使用（JavaScript）：
# const es = new EventSource(`/api/agent/${agentId}/events`);

# es.addEventListener("progress",  e => updateProgress(JSON.parse(e.data)));
# es.addEventListener("done",      e => { console.log("完成!"); es.close(); });
```

## 4. WebSocket 实时双向通信

### 4.1 基础 WebSocket

```python
from fastapi import WebSocket, WebSocketDisconnect
from typing import Dict


class ConnectionManager:
    """WebSocket 连接管理器"""

    def __init__(self):
        self.active_connections: Dict[str, WebSocket] = {}

    async def connect(self, session_id: str, websocket: WebSocket):
        await websocket.accept()
        self.active_connections[session_id] = websocket

    def disconnect(self, session_id: str):
        self.active_connections.pop(session_id, None)

    async def send_json(self, session_id: str, data: dict):
        ws = self.active_connections.get(session_id)
        if ws:
            await ws.send_json(data)


manager = ConnectionManager()


@app.websocket("/ws/chat/{session_id}")
async def websocket_chat(websocket: WebSocket, session_id: str):
    """WebSocket 对话 — 双向实时通信"""
    await manager.connect(session_id, websocket)

    try:
        while True:
            # 接收客户端消息
            data = await websocket.receive_json()

            if data.get("type") == "stop":
                # 客户端请求停止生成
                await websocket.send_json({"type": "stopped"})
                continue

            # 流式返回 LLM 响应
            async for token in stream_llm(data.get("message", "")):
                await websocket.send_json({
                    "type": "token",
                    "data": token,
                })

            await websocket.send_json({"type": "done"})

    except WebSocketDisconnect:
        manager.disconnect(session_id)
```

### 4.2 WebSocket vs SSE 选型

```text
选 SSE：
  ✅ 只需要服务器推送，不需要客户端频繁发送
  ✅ 浏览器原生支持（EventSource），无需第三方库
  ✅ 自动重连
  ✅ 走 HTTP/2 多路复用
  ❌ 单向（服务器→客户端），客户端发送需另外 POST

选 WebSocket：
  ✅ 需要双向频繁通信（对话、实时协作）
  ✅ 需要客户端主动控制（中断生成、修改参数、语音输入）
  ✅ 二进制数据（TTS 音频流）
  ❌ 浏览器需 WebSocket API
  ❌ 无自动重连（需手动实现）

AI 应用推荐：
  • LLM Token 流 → SSE（简单，自动重连）
  • 实时对话 + 中断 → WebSocket（双向控制）
  • 进度推送 → SSE（单向推送）
```

## 5. MCP Streamable HTTP

```python
"""
MCP Streamable HTTP — 混合响应模式
一次请求 → 多种响应（JSON 头部 + SSE 流式内容）
"""
from fastapi import FastAPI, Request
from fastapi.responses import StreamingResponse
import json


@app.post("/mcp")
async def mcp_streamable(request: Request):
    """
    MCP Streamable HTTP 端点
    根据 Accept 头决定 JSON 还是 SSE 响应
    """
    body = await request.json()

    accept = request.headers.get("accept", "application/json")

    if "text/event-stream" in accept:
        # SSE 流式响应
        return StreamingResponse(
            mcp_sse_stream(body),
            media_type="text/event-stream",
        )
    else:
        # 标准 JSON 响应
        result = await mcp_handle(body)
        return result


async def mcp_sse_stream(request_body: dict):
    """MCP SSE 流：先返回最终结果，再推送进度/日志"""

    method = request_body.get("method")

    if method == "tools/call":
        tool_name = request_body["params"]["name"]

        # 启动异步任务
        task = asyncio.create_task(execute_tool(tool_name, request_body["params"]["arguments"]))

        # 推送进度
        while not task.done():
            await asyncio.sleep(0.5)
            progress = get_tool_progress(tool_name)
            if progress:
                yield f"event: message\ndata: {json.dumps(progress)}\n\n"

        # 返回结果
        result = await task
        yield f"event: message\ndata: {json.dumps(result)}\n\n"
```

> 🎯 **核心要点**：LLM Token 流用 StreamingResponse（灵活），进度推送用 SSE（标准），实时对话用 WebSocket（双向），MCP Server 用 Streamable HTTP（混合模式）

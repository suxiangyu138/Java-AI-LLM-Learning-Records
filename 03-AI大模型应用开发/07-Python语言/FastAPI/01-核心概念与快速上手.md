## 1. FastAPI 是什么

```text
FastAPI = Python 现代 Web 框架，核心优势：

  ① 极速：基于 Starlette + Uvicorn，性能对标 Node.js/Go
  ② 类型安全：Python 类型提示 → 自动校验 + 自动文档
  ③ 原生异步：async/await 一等公民，不阻塞事件循环
  ④ 自动文档：Swagger UI + ReDoc 零配置生成
  ⑤ 标准兼容：OpenAPI 3.1 + JSON Schema
```

```python
# 最简 FastAPI 应用 — 5 行代码
from fastapi import FastAPI

app = FastAPI(title="我的第一个 API")

@app.get("/")
def hello():
    return {"message": "Hello, FastAPI!"}

# 启动：uvicorn main:app --reload
```

## 2. 路由与请求方法

### 2.1 路由装饰器

```python
from fastapi import FastAPI

app = FastAPI()

@app.get("/items")           # GET     — 查询资源
async def list_items():
    return [{"id": 1, "name": "Item 1"}]

@app.post("/items")          # POST    — 创建资源
async def create_item(item: dict):
    return item

@app.put("/items/{item_id}")  # PUT     — 完整更新
async def update_item(item_id: int, item: dict):
    return {"id": item_id, **item}

@app.patch("/items/{item_id}") # PATCH  — 部分更新
async def partial_update(item_id: int, updates: dict):
    return {"id": item_id, "updates": updates}

@app.delete("/items/{item_id}") # DELETE — 删除资源
async def delete_item(item_id: int):
    return {"deleted": item_id}

@app.options("/items")       # OPTIONS — CORS 预检（通常自动处理）
async def items_options(): ...

@app.head("/items")          # HEAD    — 同 GET 但不返回 Body
async def items_head(): ...

# 路由分组 — APIRouter
from fastapi import APIRouter

router = APIRouter(prefix="/api/v1", tags=["v1"])

@router.get("/users")
async def get_users(): ...

app.include_router(router)
```

### 2.2 路由优先级

```python
# 固定路径优先于路径参数
@app.get("/users/me")           # ① 固定路径
async def get_current_user(): ...

@app.get("/users/{user_id}")    # ② 路径参数（兜底）
async def get_user(user_id: int): ...

# 请求 /users/me → 匹配 ①
# 请求 /users/123 → 匹配 ②
```

## 3. 路径参数与查询参数

### 3.1 全部参数类型

```python
from fastapi import FastAPI, Path, Query
from enum import Enum

app = FastAPI()


class Category(str, Enum):
    FRONTEND = "frontend"
    BACKEND = "backend"
    AI = "ai"


@app.get("/articles/{article_id}")
async def get_article(
    # === 路径参数 ===
    article_id: int = Path(
        ...,                          # ... = 必填
        title="文章ID",
        ge=1,                         # greater or equal
        description="文章的唯一标识符",
    ),

    # === 查询参数 ===
    category: Category | None = Query(
        None,
        title="分类",
        description="文章分类过滤",
    ),
    page: int = Query(1, ge=1, description="页码"),
    size: int = Query(20, ge=1, le=100, description="每页数量"),
    keyword: str | None = Query(None, min_length=1, max_length=50),
    sort: str = Query("desc", regex="^(asc|desc)$"),
):
    """
    获取文章详情或列表。

    - **article_id**: 文章ID（路径参数）
    - **category**: 可选分类过滤
    - **page/size**: 分页参数
    """
    return {
        "article_id": article_id,
        "category": category,
        "pagination": {"page": page, "size": size},
        "keyword": keyword,
        "sort": sort,
    }
```

### 3.2 参数校验一览

```python
# 数值校验
@app.get("/items")
async def items(
    price: float = Query(..., gt=0, lt=10000),    # 0 < price < 10000
    quantity: int = Query(..., ge=1, le=999),      # 1 <= quantity <= 999
):

# 字符串校验
@app.get("/search")
async def search(
    q: str = Query(..., min_length=1, max_length=100, regex="^[\\w\\s]+$"),
):

# 数组参数（同名多值）
@app.get("/tags")
async def tags(
    tag: list[str] = Query(["default"]),   # ?tag=a&tag=b → ["a", "b"]
):
```

## 4. 请求体与 Pydantic 模型

### 4.1 请求体定义

```python
from pydantic import BaseModel, Field, EmailStr
from datetime import datetime
from typing import Optional


class ArticleCreate(BaseModel):
    """创建文章请求"""
    title: str = Field(..., min_length=1, max_length=200, description="标题")
    content: str = Field(..., min_length=10, description="正文")
    category: Category = Field(..., description="分类")
    tags: list[str] = Field(default=[], max_length=10)
    published: bool = Field(default=False)
    author_email: EmailStr | None = Field(None)


@app.post("/articles", status_code=201)
async def create_article(article: ArticleCreate):
    # article 已经是校验过的 Pydantic 模型实例
    return {
        "id": 1,
        "title": article.title,
        "created_at": datetime.now(),
    }
```

### 4.2 多种 Body 参数

```python
@app.put("/articles/{article_id}")
async def update_article(
    article_id: int,                             # 路径参数
    article: ArticleCreate,                      # Body 参数（JSON）
    importance: int = Body(..., ge=0, le=5),    # 独立 Body 字段
    token: str = Header(...),                    # Header 参数
    user_agent: str | None = Header(None),       # 可选 Header
    session: str = Cookie(None),                 # Cookie 参数
):
    """
    FastAPI 自动判断参数来源：
    - 路径参数：在 URL 路径中声明
    - Query()：显式声明查询参数，或无类型标注
    - Body()：Pydantic 模型或显式 Body()
    - Header()：请求头
    - Cookie()：Cookie
    """
```

## 5. 依赖注入 (Dependency Injection)

### 5.1 核心机制

```text
FastAPI DI = 核心特色功能

  原理：函数参数 → FastAPI 递归解析 → 注入返回值

  三大用途：
  ① 代码复用：DB Session、认证、权限 → 一次定义，处处使用
  ② 分层解耦：路由函数不直接依赖具体实现
  ③ 可选缓存：同一个请求内多次注入只执行一次
```

### 5.2 常用依赖模式

```python
from fastapi import Depends, HTTPException, status
from typing import Annotated


# ===== 依赖 1：数据库 Session =====
async def get_db():
    """每个请求创建独立 DB Session，请求结束自动关闭"""
    async with AsyncSessionLocal() as session:
        yield session

# ===== 依赖 2：当前用户认证 =====
async def get_current_user(
    token: str = Depends(oauth2_scheme),
    db = Depends(get_db),
):
    """从 Token 中解析当前用户"""
    user = await decode_token(token, db)
    if not user:
        raise HTTPException(status_code=401, detail="未登录")
    return user

# ===== 依赖 3：权限检查（依赖嵌套） =====
async def get_admin_user(
    current_user = Depends(get_current_user),  # 嵌套依赖！
):
    if not current_user.is_admin:
        raise HTTPException(status_code=403, detail="需要管理员权限")
    return current_user

# ===== 使用：Annotated 简写（推荐） =====
DBSession = Annotated[AsyncSession, Depends(get_db)]
CurrentUser = Annotated[User, Depends(get_current_user)]
AdminUser = Annotated[User, Depends(get_admin_user)]

@app.get("/users/me")
async def me(
    db: DBSession,          # 注入 DB Session
    user: CurrentUser,      # 注入当前用户
):
    return user


# ===== 依赖 4：带参数的工厂函数 =====
def require_permission(permission: str):
    """依赖工厂：为不同端点创建不同的权限检查"""
    async def checker(current_user = Depends(get_current_user)):
        if permission not in current_user.permissions:
            raise HTTPException(status_code=403, detail=f"缺少权限: {permission}")
        return current_user
    return checker

@app.post("/admin/users", dependencies=[Depends(require_permission("user:create"))])
async def create_user(...): ...
```

## 6. 自动 API 文档

### 6.1 文档访问

```text
启动服务后自动生成：
  • Swagger UI  — http://localhost:8000/docs   (交互式)
  • ReDoc       — http://localhost:8000/redoc  (美观只读)
  • OpenAPI JSON — http://localhost:8000/openapi.json

文档内容自动生成自：
  • 路由装饰器 (path, methods)
  • 类型提示 (int, str, Pydantic models)
  • Field(description=...), Query(description=...)
  • Docstring (rendered in Swagger)
  • tags=[...] (API 分组)
```

### 6.2 文档增强配置

```python
app = FastAPI(
    title="AI Agent API",
    description="""
    ## AI Agent 后端服务

    提供 Agent 管理、工具调用、RAG 检索等核心 API。

    ### 核心功能
    - **Agent**: 创建和管理 AI Agent
    - **Tools**: MCP 工具注册与调用
    - **RAG**: 文档上传与检索
    - **Chat**: 流式对话接口
    """,
    version="1.0.0",
    terms_of_service="https://example.com/terms",
    contact={
        "name": "API Support",
        "email": "support@example.com",
    },
    license_info={
        "name": "MIT",
    },
    openapi_tags=[
        {"name": "Agent", "description": "Agent 增删改查"},
        {"name": "Tools", "description": "MCP 工具管理"},
        {"name": "Chat", "description": "对话与流式输出"},
    ],
    docs_url="/docs",           # Swagger UI 路径
    redoc_url="/redoc",         # ReDoc 路径
    openapi_url="/openapi.json", # OpenAPI Schema 路径
)
```

> 🎯 **核心要点**：FastAPI = 类型提示驱动的自动校验 + 依赖注入的代码复用 + 零配置的 API 文档。掌握路由参数 + Pydantic + DI 三者，即可构建生产级 API

## 1. 时间序列基础操作

```python
import pandas as pd
import numpy as np

# 生成日期范围
dates = pd.date_range("2024-01-01", "2026-06-30", freq="D")  # 日
dates = pd.date_range("2024-01-01", periods=30, freq="W")     # 周
dates = pd.date_range("2024-01-01", periods=12, freq="ME")    # 月末

# 设置时间索引
df["date"] = pd.to_datetime(df["date"])
df = df.set_index("date")

# 重采样（改变时间频率）
df.resample("W").sum()               # 日 → 周（求和）
df.resample("ME").mean()             # 日 → 月（求均值）
df.resample("QE").agg(["sum","mean"]) # 日 → 季度（多指标）

# 时间切片
df["2025"]               # 2025 全年
df["2025-01":"2025-06"]  # 2025 上半年
df[df.index.month == 12] # 所有 12 月的数据
```

## 2. 趋势与季节性分解

```python
from statsmodels.tsa.seasonal import seasonal_decompose

# 经典分解：Y(t) = Trend(t) + Seasonal(t) + Residual(t)
result = seasonal_decompose(
    df["revenue"],
    model="additive",      # 加法模型（季节波动幅度恒定）
    period=12              # 周期（月度数据 → 12 个月）
)
# result = seasonal_decompose(df["revenue"], model="multiplicative", period=12)

# 提取各分量
trend = result.trend       # 长期趋势
seasonal = result.seasonal # 季节模式
residual = result.resid    # 随机波动（残差）

print(f"趋势方向: {'上升' if trend[-1] > trend[0] else '下降'}")
print(f"季节性强度: {1 - residual.var() / (trend + seasonal).var():.2%}")
```

## 3. 移动平均与平滑

```python
# 简单移动平均
df["ma_7"] = df["value"].rolling(window=7).mean()      # 7 日
df["ma_30"] = df["value"].rolling(window=30).mean()    # 30 日

# 指数加权移动平均（近期权重更大）
df["ewm_7"] = df["value"].ewm(span=7, adjust=False).mean()
df["ewm_alpha"] = df["value"].ewm(alpha=0.3).mean()   # 自定义衰减因子

# 动态滚动窗口
df["rolling_sum_7d"] = df["value"].rolling("7D").sum()

# 滚动统计
df["rolling_mean"] = df["value"].rolling(30).mean()
df["rolling_std"] = df["value"].rolling(30).std()
df["upper_band"] = df["rolling_mean"] + 2 * df["rolling_std"]
df["lower_band"] = df["rolling_mean"] - 2 * df["rolling_std"]
```

## 4. 同比与环比分析

```python
# 环比：与上一个周期相比
df["mom"] = df["revenue"].pct_change()        # Month-over-Month
df["qoq"] = df["revenue"].pct_change(3)       # Quarter-over-Quarter

# 同比：与去年同周期相比
df["yoy"] = df["revenue"].pct_change(12)      # Year-over-Year

# 绝对变化
df["change"] = df["revenue"].diff()            # 差分（与上一期）
df["seasonal_diff"] = df["revenue"].diff(12)    # 季节差分（与去年同月）

# YTD 累计
df["ytd_sum"] = df.groupby(df.index.year)["revenue"].cumsum()
df["ytd_avg"] = df.groupby(df.index.year)["revenue"].expanding().mean().values
```

## 5. 实战：销售预测

```python
# 生成模拟数据：2 年日均销售 + 趋势 + 周末效应 + 噪声
np.random.seed(42)
dates = pd.date_range("2024-01-01", "2025-12-31", freq="D")
trend = np.linspace(100, 150, len(dates))                 # 上升趋势
seasonal = 20 * np.sin(2 * np.pi * np.arange(len(dates)) / 365)  # 年周期
weekend = np.where(dates.dayofweek >= 5, -15, 5)          # 周末效应
noise = np.random.normal(0, 5, len(dates))

df = pd.DataFrame({
    "date": dates,
    "sales": trend + seasonal + weekend + noise
}).set_index("date")

# ---- 简单预测方法 ----
from statsmodels.tsa.holtwinters import ExponentialSmoothing

# 1. 简单指数平滑（无趋势）
# fit1 = SimpleExpSmoothing(df["sales"]).fit()

# 2. Holt 线性趋势
# fit2 = Holt(df["sales"]).fit()

# 3. Holt-Winters（趋势 + 季节，最强）
model = ExponentialSmoothing(
    df["sales"],
    seasonal_periods=7,          # 周周期
    trend="add",
    seasonal="add",
)
fit = model.fit()

# 预测未来 30 天
forecast = fit.forecast(30)
print(f"30 天预测均值: {forecast.mean():.1f}")
print(f"预测范围: [{forecast.min():.1f}, {forecast.max():.1f}]")
```

## 核心要点回顾

- 时间索引三操作：`resample`（重采样）、切片（`"2025-01":"2025-06"`）、频率转换
- 时间序列分解：趋势 + 季节 + 残差 = 原始序列
- 移动平均：`rolling(7)` 简单平均 / `ewm(span=7)` 加权平均
- 同比（`pct_change(12)`）消除季节影响，环比（`pct_change()`）看最新趋势
- Holt-Winters = 趋势 + 季节 + 平滑，经典时间序列预测方法
- Prophet = Facebook 开源，自动处理节假日 + 变点检测，工程首选

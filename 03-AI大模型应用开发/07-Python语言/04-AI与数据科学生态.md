## Pandas

```python
import pandas as pd

# ① 读取数据
df = pd.read_csv("eval_results.csv")
df = pd.read_json("api_responses.jsonl", lines=True)

# ② 分析 LLM 评估结果
df.groupby("model").agg({
    "accuracy": "mean",
    "latency_ms": ["mean", "p50", "p99"],
    "cost_usd": "sum"
})

# ③ 过滤 + 聚合
gpt4_data = df[df["model"] == "gpt-4o"]
print(f"GPT-4o 平均准确率: {gpt4_data['accuracy'].mean():.2%}")
```

## Matplotlib 可视化

```python
import matplotlib.pyplot as plt

# LLM 延迟分布
plt.hist(df["latency_ms"], bins=50, alpha=0.7, label="Qwen-7B")
plt.axvline(df["latency_ms"].quantile(0.99), color='r', 
            linestyle='--', label='P99')
plt.xlabel("延迟 (ms)")
plt.legend()
plt.savefig("latency_dist.png")

# 模型性能对比
models = df.groupby("model")["accuracy"].mean()
plt.bar(models.index, models.values)
plt.ylabel("准确率")
```

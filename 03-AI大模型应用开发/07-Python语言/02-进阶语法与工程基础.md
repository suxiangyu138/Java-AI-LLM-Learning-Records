# 02 - 进阶语法与工程基础

> 🎯 Python 的魔法就藏在进阶语法里 — 装饰器、生成器、上下文管理器、类型注解是读懂 PyTorch/LangChain 源码的钥匙

## 1. 装饰器 @

```python
# 装饰器 = 包装函数 → AOP 的 Python 实现
# Java 类比：@Transactional → 方法执行前后加逻辑

import time
from functools import wraps

def timer(func):
    """测量函数执行时间"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        print(f"{func.__name__}: {time.time()-start:.2f}s")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(1.5)

slow_function()  # slow_function: 1.50s

# 带参数的装饰器
def retry(max_attempts=3):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for i in range(max_attempts):
                try: return func(*args, **kwargs)
                except Exception as e:
                    if i == max_attempts-1: raise
                    time.sleep(2**i)
        return wrapper
    return decorator

@retry(max_attempts=5)
def call_api(): ...  # API 调用自动重试 5 次
```

## 2. 生成器 yield

```python
# 生成器 = 惰性计算 → 处理大数据不爆内存
# Java 类比：Iterator + 懒加载

def read_large_file(path):
    """逐行读取大文件（不加载到内存）"""
    with open(path) as f:
        for line in f:
            yield line.strip()

# 生成器表达式（类似 Stream.map）
squares = (x**2 for x in range(1_000_000))  # 不占内存

# LLM 流式输出本质就是生成器
def stream_tokens(prompt, model):
    for token in model.generate_stream(prompt):
        yield token  # 逐token返回，不等待全部
```

## 3. 上下文管理器 with

```python
# with = 自动资源管理
# Java 类比：try-with-resources

# 自定义上下文管理器
class Timer:
    def __enter__(self):
        self.start = time.time()
        return self
    def __exit__(self, *args):
        self.elapsed = time.time() - self.start

with Timer() as t:
    time.sleep(1)
print(f"{t.elapsed:.2f}s")  # 自动打印耗时

# 用 contextlib 简化
from contextlib import contextmanager

@contextmanager
def temp_env(**kwargs):
    old = {k: os.environ.get(k) for k in kwargs}
    os.environ.update(kwargs)
    try: yield
    finally:
        for k, v in old.items():
            if v is None: del os.environ[k]
            else: os.environ[k] = v

with temp_env(OPENAI_API_KEY="test"):
    assert os.environ["OPENAI_API_KEY"] == "test"
```

## 4. 类型注解

```python
from typing import Optional, List, Dict, Union, Callable

# Python 3.10+ 类型注解
def process_batch(
    items: list[str],               # 3.9+: list[str]
    max_tokens: int = 512,
    callback: Callable[[str], None] | None = None  # 3.10+: X | None
) -> dict[str, str | int]:
    return {"status": "ok", "count": len(items)}

# 配合 dataclass → 自动生成 __init__/__repr__/__eq__
from dataclasses import dataclass

@dataclass
class LLMConfig:
    model: str = "gpt-4o-mini"
    temperature: float = 0.7
    max_tokens: int = 2048

config = LLMConfig(temperature=0.9)  # 其他用默认值
```

try:
    response = api.chat(prompt)

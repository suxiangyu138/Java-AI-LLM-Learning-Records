## 1. 认证方案对比

```text
┌──────────────┬──────────────┬──────────────┬──────────────┐
│ 方案          │ 适用场景      │ 复杂度        │ AI 应用场景   │
├──────────────┼──────────────┼──────────────┼──────────────┤
│ JWT Bearer   │ Web 应用用户  │ 低            │ Chat UI      │
│ OAuth2 + JWT │ 第三方集成    │ 中            │ MCP OAuth   │
│ API Key      │ 服务间调用    │ 最低          │ LLM Proxy   │
│ Session      │ 传统 Web      │ 中            │ 管理后台     │
│ mTLS         │ 微服务间      │ 高            │ 内部服务     │
└──────────────┴──────────────┴──────────────┴──────────────┘

AI 应用推荐组合：
  • 用户 API  → JWT Bearer（有状态过期 + Scope）
  • MCP/SDK  → API Key（简单高效）
  • 公开 OAuth → OAuth2 + JWT（标准化）
```

## 2. JWT Token 认证

### 2.1 完整 JWT 实现

```python
"""
JWT 认证：密钥签名 + 过期控制 + Scope 权限
"""
import jwt
from datetime import datetime, timedelta, timezone
from passlib.context import CryptContext  # 密码哈希
from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from pydantic import BaseModel
from typing import Annotated

# ===== 配置 =====
SECRET_KEY = "your-secret-key-change-in-production"
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 60
REFRESH_TOKEN_EXPIRE_DAYS = 30

# ===== 密码哈希 =====
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

# ===== Token 编解码 =====
def create_access_token(data: dict, expires_delta: timedelta | None = None) -> str:
    """创建 JWT Access Token"""
    to_encode = data.copy()
    expire = datetime.now(timezone.utc) + (expires_delta or timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES))
    to_encode.update({"exp": expire, "type": "access"})
    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

def create_refresh_token(data: dict) -> str:
    """创建 Refresh Token"""
    to_encode = data.copy()
    expire = datetime.now(timezone.utc) + timedelta(days=REFRESH_TOKEN_EXPIRE_DAYS)
    to_encode.update({"exp": expire, "type": "refresh"})
    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

def decode_token(token: str) -> dict:
    """解码并验证 Token"""
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        return payload
    except jwt.ExpiredSignatureError:
        raise HTTPException(status_code=401, detail="Token 已过期")
    except jwt.InvalidTokenError:
        raise HTTPException(status_code=401, detail="Token 无效")

# ===== 安全方案 =====
security = HTTPBearer()

# ===== 依赖：获取当前用户 =====
async def get_current_user(
    credentials: Annotated[HTTPAuthorizationCredentials, Depends(security)],
    db: DBSession,
) -> User:
    """从 JWT Token 中提取用户"""
    payload = decode_token(credentials.credentials)

    user_id = payload.get("sub")
    if not user_id:
        raise HTTPException(status_code=401, detail="Token 中无用户标识")

    user = await db.get(User, int(user_id))
    if not user or not user.is_active:
        raise HTTPException(status_code=401, detail="用户不存在或已禁用")

    return user

CurrentUser = Annotated[User, Depends(get_current_user)]

# ===== 登录/注册/刷新端点 =====
from fastapi import APIRouter

auth_router = APIRouter(prefix="/auth", tags=["认证"])


class LoginRequest(BaseModel):
    email: str
    password: str


class TokenResponse(BaseModel):
    access_token: str
    refresh_token: str
    token_type: str = "bearer"
    expires_in: int


@auth_router.post("/login", response_model=TokenResponse)
async def login(request: LoginRequest, db: DBSession):
    """用户登录 — 返回 Access + Refresh Token"""
    # 查找用户
    stmt = select(User).where(User.email == request.email)
    result = await db.execute(stmt)
    user = result.scalar_one_or_none()

    if not user or not pwd_context.verify(request.password, user.hashed_password):
        raise HTTPException(status_code=401, detail="邮箱或密码错误")

    # 生成 Token
    token_data = {
        "sub": str(user.id),
        "email": user.email,
        "scopes": ["user:read", "chat:write", "admin"] if user.is_superuser else ["user:read", "chat:write"],
    }

    return {
        "access_token": create_access_token(token_data),
        "refresh_token": create_refresh_token({"sub": str(user.id)}),
        "expires_in": ACCESS_TOKEN_EXPIRE_MINUTES * 60,
    }


@auth_router.post("/refresh", response_model=TokenResponse)
async def refresh(credentials: Annotated[HTTPAuthorizationCredentials, Depends(security)]):
    """刷新 Access Token"""
    payload = decode_token(credentials.credentials)
    if payload.get("type") != "refresh":
        raise HTTPException(status_code=401, detail="不是 Refresh Token")

    user_id = payload["sub"]
    # ... 重新生成（省略用户状态检查）
    return {
        "access_token": create_access_token({"sub": user_id}),
        "refresh_token": create_refresh_token({"sub": user_id}),
        "expires_in": ACCESS_TOKEN_EXPIRE_MINUTES * 60,
    }


@auth_router.post("/register")
async def register(username: str, email: str, password: str, db: DBSession):
    """用户注册"""
    existing = await db.execute(select(User).where(User.email == email))
    if existing.scalar_one_or_none():
        raise HTTPException(status_code=400, detail="邮箱已被注册")

    user = User(
        username=username,
        email=email,
        hashed_password=pwd_context.hash(password),
    )
    db.add(user)
    await db.commit()
    return {"message": "注册成功", "user_id": user.id}
```

### 2.2 Scope 权限控制

```python
def require_scope(required_scope: str):
    """Scope 权限依赖工厂"""

    async def scope_checker(
        credentials: Annotated[HTTPAuthorizationCredentials, Depends(security)],
    ):
        payload = decode_token(credentials.credentials)
        scopes = payload.get("scopes", [])

        if required_scope not in scopes:
            raise HTTPException(
                status_code=403,
                detail=f"权限不足：需要 {required_scope} 权限",
            )
        return payload

    return scope_checker


# 使用
@app.get("/admin/stats", dependencies=[Depends(require_scope("admin"))])
async def admin_stats(): ...


@app.post("/api/chat", dependencies=[Depends(require_scope("chat:write"))])
async def chat(request: ChatRequest, user: CurrentUser): ...
```

```python
from fastapi import Form

# OAuth2 方案（自动生成 Swagger UI 的 Authorize 按钮）
oauth2_scheme = OAuth2PasswordBearer(
    tokenUrl="/auth/token",           # 获取 Token 的端点
    scopes={
        "user:read": "读取用户信息",
        "chat:write": "发送聊天消息",
        "admin": "管理员权限",
    },
)


@app.post("/auth/token")
async def login_for_access_token(
    form_data: Annotated[OAuth2PasswordRequestForm, Depends()],  # username + password + scope 表单
    db: DBSession,
) -> TokenResponse:
    """
    OAuth2 标准 Token 端点
    Swagger UI 会自动识别并显示 Authorize 按钮
    """
    user = await authenticate_user(db, form_data.username, form_data.password)
    if not user:
        raise HTTPException(status_code=401)

    # 用户请求的 scope 必须在其拥有的 scope 之内
    requested_scopes = form_data.scopes
    user_scopes = ["user:read", "chat:write"]
    if "admin" in user_scopes:
        user_scopes.append("admin")

    granted_scopes = [s for s in requested_scopes if s in user_scopes]

    token_data = {
        "sub": str(user.id),
        "scopes": granted_scopes,
    }
    return {
        "access_token": create_access_token(token_data),
        "refresh_token": create_refresh_token({"sub": str(user.id)}),
        "token_type": "bearer",
        "expires_in": ACCESS_TOKEN_EXPIRE_MINUTES * 60,
    }
```

## 4. API Key 认证

### 4.1 API Key 实现

```python
from fastapi.security import APIKeyHeader

api_key_header = APIKeyHeader(name="X-API-Key", auto_error=False)


async def get_api_key_user(
    api_key: Annotated[str | None, Depends(api_key_header)],
    bearer: Annotated[HTTPAuthorizationCredentials | None, Depends(HTTPBearer(auto_error=False))],
    db: DBSession,
) -> User | None:
    """同时支持 API Key 和 Bearer Token"""

    if bearer:
        # Bearer Token 优先
        return await get_current_user(bearer, db)

    if api_key:
        # API Key 认证
        from hashlib import sha256
        key_hash = sha256(api_key.encode()).hexdigest()

        stmt = (
            select(User)
            .join(APIKey)
            .where(APIKey.key_hash == key_hash, APIKey.is_active == True)
        )
        result = await db.execute(stmt)
        user = result.scalar_one_or_none()

        if not user:
            raise HTTPException(status_code=401, detail="API Key 无效")

        # 更新最后使用时间
        await db.execute(
            update(APIKey)
            .where(APIKey.key_hash == key_hash)
            .values(last_used_at=datetime.utcnow())
        )
        await db.commit()

        return user

    raise HTTPException(status_code=401, detail="请提供 Bearer Token 或 X-API-Key")


# 注意：API Key 不如 JWT 安全（无法过期、无法区分用户权限）
# 适用于：CI/CD、内部服务调用、开发调试
```

## 6. 速率限制 (Rate Limiting)

### 6.1 基于 Redis 的限流

```python
"""
AI API 特别需要速率限制 — 防止 Token 滥用
"""
import redis.asyncio as redis
from fastapi import Request
import time


class RateLimiter:
    """滑动窗口速率限制器（基于 Redis）"""

    def __init__(self, redis_url: str = "redis://localhost:6379"):
        self.redis = redis.from_url(redis_url)

    async def check(
        self,
        key: str,
        max_requests: int,
        window_seconds: int = 60,
    ) -> tuple[bool, int, int]:
        """
        滑动窗口限流检查

        返回：(是否允许, 剩余次数, 重置秒数)
        """
        now = time.time()
        window_start = now - window_seconds

        # Lua 脚本：原子性地清理旧记录 + 检查 + 添加
        lua_script = """
        local key = KEYS[1]
        local now = tonumber(ARGV[1])
        local window_start = tonumber(ARGV[2])
        local max_requests = tonumber(ARGV[3])
        local window_seconds = tonumber(ARGV[4])

        -- 清理窗口外的记录
        redis.call('ZREMRANGEBYSCORE', key, 0, window_start)

        -- 当前窗口内的请求数
        local current = redis.call('ZCARD', key)

        if current < max_requests then
            -- 允许：添加当前请求时间戳
            redis.call('ZADD', key, now, now .. ':' .. current)
            redis.call('EXPIRE', key, window_seconds)
            return {1, max_requests - current - 1, window_seconds}
        else
            -- 拒绝
            local oldest = redis.call('ZRANGE', key, 0, 0, 'WITHSCORES')[2]
            local reset_in = math.ceil(oldest + window_seconds - now)
            return {0, 0, reset_in}
        end
        """

        allowed, remaining, reset = await self.redis.eval(
            lua_script,
            1, key,
            now, window_start, max_requests, window_seconds,
        )

        return bool(allowed), int(remaining), int(reset)


# ===== FastAPI 中间件 =====
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

limiter = RateLimiter()


@app.middleware("http")
async def rate_limit_middleware(request: Request, call_next):
    """全局限流中间件"""

    # 获取用户标识（Token → user_id，或 IP）
    user_id = request.headers.get("X-User-ID") or request.client.host
    key = f"rate_limit:{user_id}"

    # 检查限流
    allowed, remaining, reset = await limiter.check(
        key, max_requests=100, window_seconds=60
    )

    if not allowed:
        return JSONResponse(
            status_code=429,
            content={
                "error": "RATE_LIMIT_EXCEEDED",
                "message": f"请求频率超限，请在 {reset} 秒后重试",
                "retry_after": reset,
            },
            headers={
                "Retry-After": str(reset),
                "X-RateLimit-Limit": "100",
                "X-RateLimit-Remaining": "0",
                "X-RateLimit-Reset": str(reset),
            },
        )

    response = await call_next(request)
    response.headers["X-RateLimit-Remaining"] = str(remaining)
    return response
```

### 6.2 Token 用量预算

```python
"""
AI 应用特殊限流：Token 用量预算
"""
class TokenBudgetLimiter:
    """Token 预算限流 — 限制每日/每月 Token 消耗"""

    async def check_and_deduct(
        self,
        user_id: int,
        tokens: int,
        daily_limit: int = 100_000,
    ) -> bool:
        """检查 Token 预算，超限则拒绝"""
        today = datetime.now(timezone.utc).strftime("%Y-%m-%d")
        key = f"token_budget:{user_id}:{today}"

        current = int(await self.redis.get(key) or 0)

        if current + tokens > daily_limit:
            return False

        await self.redis.incr(key, tokens)
        await self.redis.expire(key, 86400)  # 明天自动清理
        return True
```

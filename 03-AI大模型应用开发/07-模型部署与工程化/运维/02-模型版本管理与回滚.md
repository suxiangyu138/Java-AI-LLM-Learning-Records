### 自动回滚条件

```python
class AutoRollback:
    """自动回滚监控"""
    
    def __init__(self, baseline_metrics, threshold):
        self.baseline = baseline_metrics  # v4 的指标基线
        self.threshold = threshold        # 允许的退化比例
    
    def check_and_rollback(self, canary_metrics):
        """检查金丝雀指标 → 不达标则回滚"""
        triggers = []
        
        # 错误率超过基线 2×
        if canary_metrics.error_rate > self.baseline.error_rate * 2:
            triggers.append("错误率超标")
        
        # P99 延迟超过基线 1.5×
        if canary_metrics.p99_latency > self.baseline.p99_latency * 1.5:
            triggers.append("延迟超标")
        
        # 用户负反馈率超过基线 3×
        if canary_metrics.negative_rate > self.baseline.negative_rate * 3:
            triggers.append("用户满意度下降")
        
        if triggers:
            logger.error(f"金丝雀不达标: {triggers} → 执行回滚！")
            self.rollback()
            return True
        return False
    
    def rollback(self):
        # K8s 回滚到上一个版本
        subprocess.run([
            "kubectl", "rollout", "undo", "deployment/llm-inference"
        ])
```

## 4. Prompt 版本管理

```python
# Prompt 模板也纳入版本管理
class PromptRegistry:
    def __init__(self):
        self.prompts = {}  # name → [(version, template, eval_score), ...]
    
    def register(self, name, template, score):
        if name not in self.prompts:
            self.prompts[name] = []
        version = len(self.prompts[name]) + 1
        self.prompts[name].append((version, template, score))
        # Git commit 记录
        self._git_commit(f"prompt/{name}/v{version}")
    
    def get_best(self, name):
        """获取评估分数最高的 Prompt 版本"""
        return max(self.prompts[name], key=lambda x: x[2])
    
    def rollback(self, name, version):
        """回滚到指定 Prompt 版本"""
        return self.prompts[name][version - 1][1]
```

### 关键要点

```text
版本回滚 Checklist：
  □ 模型权重有备份（S3/HuggingFace）
  □ 配置文件版本化（Git）
  □ Prompt 模板版本化（Prompt Registry）
  □ LoRA Adapter 与 Base Model 版本对应关系
  □ 回滚演练（每季度至少 1 次）
  □ 推理引擎版本锁定（vLLM 版本要和模型版本绑定）
```

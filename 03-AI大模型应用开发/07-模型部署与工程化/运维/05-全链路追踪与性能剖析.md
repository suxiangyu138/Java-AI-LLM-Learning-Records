> 🎯 用户说"回答好慢" — 是 Embedding 慢？检索慢？还是 LLM 推理慢？没有全链路追踪，你只能靠猜

## 1. OpenTelemetry 集成

```python
from opentelemetry import trace
from opentelemetry.instrumentation.openai import OpenAIInstrumentor

# 自动注入 OpenAI 追踪
OpenAIInstrumentor().instrument()

tracer = trace.get_tracer(__name__)

@tracer.start_as_current_span("rag_query")
def rag_query(question):
    with tracer.start_as_current_span("embedding"):
        q_vec = embed(question)          # Span: Embedding 耗时
    
    with tracer.start_as_current_span("retrieval"):
        docs = vector_search(q_vec)      # Span: 检索耗时
    
    with tracer.start_as_current_span("llm_generation"):
        answer = llm(question, docs)     # Span: LLM 耗时
    
    return answer

# Trace 输出：
# rag_query (2.5s)
#   ├── embedding (120ms)
#   ├── retrieval (80ms)
#   └── llm_generation (2.3s)
#         ├── prefill (500ms)     ← Tokenize+编码
#         └── decode (1.8s)       ← 逐 token 生成
```

## 3. 慢请求定位

```python
# 慢请求自动捕获
class SlowQueryDetector:
    def __init__(self, p99_threshold_ms=5000):
        self.threshold = p99_threshold_ms
        self.slow_queries = []
    
    def check(self, trace_id, operation, duration_ms, context):
        if duration_ms > self.threshold:
            self.slow_queries.append({
                "trace_id": trace_id,
                "operation": operation,
                "duration_ms": duration_ms,
                "context": context,
                "timestamp": time.time()
            })
            logger.warning(f"慢查询! {operation} 耗时 {duration_ms}ms")
```

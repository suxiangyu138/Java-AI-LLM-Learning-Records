## 3. 核心代码

```python
"""main.py — 生产化入口（治理层组装，业务层复用阶段 3 代码）"""
from agent_guard import CircuitBreaker          # 01 篇
from policy_engine import PolicyEngine          # 05 篇
from input_guard import check_injection         # 05 篇
from context_budget import ContextBudget        # 04 篇
from cost_control import CostController         # 05 篇
from agent.business import query_agent, extract_agent   # 阶段 3 成果


class ProductionAgent:
    """业务 Agent + 治理层的组合（洋葱模型：输入→治理→业务→校验→输出）"""

    def __init__(self):
        self.breaker = CircuitBreaker(max_turns=25, max_cost_usd=2.0)
        self.cost = CostController(session_budget=2.0, daily_budget=50.0)
        self.policy = PolicyEngine(approve_callback=self._human_approve)
        self.budget = ContextBudget()
        self.sessions = {}

    def _human_approve(self, tool_name, args):
        """HITL：高危动作人工确认（08 篇；生产接审批系统/IM）"""
        print(f"[审批] 高危操作 {tool_name} {args} → 输入 y/n: ", end="")
        return input().strip().lower() == "y"

    def handle(self, user_text: str, session: str = "s-1") -> str:
        # ① 输入防护（请求前）
        flagged, _ = check_injection(user_text)
        if flagged:
            return "（检测到可疑输入，已拒绝处理）"

        # ② 熔断与预算（请求前）
        if not self.breaker.check() or not self.cost.check():
            return "（会话已达安全限制，请重新开始）"

        # ③ 路由（业务层）
        agent = extract_agent if route(user_text) == "extract" else query_agent
        result = agent.invoke(
            {"messages": [("user", user_text)]},
            config={"configurable": {"thread_id": session}},
        )
        answer = result["messages"][-1].content

        # ④ 输出校验（请求后）：敏感检测 + 格式抽查
        sensitive = check_sensitive(answer)
        if sensitive:
            return "（回答包含敏感信息，已拦截展示）"

        # ⑤ 记账与遥测（成本标签）
        usage = estimate_usage(result)
        self.breaker.record(cost=usage, tool_calls=count_tool_calls(result))
        self.cost.record(usage)
        return answer


def route(text: str) -> str:
    return "extract" if ("提取" in text or "记一下" in text) else "query"
```

## 4. 七项验收标准

| # | 验收项 | 判定 |
|:---:|--------|------|
| 1 | 三重熔断生效 | 注入"循环计算"类请求，轮数/成本熔断触发并有用户提示 |
| 2 | 输入防护生效 | 注入模式输入被拒绝（不进入模型） |
| 3 | 高危动作审批 | 触发 risky 工具（如 send_email 模拟）→ 人工 y/n 生效 |
| 4 | 输出敏感检测 | 构造含手机号的回答 → 拦截展示 |
| 5 | 评估套件跑通 | golden dataset ≥50 条 + 五模式测试 + 门禁脚本通过 |
| 6 | 遥测可见 | 一次完整会话的 span 树可查（工具/成本/耗时标签） |
| 7 | 版本化可回滚 | prompt/工具/护栏变更走 CI 门禁；回滚一条命令 |

### 2.1 Spring AI 方式

```yaml
# application.yml — 同时配置多个模型
spring:
  ai:
    openai:
      api-key: ${OPENAI_API_KEY}
      chat:
        options:
          model: gpt-4o-mini      # 默认模型
          temperature: 0.7

    anthropic:
      api-key: ${ANTHROPIC_API_KEY}
      chat:
        options:
          model: claude-sonnet-5

    vertex:
      ai:
        gemini:
          project-id: ${GCP_PROJECT}
          location: us-central1
          chat:
            options:
              model: gemini-2.5-flash
```

```java
// 多模型 ChatClient Bean
@Configuration
public class MultiModelConfig {

    @Bean
    @Qualifier("openai")
    public ChatClient openAiClient(ChatClient.Builder builder) {
        return builder.build();  // 默认用 OpenAI
    }

    @Bean
    @Qualifier("claude")
    public ChatClient claudeClient(
            ChatClient.Builder builder,
            AnthropicChatModel anthropicModel) {
        return builder
            .defaultOptions(
                AnthropicChatOptions.builder()
                    .withModel("claude-opus-5")
                    .withTemperature(0.3)
                    .build()
            )
            .build();
    }
}
```

### 2.2 LangChain4j 方式

```java
// LangChain4j —— 统一抽象，切换模型只需换 Builder
public class ModelFactory {

    // OpenAI
    public static ChatLanguageModel openAi() {
        return OpenAiChatModel.builder()
            .apiKey(System.getenv("OPENAI_API_KEY"))
            .modelName("gpt-4o")
            .temperature(0.7)
            .build();
    }

    // Anthropic Claude
    public static ChatLanguageModel claude() {
        return AnthropicChatModel.builder()
            .apiKey(System.getenv("ANTHROPIC_API_KEY"))
            .modelName("claude-sonnet-5")
            .maxTokens(4096)
            .build();
    }

    // Google Gemini
    public static ChatLanguageModel gemini() {
        return VertexAiGeminiChatModel.builder()
            .project(System.getenv("GCP_PROJECT"))
            .location("us-central1")
            .modelName("gemini-2.5-flash")
            .build();
    }

    // 关键优势：所有模型都实现 ChatLanguageModel 接口
    // 下游代码不需要改，切换模型 = 换一个 builder
    public static ChatLanguageModel getModel(String provider) {
        return switch (provider) {
            case "openai"  -> openAi();
            case "claude"  -> claude();
            case "gemini"  -> gemini();
            default        -> openAi();
        };
    }
}
```

### 4.3 本地 vs 云 API 决策

| 维度 | 本地部署（Ollama/vLLM） | 云 API |
|------|:---:|:---:|
| **成本** | 固定硬件成本 | 按量付费 |
| **延迟** | 取决于 GPU | 网络延迟 + 推理延迟 |
| **隐私** | ✅ 数据不出企业 | ⚠️ 数据经过第三方 |
| **模型选择** | 仅开源模型 | 最前沿的闭源模型 |
| **运维** | 需要 GPU 运维 | 零运维 |
| **适用体量** | 日均 > 100 万 Token | 日均 < 100 万 Token |
| **合规** | 满足数据不出境要求 | 需确认合规性 |

## 5. ModelRouter：智能路由

### 5.1 LangChain4j ModelRouter

```java
// LangChain4j 1.12+ 的 ModelRouter —— 按复杂度自动路由
@Service
public class IntelligentRouter {

    private final ChatLanguageModel router;

    public IntelligentRouter() {
        // 成本模型
        ChatLanguageModel cheapModel = OpenAiChatModel.builder()
            .modelName("gpt-4o-mini")  // 简单任务
            .build();

        ChatLanguageModel mediumModel = DeepSeekChatModel.builder()
            .modelName("deepseek-chat")  // 中等任务
            .build();

        ChatLanguageModel premiumModel = OpenAiChatModel.builder()
            .modelName("gpt-4o")  // 复杂任务
            .build();

        ChatLanguageModel fallbackModel = OllamaChatModel.builder()
            .modelName("qwen2.5:7b")  // 兜底模型
            .build();

        // ModelRouter 配置
        this.router = new ModelRouter()
            .addModel("cheap", cheapModel)
            .addModel("medium", mediumModel)
            .addModel("premium", premiumModel)
            .addModel("fallback", fallbackModel)

            // 路由规则：按复杂度自动选择
            .complexityRouter(complexity -> {
                if (complexity == Complexity.LOW) return "cheap";
                if (complexity == Complexity.MEDIUM) return "medium";
                return "premium";
            })

            // 降级规则：失败时自动切换
            .fallbackStrategy(List.of("premium", "medium", "cheap", "fallback"))

            .build();
    }

    public String chat(String message) {
        return router.generate(message);
    }
}
```

### 5.2 手动路由策略

```java
@Service
public class ManualModelRouter {

    private final Map<String, ChatLanguageModel> models;

    /**
     * 基于任务特征的手动路由
     */
    public ChatLanguageModel route(Task task) {
        // 规则 1：代码生成/审查 → Claude
        if (task.getType().equals("code") || task.getType().equals("review")) {
            return models.get("claude");
        }

        // 规则 2：中文内容 → 通义千问
        if (task.getLanguage().equals("zh")) {
            return models.get("qwen");
        }

        // 规则 3：简单分类/提取 → 小模型
        if (task.getComplexity() == Complexity.LOW) {
            return models.get("mini");
        }

        // 规则 4：复杂推理 → 大模型
        if (task.getComplexity() == Complexity.HIGH) {
            return models.get("premium");
        }

        // 默认：性价比模型
        return models.get("default");
    }
}
```

### 5.3 Spring AI 中的模型切换

```java
@Service
public class AdaptiveChatService {

    private final ChatClient openAiClient;
    private final ChatClient deepSeekClient;
    private final ChatClient fallbackClient;

    public String chat(String message) {
        // 基于上下文长度选择模型
        int estimatedTokens = estimateTokens(message);

        if (estimatedTokens > 64000) {
            return deepSeekClient.prompt().user(message).call().content();
        }

        try {
            return openAiClient.prompt().user(message).call().content();
        } catch (Exception e) {
            log.warn("OpenAI 调用失败，降级到 DeepSeek", e);
            try {
                return deepSeekClient.prompt().user(message).call().content();
            } catch (Exception e2) {
                log.error("DeepSeek 也失败，使用本地兜底", e2);
                return fallbackClient.prompt().user(message).call().content();
            }
        }
    }
}
```

## 6. 多模型降级策略

### 6.1 降级链设计

```text
降级链：主模型 → 备用模型 → 本地模型 → 规则/缓存兜底

┌──────────────────────────────────────────────────────┐
│                                                      │
│  Level 1: 主模型（最高质量）                           │
│  Claude Opus / GPT-4o                                │
│  超时: 120s                                          │
│      ↓ 失败/超时/配额不足                             │
│                                                      │
│  Level 2: 备用模型（平衡质量与成本）                    │
│  DeepSeek V4 / Qwen-Max                              │
│  超时: 60s                                           │
│      ↓ 失败/超时                                      │
│                                                      │
│  Level 3: 本地模型（零成本兜底）                        │
│  Ollama qwen2.5:7b / Llama 3.1 8B                    │
│  超时: 30s                                           │
│      ↓ 失败                                           │
│                                                      │
│  Level 4: 规则/缓存（确定性兜底）                       │
│  关键词匹配 / 缓存答案 / 降级提示                       │
│  100% 可用                                           │
│                                                      │
└──────────────────────────────────────────────────────┘
```

### 6.2 实现代码

```java
@Component
@Slf4j
public class DegradingModelChain {

    private final List<ModelLevel> chain;

    public DegradingModelChain() {
        this.chain = List.of(
            new ModelLevel("primary",
                OpenAiChatModel.builder()
                    .modelName("gpt-4o").build(),
                Duration.ofSeconds(120)),

            new ModelLevel("secondary",
                DeepSeekChatModel.builder()
                    .modelName("deepseek-chat").build(),
                Duration.ofSeconds(60)),

            new ModelLevel("local",
                OllamaChatModel.builder()
                    .modelName("qwen2.5:7b").build(),
                Duration.ofSeconds(30))
        );
    }

    /**
     * 按降级链顺序尝试，直到成功或全部失败
     */
    public ModelResponse generateWithDegradation(String prompt) {
        List<String> attemptedLevels = new ArrayList<>();

        for (ModelLevel level : chain) {
            attemptedLevels.add(level.name());
            try {
                log.info("尝试 {} 模型...", level.name());
                String result = level.model()
                    .generate(prompt, level.timeout());
                log.info("✅ {} 模型成功", level.name());

                return new ModelResponse(result, level.name(), attemptedLevels);

            } catch (TimeoutException e) {
                log.warn("⏱️ {} 模型超时 ({}s)，降级...",
                    level.name(), level.timeout().getSeconds());
            } catch (Exception e) {
                log.warn("❌ {} 模型失败: {}，降级...",
                    level.name(), e.getMessage());
            }
        }

        // 全部失败 —— 返回友好降级提示
        log.error("🚨 所有模型不可用！返回降级提示");
        return new ModelResponse(
            "抱歉，AI 服务暂时不可用，请稍后重试或联系技术支持。",
            "fallback", attemptedLevels);
    }

    record ModelLevel(
        String name,
        ChatLanguageModel model,
        Duration timeout
    ) {}
}
```

### 7.2 成本控制策略

```java
@Component
public class CostController {

    private final MeterRegistry meterRegistry;  // Micrometer

    // 策略 1：短问题用小模型
    public ChatLanguageModel selectModel(String prompt) {
        int tokens = estimateTokens(prompt);

        if (tokens < 500) return getModel("mini");       // 简单问题
        if (tokens < 4000) return getModel("standard");   // 一般问题
        return getModel("premium");                       // 复杂问题
    }

    // 策略 2：优先使用带缓存的结果
    public String chatWithCache(String prompt) {
        String cacheKey = hashPrompt(prompt);
        String cached = cacheService.get(cacheKey);

        if (cached != null) {
            meterRegistry.counter("llm.cache.hit").increment();
            return cached;  // 零成本！
        }

        meterRegistry.counter("llm.cache.miss").increment();
        String result = selectModel(prompt).generate(prompt);
        cacheService.set(cacheKey, result, Duration.ofHours(1));
        return result;
    }

    // 策略 3：Token 消耗实时监控和限流
    public void checkBudget() {
        double monthUsage = getMonthlyCreditUsage();
        double budget = getMonthlyBudget();

        if (monthUsage > budget * 0.8) {
            log.warn("⚠️ 当月用量已达 80%，自动切换到经济模式");
            setGlobalModelLevel("economy");
        }
        if (monthUsage > budget * 0.95) {
            log.error("🚨 当月用量达 95%，限制仅使用免费模型");
            setGlobalModelLevel("free_only");
        }
    }
}
```

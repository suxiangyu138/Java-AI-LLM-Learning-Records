## 2. GraalVM 原生镜像

### 2.1 Spring Boot 3.5 + GraalVM

```xml
<!-- pom.xml — 添加 GraalVM Native 支持 -->
<plugin>
    <groupId>org.graalvm.buildtools</groupId>
    <artifactId>native-maven-plugin</artifactId>
    <version>0.10.5</version>
    <executions>
        <execution>
            <goals>
                <goal>compile-no-fork</goal>
            </goals>
        </execution>
    </executions>
    <configuration>
        <imageName>${project.artifactId}</imageName>
        <mainClass>com.example.AiApplication</mainClass>
        <buildArgs>
            <arg>--enable-url-protocols=https</arg>
            <arg>-H:+ReportExceptionStackTraces</arg>
            <arg>-march=compatibility</arg>   <!-- 兼容性优化 -->
        </buildArgs>
    </configuration>
</plugin>
```

### 2.2 Native Hint 配置

```java
// GraalVM 原生镜像需要反射/代理/资源等 Hint
@Configuration
@ImportRuntimeHints(MyRuntimeHints.class)
public class NativeConfig {

    // Spring AI 大部分模块已自动提供 Hint
    // 自定义组件需要手动注册
}

class MyRuntimeHints implements RuntimeHintsRegistrar {

    @Override
    public void registerHints(
            RuntimeHints hints, ClassLoader classLoader) {

        // 注册需要反射的类（如动态代理的 AiService）
        hints.reflection()
            .registerType(MyAiService.class,
                MemberCategory.INVOKE_DECLARED_METHODS,
                MemberCategory.DECLARED_FIELDS);

        // 注册资源文件
        hints.resources()
            .registerPattern("prompts/*.txt")
            .registerPattern("config/*.yml");
    }
}
```

### 2.3 构建命令

```bash
# Maven 构建原生镜像
./mvnw -Pnative native:compile

# 或直接运行（生成临时镜像）
./mvnw -Pnative spring-boot:build-image

# Gradle 构建原生镜像
./gradlew nativeCompile

# 生成的可执行文件
./target/ai-application

# Docker 镜像
docker build -f Dockerfile.native -t ai-app:native .
```

### 3.2 Quarkus AI Service

```java
@ApplicationScoped
public class QuarkusAiService {

    @Inject
    @OpenAi(modelName = "gpt-4o-mini")
    ChatLanguageModel model;

    @Inject
    @Qwen(modelName = "qwen-max")
    ChatLanguageModel qwenModel;

    /**
     * Quarkus 响应式 AI 端点
     */
    public Uni<String> chatAsync(String message) {
        return Uni.createFrom().item(() -> model.generate(message));
    }

    // Quarkus 的优势：响应式 + 非阻塞 I/O
    // 高并发场景下资源利用率远超传统 Servlet 模型
}
```

```properties
# 日志

```

## 4. 可观测性体系

### 4.1 监控指标定义

```java
@Component
public class AiObservability {

    private final MeterRegistry registry;

    // LLM 调用指标
    @EventListener
    public void onLlmCall(LlmCallEvent event) {
        // 调用计数
        Counter.builder("llm.calls.total")
            .tag("model", event.getModelName())
            .tag("status", event.getStatus())
            .register(registry)
            .increment();

        // 延迟直方图
        Timer.builder("llm.calls.duration")
            .tag("model", event.getModelName())
            .register(registry)
            .record(event.getDuration());

        // Token 消耗
        Counter.builder("llm.tokens.input")
            .tag("model", event.getModelName())
            .register(registry)
            .increment(event.getInputTokens());

        Counter.builder("llm.tokens.output")
            .tag("model", event.getModelName())
            .register(registry)
            .increment(event.getOutputTokens());
    }

    // RAG 检索指标
    public void recordRetrieval(String source, long durationMs,
                                 int resultCount, float avgScore) {
        Timer.builder("rag.retrieval.duration")
            .tag("source", source)
            .register(registry)
            .record(durationMs, TimeUnit.MILLISECONDS);

        Gauge.builder("rag.retrieval.result_count", resultCount,
                Number::intValue)
            .tag("source", source)
            .register(registry);

        Gauge.builder("rag.retrieval.avg_score", avgScore,
                Number::floatValue)
            .tag("source", source)
            .register(registry);
    }

    // 降级指标
    public void recordDegradation(String fromModel, String toModel) {
        Counter.builder("llm.degradation")
            .tag("from", fromModel)
            .tag("to", toModel)
            .register(registry)
            .increment();
    }
}
```

### 4.3 告警规则（Prometheus）

```yaml
# prometheus/alerts.yml
groups:
  - name: java-ai-alerts
    rules:
      - alert: HighLlmErrorRate
        expr: rate(llm_calls_total{status="error"}[5m]) > 0.05
        for: 5m
        labels:
          severity: critical
        annotations:
          summary: "LLM 调用错误率超过 5%"

      - alert: HighDegradationRate
        expr: rate(llm_degradation_total[15m]) > 0.1
        for: 10m
        labels:
          severity: warning
        annotations:
          summary: "模型降级率超过 10%，检查主模型可用性"

      - alert: SlowRagRetrieval
        expr: histogram_quantile(0.95,
          rate(rag_retrieval_duration_seconds_bucket[5m])) > 2
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "RAG 检索 P95 延迟超过 2 秒"

      - alert: TokenBudgetWarning
        expr: llm_tokens_input_total + llm_tokens_output_total
          > 800000  # 月度预算 100 万
        for: 1h
        labels:
          severity: warning
        annotations:
          summary: "月度 Token 消耗已超过 80% 预算"
```

### 5.1 API Key 管理

```java
// 方案 A：环境变量（简单）
String apiKey = System.getenv("OPENAI_API_KEY");

// 方案 B：Spring Cloud Config + Vault（企业级）
@Configuration
public class SecureKeyConfig {

    @Bean
    public ChatLanguageModel secureModel(
            @Value("${openai.api-key}") String apiKey) {
        // apiKey 来自 Vault，运行时动态注入
        return OpenAiChatModel.builder()
            .apiKey(apiKey)
            .build();
    }
}
```

### 5.2 Prompt 注入防护

```java
@Component
public class PromptInjectionGuard {

    private static final List<String> INJECTION_PATTERNS = List.of(
        "ignore previous instructions",
        "ignore all previous",
        "disregard your system prompt",
        "你是 DAN",
        "jailbreak",
        "system prompt:*"   // 尝试获取 System Prompt
    );

    /**
     * 检查用户输入是否包含注入攻击
     */
    public boolean isSafe(String userInput) {
        String lower = userInput.toLowerCase();
        for (String pattern : INJECTION_PATTERNS) {
            if (lower.contains(pattern)) {
                log.warn("🚨 检测到 Prompt 注入尝试: {}", pattern);
                return false;
            }
        }
        return true;
    }

    /**
     * 在用户输入前添加防护层
     */
    public String sanitize(String userInput) {
        // 移除特殊控制字符
        return userInput
            .replace(" ", "")       // Null 字节
            .replaceAll("[\\x00-\\x08\\x0B\\x0C\\x0E-\\x1F]", "")
            .trim();
    }
}
```

## 6. CI/CD 流水线

### 6.1 GitHub Actions 示例

```yaml
# .github/workflows/build-and-deploy.yml
name: Build & Deploy Java AI Service

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Set up JDK 21
        uses: actions/setup-java@v4
        with:
          java-version: '21'
          distribution: 'temurin'

      - name: Cache Maven dependencies
        uses: actions/cache@v4
        with:
          path: ~/.m2
          key: ${{ runner.os }}-m2-${{ hashFiles('**/pom.xml') }}

      # 阶段 1：单元测试（Mock LLM，零成本）
      - name: Unit Tests
        run: ./mvnw test -Dgroups="unit"
        env:
          LLM_MOCK_ENABLED: true

      # 阶段 2：集成测试（可选真实 LLM）
      - name: Integration Tests
        if: github.event_name == 'pull_request'
        run: ./mvnw verify -Dgroups="integration"
        env:
          OPENAI_API_KEY: ${{ secrets.OPENAI_API_KEY }}

      # 阶段 3：GraalVM 原生镜像构建
      - name: Build Native Image
        if: github.ref == 'refs/heads/main'
        run: ./mvnw -Pnative native:compile

      # 阶段 4：Docker 镜像构建与推送
      - name: Build & Push Docker Image
        if: github.ref == 'refs/heads/main'
        run: |
          docker build -f Dockerfile.native \
            -t ghcr.io/${{ github.repository }}:${{ github.sha }} .
          docker push ghcr.io/${{ github.repository }}:${{ github.sha }}

      # 阶段 5：部署到 K8s
      - name: Deploy to Kubernetes
        if: github.ref == 'refs/heads/main'
        run: |
          kubectl set image deployment/java-ai-service \
            ai-service=ghcr.io/${{ github.repository }}:${{ github.sha }}
          kubectl rollout status deployment/java-ai-service
```

## 7. 性能基准与调优

### 7.1 关键性能指标

| 指标 | 目标值 | 优化策略 |
|------|:---:|------|
| **首 Token 延迟（TTFT）** | < 1s | 使用流式响应、Prompt Cache |
| **完整响应延迟** | < 5s | 合理设置 maxTokens、模型选择 |
| **RAG 检索延迟** | < 200ms | 混合检索 + HNSW 索引 + 连接池 |
| **向量化延迟** | < 100ms | 本地部署 Embedding 模型（Ollama） |
| **缓存命中率** | > 60% | Redis 缓存 + 语义去重 |
| **错误率** | < 1% | 降级链 + 重试 + 熔断 |

### 7.2 常见性能瓶颈与解决

| 瓶颈 | 现象 | 解决 |
|------|------|------|
| **PGVector 慢查询** | 检索 > 500ms | 建 HNSW 索引 + REINDEX + ef_search 调优 |
| **连接池耗尽** | 高峰期超时 | 增大 HikariCP max-pool-size 到 20 |
| **大文档向量化慢** | 摄入 10min+ | 批量向量化 + 异步 ES 写入 |
| **OOM** | Native 镜像 crash | 减小 maxTokens + 分片大小 |
| **冷启动慢** | Spring Boot 启动 8s+ | GraalVM Native Image 或 Quarkus |
| **跨 Region 延迟** | API 延迟 2s+ | 就近选择 API 区域 / 本地部署 Ollama |

### 7.3 JVM 调优参数

```bash
# 标准 JVM（非 Native）生产推荐参数
java \
  -Xms512m -Xmx1024m \
  -XX:+UseZGC \                          # ZGC 低延迟 GC
  -XX:MaxGCPauseMillis=50 \              # GC 暂停目标 50ms
  -XX:+ZGenerational \                   # 分代 ZGC（Java 21+）
  -XX:ConcGCThreads=2 \
  -Djava.net.preferIPv4Stack=true \      # 容器环境优化
  -jar ai-application.jar
```

> 🎯 **核心要点**：Java AI 应用的企业级部署核心决策是 JVM vs GraalVM Native。90% 场景建议先用标准 JVM 部署（调试方便、生态兼容好），在确实遇到冷启动瓶颈或 Serverless 场景时再考虑 Native 方案。可观测性是 Day One 就要做的事情——LLM 调用的 Token 消耗、延迟和降级率必须实时可见。

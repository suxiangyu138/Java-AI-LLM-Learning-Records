## 1. 控制流的四种基本模式

```text
模式 1：线性（Linear）
  A → B → C → D
  Chain 就能做，LangGraph 用普通边

模式 2：分支（Branching）
       ┌─→ B ─┐
  A ───┤       ├──→ D
       └─→ C ─┘
  根据条件从 A 到 B 或 C

模式 3：循环（Looping）
  A → B → C → 判断 → 回 A
  Chain 做不到！LangGraph 专属

模式 4：并行（Parallelism）
       ┌─→ B ─┐
  A ───┼─→ C ─┼──→ E
       └─→ D ─┘
  同时执行 B、C、D
```

## 2. Conditional Edge 深度指南

### 2.1 基本语法

```python
from langgraph.graph import StateGraph, END

# 路由函数签名：State → str
# 返回值：下一个 Node 名称 或 END 或 "__end__"
def router(state: AgentState) -> str:
    if state.get("error"):
        return "handle_error"
    if state.get("needs_verification"):
        return "verify"
    return END

# 注册条件边
graph.add_conditional_edges(
    "source_node",      # 从哪个 Node 出发
    router,             # 路由函数
    {                   # 路由表（可选，文档作用）
        "handle_error": "handle_error",
        "verify": "verify",
        END: END
    }
)
```

### 2.2 多条件路由

```python
def complex_router(state: AgentState) -> str:
    """模型响应后的路由决策"""
    last_msg = state["messages"][-1]
    step = state.get("current_step", "")

    # 优先级 1：错误处理
    if state.get("error_occurred"):
        return "error_handler"

    # 优先级 2：工具调用
    if hasattr(last_msg, "tool_calls") and last_msg.tool_calls:
        # 进一步区分工具类型
        tool_name = last_msg.tool_calls[0]["name"]
        if tool_name == "search":
            return "search_tool"
        elif tool_name == "execute_code":
            return "code_sandbox"
        else:
            return "generic_tools"

    # 优先级 3：步骤路由
    if step == "collecting_info":
        return "analyze_info"

    # 优先级 4：默认结束
    return END
```

### 2.3 条件路由最佳实践

```python
# ✅ 好：路由函数纯函数，只读 State
def good_router(state):
    return "tools" if state["messages"][-1].tool_calls else END

# ❌ 坏：路由函数有副作用
def bad_router(state):
    log_to_db(state)           # 副作用！
    state["_counter"] += 1     # 修改 State！
    update_external_cache()    # 外部调用！
    return "next"

# ✅ 好：复杂判断拆分为多个小函数
def should_use_tools(state):
    last = state["messages"][-1]
    return hasattr(last, "tool_calls") and last.tool_calls

def is_conversation_done(state):
    return state.get("turn_count", 0) >= state.get("max_turns", 10)

def master_router(state):
    if is_conversation_done(state):
        return END
    if should_use_tools(state):
        return "tools"
    return END
```

## 3. Agentic Loop：ReAct 循环实现

### 3.1 什么是 Agentic Loop

```text
Agentic Loop = LLM 推理 → 工具调用 → 观察结果 → 继续推理 → ...
直到 LLM 认为任务完成（不调用工具了）

循环结构：
        ┌──────────────────────┐
        ▼                      │
   ┌────────┐     ┌─────────┐  │
   │  LLM   │────▷│  Tools  │──┘
   │ 思考   │     │ 执行    │
   └────────┘     └─────────┘
        │
        │ finish_reason="stop"
        ▼
      END
```

### 3.2 完整实现

```python
from langgraph.graph import StateGraph, START, END
from langgraph.prebuilt import ToolNode
from typing import TypedDict, Annotated
import operator

# ① State
class AgentState(TypedDict):
    messages: Annotated[list, operator.add]
    iteration_count: int  # 安全阀：防止无限循环

# ② LLM 节点
def call_model(state: AgentState) -> dict:
    """调用 LLM，返回 AI 消息（可能含 tool_calls）"""
    response = llm_with_tools.invoke(state["messages"])
    return {
        "messages": [response],
        "iteration_count": state.get("iteration_count", 0) + 1
    }

# ③ 工具节点
tool_node = ToolNode(tools=[get_weather, search_web])

# ④ 路由函数：决定继续还是结束
def should_continue(state: AgentState) -> str:
    messages = state["messages"]
    last_message = messages[-1]

    # 安全检查：防止无限循环
    if state.get("iteration_count", 0) > 15:
        return END

    # 如果有 tool_calls → 去执行工具
    if hasattr(last_message, "tool_calls") and last_message.tool_calls:
        return "tools"

    # 否则 → 结束
    return END

# ⑤ 构建图
builder = StateGraph(AgentState)
builder.add_node("agent", call_model)
builder.add_node("tools", tool_node)

builder.add_edge(START, "agent")
builder.add_conditional_edges("agent", should_continue, {
    "tools": "tools",
    END: END
})
# 关键：tools → agent（回到 LLM），形成循环！
builder.add_edge("tools", "agent")

graph = builder.compile()

# ⑥ 使用
result = graph.invoke({
    "messages": [HumanMessage(content="北京今天天气怎么样？")],
    "iteration_count": 0
})
```

### 3.3 Prebuilt ReAct Agent（开箱即用）

```python
from langgraph.prebuilt import create_react_agent

# 一行代码创建 ReAct Agent
agent = create_react_agent(
    model=llm,
    tools=[get_weather, search_web, calculator],
    # 等价于手写上面的 ②③④⑤
)

# 使用
result = agent.invoke({
    "messages": [HumanMessage(content="北京今天天气怎么样？")]
})

# 可视化图结构
print(agent.get_graph().draw_ascii())
# 输出：
# START → agent → [conditional] → tools → agent → ... → END
```

### 3.4 循环安全阀

```python
# 方式 1：最大迭代次数（最简单）
def should_continue(state):
    if state["iteration_count"] > MAX_ITERATIONS:
        return "summarize_and_exit"  # 优雅结束
    if has_tool_calls(state):
        return "tools"
    return END

# 方式 2：最大执行时间
import time

class AgentState(TypedDict):
    messages: Annotated[list, operator.add]
    start_time: float
    max_seconds: float

def should_continue(state):
    elapsed = time.time() - state["start_time"]
    if elapsed > state["max_seconds"]:
        return "timeout_handler"
    return route_normal(state)

# 方式 3：Token 预算
def should_continue(state):
    total_tokens = sum(
        getattr(m, "usage_metadata", {}).get("total_tokens", 0)
        for m in state["messages"]
        if hasattr(m, "usage_metadata")
    )
    if total_tokens > state["token_budget"]:
        return "budget_exceeded_handler"
    return route_normal(state)
```

## 4. 并行执行：Fan-out / Fan-in

### 4.1 Send API

```python
from langgraph.graph import Send

class ParallelState(TypedDict):
    tasks: list[str]                    # 待处理任务
    results: Annotated[list, operator.add]  # 合并结果
    completed_count: int

# 分发器：为每个任务创建一个 Send
def dispatcher(state: ParallelState) -> list[Send]:
    tasks = state["tasks"]
    # 每个 Send → 一个独立的 worker 节点实例
    return [
        Send("worker", {"current_task": task, "task_index": i})
        for i, task in enumerate(tasks)
    ]

# Worker：处理单个任务
def worker(state: ParallelState) -> dict:
    task = state["current_task"]
    result = process_task(task)  # 实际处理逻辑
    return {"results": [result]}

# 构建
builder = StateGraph(ParallelState)
builder.add_node("dispatcher", dispatcher)
builder.add_node("worker", worker)
builder.add_node("aggregator", aggregator)  # 汇总结果

builder.add_edge(START, "dispatcher")
builder.add_conditional_edges(
    "dispatcher",
    dispatcher,  # 返回 Send 列表 → 自动并行
    {"worker": "worker"}
)
builder.add_edge("worker", "aggregator")
builder.add_edge("aggregator", END)
```

### 4.2 并行执行的执行模型

```text
dispatcher 返回 3 个 Send → 3 个 worker 同时启动

时间 →
dispatcher ████
              │
worker_1     ████████ (task A)
worker_2     ████████████████ (task B)  ← 最慢
worker_3     ██████ (task C)
                        │
aggregator              └────── ████
                                  │
所有 worker 完成后，aggregator 收到合并的 State
```

### 4.3 Map-Reduce 模式

```python
class MapReduceState(TypedDict):
    documents: list[str]
    summaries: Annotated[list, operator.add]
    final_summary: str

# Map 阶段：并行生成摘要
def map_to_summaries(state):
    return [
        Send("summarizer", {"doc": doc})
        for doc in state["documents"]
    ]

def summarizer(state):
    summary = llm.invoke(f"请总结：{state['doc']}")
    return {"summaries": [summary.content]}

# Reduce 阶段：汇总所有摘要
def reduce_summaries(state):
    all_summaries = "\n\n".join(state["summaries"])
    final = llm.invoke(f"综合以下摘要：\n{all_summaries}")
    return {"final_summary": final.content}
```

## 5. Subgraph：图的组合与复用

### 5.1 基本用法

```python
# ① 定义子图（独立的 StateGraph）
def build_sub_agent():
    sub_builder = StateGraph(SubState)
    sub_builder.add_node("analyze", analyze_func)
    sub_builder.add_node("generate", generate_func)
    # ... 子图的边
    return sub_builder.compile()

# ② 将子图作为主图的一个节点
main_builder = StateGraph(MainState)
main_builder.add_node("sub_agent", build_sub_agent())
#                                        ↑
#                              直接传入编译好的 Graph

# ③ 子图和主图共享同一个 State
# Subgraph 只能读取和修改 MainState 中定义的字段
```

### 5.2 Subgraph 的 State 映射

```python
# 场景：子图有自己的 State Schema
class SubState(TypedDict):
    internal_messages: Annotated[list, operator.add]
    analysis_result: str

class MainState(TypedDict):
    messages: Annotated[list, operator.add]
    analysis_result: str  # 与 SubState 同名字段

# LangGraph 自动映射同名字段：
#   SubState.internal_messages ←→ MainState.messages ❌（不同名，不映射）
#   SubState.analysis_result   ←→ MainState.analysis_result ✅（同名，自动映射）
```

### 5.3 实际案例：多步骤 Pipeline

```python
# 子图 1：意图识别
def build_intent_agent():
    # graph that identifies user intent
    ...

# 子图 2：代码生成
def build_code_agent():
    # graph that generates code
    ...

# 子图 3：代码审查
def build_review_agent():
    # graph that reviews code
    ...

# 主管道
main = StateGraph(MainState)
main.add_node("intent", build_intent_agent())
main.add_node("code_gen", build_code_agent())
main.add_node("review", build_review_agent())

main.add_edge(START, "intent")
main.add_conditional_edges("intent", route_by_intent, {
    "code_gen": "code_gen",
    END: END
})
main.add_edge("code_gen", "review")
main.add_edge("review", END)
```

## 6. 高级控制流模式

### 6.1 错误恢复循环

```python
class ErrorRecoveryState(TypedDict):
    messages: Annotated[list, operator.add]
    retry_count: int
    max_retries: int

def call_with_retry(state):
    try:
        result = fragile_operation(state)
        return {"messages": [AIMessage(content=result)]}
    except Exception as e:
        return {
            "messages": [AIMessage(content=f"错误: {e}")],
            "retry_count": state["retry_count"] + 1
        }

def should_retry(state) -> str:
    if state["retry_count"] < state["max_retries"]:
        return "retry_handler"  # 去修复 → 重试
    return "fallback_handler"   # 放弃 → 降级

# 图结构：LLM → [尝试] → 失败 → 修复 → 重试 → 成功 → END
#                          ↓
#                        放弃 → 降级 → END
```

### 6.2 流水线 + 回流（Pipeline with Backflow）

```text
设计中常见：审核不通过 → 回到生成阶段

         ┌────── 通过 ────────→ END
生成 → 审核
   ↑      │
   └── 驳回（带反馈）
```

```python
def review_and_route(state) -> str:
    if state["review_passed"]:
        return END
    if state["revision_count"] < 3:
        return "generator"  # 带反馈回去重做
    return "manual_review"  # 太多次，人工介入

graph.add_conditional_edges("reviewer", review_and_route, {
    "generator": "generator",
    "manual_review": "manual_review",
    END: END
})
graph.add_edge("generator", "reviewer")  # 生成 → 审核
# 形成了：generator → reviewer → generator 的循环
```

### 6.3 动态图修改

```python
# LangGraph 0.2+ 支持在运行时动态修改图
# 场景：根据工具调用的结果，插入额外的处理步骤

def dynamic_router(state):
    tool_result = state.get("latest_tool_result", "")

    # 根据工具结果决定下一步
    if "需要更多信息" in tool_result:
        # 动态插入信息收集步骤
        return "collect_more_info"
    elif "任务完成" in tool_result:
        return END
    elif "发生错误" in tool_result:
        # 动态插入错误恢复步骤
        return "error_recovery"
    else:
        return "continue_processing"
```

> 🎯 **核心要点**：LangGraph 的核心优势是**循环 + 条件分支**的任意组合。Agentic Loop（LLM ↔ Tool 循环）是最高频的模式，用 `create_react_agent` 可以一行代码搞定。但理解底层 `add_conditional_edges` + `add_edge("tools", "agent")` 的原理，才能定制出真正符合业务需求的 Agent。

## 1. 多 Provider 切换架构

### 1.1 SDK 层抽象

**方案一：工厂模式**（适合中小项目）

```python
from anthropic import Anthropic
from dataclasses import dataclass
from typing import Optional

@dataclass
class ProviderConfig:
    name: str
    base_url: str
    api_key: str
    models: list[str]
    weight: int = 1  # 负载均衡权重

class AIProviderFactory:
    def __init__(self):
        self.providers: dict[str, ProviderConfig] = {}
        self.clients: dict[str, Anthropic] = {}
    
    def register(self, config: ProviderConfig):
        self.providers[config.name] = config
        self.clients[config.name] = Anthropic(
            base_url=config.base_url,
            api_key=config.api_key
        )
    
    def get_client(self, model: str) -> tuple[Anthropic, ProviderConfig]:
        """根据模型选择对应的 Provider"""
        for name, config in self.providers.items():
            if model in config.models:
                return self.clients[name], config
        raise ValueError(f"No provider found for model: {model}")
    
    def get_client_round_robin(self) -> tuple[Anthropic, ProviderConfig]:
        """轮询选择 Provider"""
        import itertools
        from random import choices
        
        names, weights = zip(*[
            (n, p.weight) for n, p in self.providers.items()
        ])
        chosen = choices(names, weights=weights, k=1)[0]
        return self.clients[chosen], self.providers[chosen]

# 使用示例
factory = AIProviderFactory()
factory.register(ProviderConfig(
    name="deepseek",
    base_url="https://api.deepseek.com/beta",
    api_key="sk-ds-xxx",
    models=["deepseek-chat", "deepseek-reasoner"],
    weight=5
))
factory.register(ProviderConfig(
    name="siliconflow",
    base_url="https://api.siliconflow.cn/v1",
    api_key="sk-sf-xxx",
    models=["Qwen2.5-72B"],
    weight=3
))
```

**动态切换实现**：

```python
import os
from anthropic import Anthropic

class DynamicAIProvider:
    def __init__(self):
        self._current_provider = os.getenv("AI_PROVIDER", "deepseek")
        self._init_clients()
    
    def _init_clients(self):
        """初始化所有 Provider 的客户端"""
        self._clients = {
            "deepseek": Anthropic(
                base_url=os.getenv("DEEPSEEK_BASE_URL"),
                api_key=os.getenv("DEEPSEEK_API_KEY")
            ),
            "anthropic": Anthropic(
                base_url=os.getenv("ANTHROPIC_BASE_URL"),
                api_key=os.getenv("ANTHROPIC_API_KEY")
            ),
        }
    
    @property
    def client(self) -> Anthropic:
        return self._clients[self._current_provider]
    
    @property
    def model(self) -> str:
        return os.getenv(f"{self._current_provider.upper()}_MODEL")
    
    def switch(self, provider: str):
        """运行时切换 Provider"""
        if provider in self._clients:
            self._current_provider = provider
        else:
            raise ValueError(f"Unknown provider: {provider}")
    
    def chat(self, messages: list[dict], **kwargs):
        return self.client.messages.create(
            model=self.model,
            messages=messages,
            **kwargs
        )
```

## 2. 成本优化策略

### 2.1 按任务选择模型

| 任务类型 | 推荐模型 | 理由 |
|---------|---------|------|
| 简单对话、翻译 | Qwen2.5-72B / 硅基流动 | 成本低，延迟适中 |
| 代码生成、复杂推理 | DeepSeek-chat / Claude Sonnet | 性价比高 |
| 数学、逻辑推理 | DeepSeek-reasoner / o系列 | 擅长推理任务 |
| 多模态分析 | DeepSeek-V4 / Gemini | 支持图片输入 |
| 低成本大批量 | 开源模型（硅基流动） | 通常最低价 |

### 2.2 成本计算公式

```text
单次调用成本 = 输入 Token 数 × 输入单价 / 1M + 输出 Token 数 × 输出单价 / 1M
```

**参考价格对比**（2026.07）：

| Anthropic | Claude Sonnet 5 | ~$3 | ~$15 |
| DeepSeek | deepseek-chat | ~¥0.5 | ~¥2 |

| 硅基流动 | Qwen2.5-72B | ~¥1 | ~¥1 |
| 硅基流动 | Llama-3.1-70B | ~¥1.5 | ~¥1.5 |

## 3. 故障转移与高可用

### 3.1 重试策略

```python
import time
from functools import wraps

def retry_with_fallback(max_retries=3, fallback_providers=None):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_error = None
            providers = ['primary'] + (fallback_providers or [])
            
            for attempt, provider in enumerate(providers):
                try:
                    if attempt > 0:
                        # 切换到备用 Provider
                        kwargs['provider_override'] = provider
                        print(f"Failing over to {provider}")
                    return func(*args, **kwargs)
                except Exception as e:
                    last_error = e
                    wait = min(2 ** attempt, 10)  # 指数退避
                    time.sleep(wait)
            
            raise last_error
        return wrapper
    return decorator
```

### 3.2 熔断机制

```python
import time
from collections import deque

class CircuitBreaker:
    """简单的熔断器实现"""
    
    STATE_CLOSED = "closed"      # 正常
    STATE_OPEN = "open"          # 熔断
    STATE_HALF_OPEN = "half_open"  # 半开
    
    def __init__(self, failure_threshold=5, recovery_timeout=30):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.failures = deque(maxlen=failure_threshold)
        self.state = self.STATE_CLOSED
        self.last_failure_time = 0
    
    def call(self, func, *args, **kwargs):
        if self.state == self.STATE_OPEN:
            if time.time() - self.last_failure_time > self.recovery_timeout:
                self.state = self.STATE_HALF_OPEN
            else:
                raise Exception("Circuit breaker is OPEN")
        
        try:
            result = func(*args, **kwargs)
            if self.state == self.STATE_HALF_OPEN:
                self.state = self.STATE_CLOSED
                self.failures.clear()
            return result
        except Exception as e:
            self.failures.append(time.time())
            self.last_failure_time = time.time()
            if len(self.failures) >= self.failure_threshold:
                self.state = self.STATE_OPEN
            raise e
```

### 4.2 结构化日志

```python
import logging
import json

logger = logging.getLogger("ai_provider")

def log_api_call(provider, model, latency_ms, input_tokens, output_tokens, status):
    logger.info(json.dumps({
        "event": "api_call",
        "provider": provider,
        "model": model,
        "latency_ms": latency_ms,
        "input_tokens": input_tokens,
        "output_tokens": output_tokens,
        "status": status,
        "timestamp": time.time(),
    }))

# 使用装饰器自动记录
def monitor(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        try:
            result = func(*args, **kwargs)
            latency = int((time.time() - start) * 1000)
            usage = result.usage
            log_api_call(
                provider=kwargs.get("provider", "unknown"),
                model=kwargs.get("model", "unknown"),
                latency_ms=latency,
                input_tokens=usage.input_tokens if usage else 0,
                output_tokens=usage.output_tokens if usage else 0,
                status="success"
            )
            return result
        except Exception as e:
            latency = int((time.time() - start) * 1000)
            log_api_call(
                provider=kwargs.get("provider", "unknown"),
                model=kwargs.get("model", "unknown"),
                latency_ms=latency,
                input_tokens=0,
                output_tokens=0,
                status=f"error: {type(e).__name__}"
            )
            raise
    return wrapper
```

## 5. 安全最佳实践

### 5.1 API Key 管理

| 实践 | 说明 |
|------|------|
| 环境变量 | 禁止硬编码，使用 `.env` 或密钥管理服务 |

### 5.2 请求验证

```python
def validate_messages(messages: list[dict]) -> bool:
    """验证消息格式，防止注入"""
    allowed_roles = {"user", "assistant"}
    max_content_length = 100000  # 100k chars
    
    for msg in messages:
        if msg.get("role") not in allowed_roles:
            return False
        content = msg.get("content", "")
        if isinstance(content, str) and len(content) > max_content_length:
            return False
        # 检查系统指令注入
        if isinstance(content, str) and "ignore previous instructions" in content.lower():
            logger.warning(f"Potential prompt injection detected")
    return True
```

### 5.3 速率限制

```python
import asyncio
from collections import defaultdict
import time

class RateLimiter:
    def __init__(self, max_rps: int = 10):
        self.max_rps = max_rps
        self.timestamps = defaultdict(list)
    
    async def acquire(self, provider: str):
        now = time.time()
        self.timestamps[provider] = [
            t for t in self.timestamps[provider]
            if now - t < 1.0  # 1秒窗口
        ]
        
        if len(self.timestamps[provider]) >= self.max_rps:
            sleep_time = 1.0 - (now - self.timestamps[provider][0])
            if sleep_time > 0:
                await asyncio.sleep(sleep_time)
        
        self.timestamps[provider].append(time.time())
```

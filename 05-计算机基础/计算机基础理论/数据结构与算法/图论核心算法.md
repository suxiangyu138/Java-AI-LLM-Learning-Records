一、图的存储模板
1. 邻接矩阵
    适用场景：稠密图，顶点数少
    时间复杂度：查询边O(1)，遍历邻接点O(n)
```java
// 无向带权图，初始化INF表示无边
public class GraphMatrix {
    static final int INF = 0x3f3f3f3f;
    int[][] graph;
    int n; // 顶点数
    public GraphMatrix(int n) {
        this.n = n;
        graph = new int[n+1][n+1]; // 顶点编号从1开始
        // 初始化
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (i == j) graph[i][j] = 0;
                else graph[i][j] = INF;
            }
        }
    }
    // 添加无向边
    public void addEdge(int u, int v, int w) {
        graph[u][v] = w;
        graph[v][u] = w;
    }
}
```

```java
}
```

3. 链式前向星
    适用场景：高效存储，适合竞赛
```java
// 链式前向星
public class GraphForwardStar {
    static class Edge {
        int to;
        int weight;
        int next; // 下一条边的索引
        Edge(int to, int weight, int next) {
            this.to = to;
            this.weight = weight;
            this.next = next;
        }
    }
    Edge[] edges;
    int[] head; // head[u]表示u的第一条边索引
    int idx; // 边的计数器
    int n, m; // 顶点数、边数
    public GraphForwardStar(int n, int m) {
        this.n = n;
        this.m = m;
        edges = new Edge[m*2]; // 无向边乘2
        head = new int[n+1];
        // 初始化head为-1
        for (int i = 1; i <= n; i++) {
            head[i] = -1;
        }
        idx = 0;
    }
    // 添加无向边
    public void addEdge(int u, int v, int w) {
        edges[idx] = new Edge(v, w, head[u]);
        head[u] = idx++;
        edges[idx] = new Edge(u, w, head[v]);
        head[v] = idx++;
    }
}
```

二、遍历算法
1. 深度优先搜索(DFS)
```java
// 邻接表版DFS，标记访问过的顶点
import java.util.ArrayList;
import java.util.List;
public class DFS {
    static class Edge {
        int to;
        Edge(int to) {
            this.to = to;
        }
    }
    static List<List<Edge>> adj;
    static boolean[] visited;
    public static void dfs(int u) {
        visited[u] = true;
        // 处理当前顶点u
        System.out.print(u + " ");
        // 遍历邻接点
        for (Edge e : adj.get(u)) {
            int v = e.to;
            if (!visited[v]) {
                dfs(v);
            }
        }
    }
    public static void main(String[] args) {
        int n = 5; // 顶点数1-5
        adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        // 建边
        adj.get(1).add(new Edge(2));
        adj.get(1).add(new Edge(3));
        adj.get(2).add(new Edge(4));
        adj.get(3).add(new Edge(5));
        visited = new boolean[n+1];
        dfs(1); // 从顶点1开始遍历
    }
}
```

2. 广度优先搜索(BFS)
```java
// 邻接表版BFS，求无权图最短路径
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
public class BFS {
    static class Edge {
        int to;
        Edge(int to) {
            this.to = to;
        }
    }
    static List<List<Edge>> adj;
    static boolean[] visited;
    static int[] dist; // 距离起点的距离
    public static void bfs(int start) {
        Queue<Integer> queue = new LinkedList<>();
        visited[start] = true;
        dist[start] = 0;
        queue.offer(start);
        while (!queue.isEmpty()) {
            int u = queue.poll();
            // 处理当前顶点u
            System.out.print(u + " ");
            for (Edge e : adj.get(u)) {
                int v = e.to;
                if (!visited[v]) {
                    visited[v] = true;
                    dist[v] = dist[u] + 1;
                    queue.offer(v);
                }
            }
        }
    }
    public static void main(String[] args) {
        int n = 5;
        adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        // 建边
        adj.get(1).add(new Edge(2));
        adj.get(1).add(new Edge(3));
        adj.get(2).add(new Edge(4));
        adj.get(3).add(new Edge(5));
        visited = new boolean[n+1];
        dist = new int[n+1];
        bfs(1);
    }
}
```

三、最短路径算法
1. Dijkstra(堆优化版)
```java
// 邻接表版，求单源最短路径（无负权边）
import java.util.*;
public class Dijkstra {
    static class Edge {
        int to;
        int weight;
        Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }
    static final int INF = 0x3f3f3f3f;
    static List<List<Edge>> adj;
    static int[] dist;
    static boolean[] visited;
    public static void dijkstra(int n, int start) {
        dist = new int[n+1];
        visited = new boolean[n+1];
        Arrays.fill(dist, INF);
        dist[start] = 0;
        // 优先队列：(距离, 顶点)，小顶堆
        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(a -> a[0]));
        pq.offer(new int[]{0, start});
        while (!pq.isEmpty()) {
            int[] curr = pq.poll();
            int u = curr[1];
            if (visited[u]) continue;
            visited[u] = true;
            for (Edge e : adj.get(u)) {
                int v = e.to;
                int w = e.weight;
                if (dist[v] > dist[u] + w) {
                    dist[v] = dist[u] + w;
                    pq.offer(new int[]{dist[v], v});
                }
            }
        }
    }
    public static void main(String[] args) {
        int n = 5;
        adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        // 加边
        adj.get(1).add(new Edge(2, 2));
        adj.get(1).add(new Edge(3, 5));
        adj.get(2).add(new Edge(4, 3));
        adj.get(3).add(new Edge(5, 1));
        dijkstra(n, 1);
        // 输出1到各顶点的最短距离
        for (int i = 1; i <= n; i++) {
            System.out.println("1->" + i + ": " + dist[i]);
        }
    }
}
```
2. Floyd-Warshall
```java
// 邻接矩阵版，求任意两点最短路径
public class Floyd {
    static final int INF = 0x3f3f3f3f;
    static int[][] dist;
    static int n;
    public static void floyd(int[][] graph) {
        n = graph.length - 1;
        dist = new int[n+1][n+1];
        // 初始化
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                dist[i][j] = graph[i][j];
            }
        }
        // 动态规划：k为中间点
        for (int k = 1; k <= n; k++) {
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= n; j++) {
                    if (dist[i][k] + dist[k][j] < dist[i][j]) {
                        dist[i][j] = dist[i][k] + dist[k][j];
                    }
                }
            }
        }
    }
    public static void main(String[] args) {
        int n = 4;
        int[][] graph = new int[n+1][n+1];
        // 初始化
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (i == j) graph[i][j] = 0;
                else graph[i][j] = INF;
            }
        }
        // 加边
        graph[1][2] = 2;
        graph[1][3] = 6;
        graph[2][3] = 3;
        graph[3][4] = 1;
        floyd(graph);
        // 输出所有点对的最短路径
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                System.out.print(dist[i][j] + " ");
            }
            System.out.println();
        }
    }
}
```

四、最小生成树
1. Kruskal(并查集实现)
```java
// Kruskal算法：按边权排序，并查集判环
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
public class Kruskal {
    static class Edge implements Comparable<Edge> {
        int u, v, weight;
        Edge(int u, int v, int weight) {
            this.u = u;
            this.v = v;
            this.weight = weight;
        }
        @Override
        public int compareTo(Edge o) {
            return this.weight - o.weight; // 升序排序
        }
    }
    static int[] parent;
    static List<Edge> edges;
    // 并查集查找
    public static int find(int x) {
        if (parent[x] != x) {
            parent[x] = find(parent[x]); // 路径压缩
        }
        return parent[x];
    }
    // 并查集合并
    public static void union(int x, int y) {
        int fx = find(x);
        int fy = find(y);
        if (fx != fy) {
            parent[fy] = fx;
        }
    }
    public static int kruskal(int n) {
        // 初始化并查集
        parent = new int[n+1];
        for (int i = 1; i <= n; i++) {
            parent[i] = i;
        }
        // 边排序
        Collections.sort(edges);
        int sum = 0; // 最小生成树总权值
        int cnt = 0; // 已选边数
        for (Edge e : edges) {
            int u = e.u;
            int v = e.v;
            int w = e.weight;
            if (find(u) != find(v)) {
                union(u, v);
                sum += w;
                cnt++;
                if (cnt == n-1) break; // 选够n-1条边
            }
        }
        return sum;
    }
    public static void main(String[] args) {
        int n = 4; // 顶点数
        edges = new ArrayList<>();
        // 加边
        edges.add(new Edge(1, 2, 1));
        edges.add(new Edge(1, 3, 3));
        edges.add(new Edge(2, 3, 1));
        edges.add(new Edge(2, 4, 5));
        edges.add(new Edge(3, 4, 2));
        int mstSum = kruskal(n);
        System.out.println("最小生成树总权值：" + mstSum);
    }
}
```

2. Prim(堆优化版)
```java
// 邻接表版Prim算法，求最小生成树
import java.util.*;
public class Prim {
    static class Edge {
        int to;
        int weight;
        Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }
    static final int INF = 0x3f3f3f3f;
    static List<List<Edge>> adj;
    static boolean[] visited;
    static int[] dist; // dist[v]表示v到已选集合的最小边权
    public static int prim(int n) {
        visited = new boolean[n+1];
        dist = new int[n+1];
        Arrays.fill(dist, INF);
        dist[1] = 0; // 从顶点1开始
        int sum = 0; // 最小生成树总权值
        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(a -> a[0]));
        pq.offer(new int[]{0, 1});
        while (!pq.isEmpty()) {
            int[] curr = pq.poll();
            int u = curr[1];
            int w = curr[0];
            if (visited[u]) continue;
            visited[u] = true;
            sum += w;
            for (Edge e : adj.get(u)) {
                int v = e.to;
                int weight = e.weight;
                if (!visited[v] && weight < dist[v]) {
                    dist[v] = weight;
                    pq.offer(new int[]{dist[v], v});
                }
            }
        }
        return sum;
    }
    public static void main(String[] args) {
        int n = 4;
        adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        // 加边
        adj.get(1).add(new Edge(2, 1));
        adj.get(1).add(new Edge(3, 3));
        adj.get(2).add(new Edge(3, 1));
        adj.get(2).add(new Edge(4, 5));
        adj.get(3).add(new Edge(4, 2));
        int mstSum = prim(n);
        System.out.println("最小生成树总权值：" + mstSum);
    }
}
```

```java
// 邻接表版，入度表实现拓扑排序
import java.util.*;
public class TopologicalSort {
    static List<List<Integer>> adj;
    static int[] inDegree; // 入度
    public static List<Integer> topologicalSort(int n) {
        List<Integer> res = new ArrayList<>();
        Queue<Integer> queue = new LinkedList<>();
        // 入度为0的点入队
        for (int i = 1; i <= n; i++) {
            if (inDegree[i] == 0) {
                queue.offer(i);
            }
        }
        while (!queue.isEmpty()) {
            int u = queue.poll();
            res.add(u);
            // 遍历邻接点，入度减1
            for (int v : adj.get(u)) {
                inDegree[v]--;
                if (inDegree[v] == 0) {
                    queue.offer(v);
                }
            }
        }
        // 若结果长度不等于顶点数，说明有环
        if (res.size() != n) {
            return new ArrayList<>(); // 有环返回空
        }
        return res;
    }
    public static void main(String[] args) {
        int n = 5; // 顶点数1-5
        adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        inDegree = new int[n+1];
        // 建边：1->2, 1->3, 2->4, 3->4, 4->5
        adj.get(1).add(2); inDegree[2]++;
        adj.get(1).add(3); inDegree[3]++;
        adj.get(2).add(4); inDegree[4]++;
        adj.get(3).add(4); inDegree[4]++;
        adj.get(4).add(5); inDegree[5]++;
        List<Integer> res = topologicalSort(n);
        System.out.println(res); // 输出拓扑序
    }
}
```
